\chapter{計算に必要な特殊関数}
\label{ch:tb-functions}

$\sin,\cos,\tan$、指数・対数の基本公式は既知とし、
workbook で記号または置換を知らないと止まる箇所だけを扱う。

\section{\texorpdfstring{$\sec,\csc,\cot$}{sec, csc, cot}}

\begin{definition}{逆数三角関数}{tb-reciprocal-trig}
\[
\sec x=\frac1{\cos x},\qquad
\csc x=\frac1{\sin x},\qquad
\cot x=\frac{\cos x}{\sin x}=\frac1{\tan x}.
\]
$\sec x$ は $\cos x\ne0$、$\csc x,\cot x$ は $\sin x\ne0$ で定義される。
\end{definition}

\[
1+\tan^2x=\sec^2x,
\qquad
1+\cot^2x=\csc^2x.
\]
微分は
\[
(\tan x)'=\sec^2x,\qquad
(\sec x)'=\sec x\tan x,
\]
\[
(\cot x)'=-\csc^2x,\qquad
(\csc x)'=-\csc x\cot x,
\]
原始関数は
\[
\int\sec x\,\d x=\log\abs{\sec x+\tan x}+C,
\qquad
\int\csc x\,\d x=\log\abs{\csc x-\cot x}+C
\]
である。特に
\[
\frac{\d}{\d x}\log(\tan x+\sec x)=\sec x
\]
は、2026年度第2期の極座標積分でそのまま使う。

\section{逆三角関数}

\begin{definition}{標準的な逆三角関数}{tb-inverse-trig}
\[
\begin{array}{c|c|c}
 &\text{定義域}&\text{値域（主値）}\\ \hline
\arcsin x&[-1,1]&[-\pi/2,\pi/2]\\
\arccos x&[-1,1]&[0,\pi]\\
\arctan x&\R&(-\pi/2,\pi/2)
\end{array}
\]
\end{definition}

$\sin(\arcsin x)=x$ は常に成り立つが、
$\arcsin(\sin x)=x$ には $x\in[-\pi/2,\pi/2]$ が必要である。
主値から符号を決めれば
\[
\cos(\arcsin x)=\sqrt{1-x^2},
\qquad
\sin(\arccos x)=\sqrt{1-x^2}.
\]
また
\[
(\arcsin x)'=\frac1{\sqrt{1-x^2}},
\qquad
(\arccos x)'=-\frac1{\sqrt{1-x^2}},
\qquad
(\arctan x)'=\frac1{1+x^2}.
\]
従って $1/\sqrt{1-x^2}$ と $1/(1+x^2)$ の原始関数は、それぞれ
$\arcsin x$ と $\arctan x$ である。

\section{根号の有理化置換}

2015年度応用数理の積分では
\[
t=x+\sqrt{x^2+1}
\]
と置く。共役との積が1なので
\[
t^{-1}=\sqrt{x^2+1}-x,
\qquad
x=\frac{t-t^{-1}}2,
\qquad
\sqrt{x^2+1}=\frac{t+t^{-1}}2,
\qquad
\d x=\frac{1+t^{-2}}2\,\d t.
\]
従って
\[
\frac{\d x}{\sqrt{x^2+1}}=\frac{\d t}{t},
\qquad
\sqrt{x^2+1}\,\d x
=\frac14(t+2t^{-1}+t^{-3})\,\d t.
\]
$x=0,1$ はそれぞれ $t=1,1+\sqrt2$ に移る。
