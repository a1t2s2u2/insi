\begin{problem}{第1問｜不変部分空間}{2017-b-1}
$f:\R^3\to\R^3$ を
\[
f(x)=\begin{pmatrix}1&0&0\\1&3&-3\\-2&2&-4\end{pmatrix}x
\]
で定める。$f(W)\subset W$ となる1次元および2次元の部分空間 $W$ を全て求めよ。
\end{problem}
\begin{proof}[解答]
行列を $A$ と書く。まず特性多項式を第1行で展開すると
\begin{align*}
\det(\lambda I-A)
&=(\lambda-1)\det\begin{pmatrix}\lambda-3&3\\-2&\lambda+4\end{pmatrix}\\
&=(\lambda-1)\{(\lambda-3)(\lambda+4)+6\}\\
&=(\lambda-1)(\lambda-2)(\lambda+3).
\end{align*}
従って固有値は $1,2,-3$ で相異なる。各方程式 $(A-\lambda I)x=0$ を解くと、対応する固有ベクトルを
\[
v_1=(-4,11,6)^{\mathsf T},\quad v_2=(0,3,1)^{\mathsf T},\quad
v_3=(0,1,2)^{\mathsf T}
\]
と取れる。

1次元部分空間 $W=\langle w\rangle$ が不変なら $Aw\in\langle w\rangle$ だから、
$Aw=\lambda w$ となり、$w$ は固有ベクトルである。各固有空間は1次元なので、
1次元不変部分空間は
$\langle v_1\rangle,\langle v_2\rangle,\langle v_3\rangle$ に限る。

2次元不変部分空間 $W$ について、$A$ の最小多項式
$(t-1)(t-2)(t+3)$ は $A|_W$ も零化する。これは重根を持たないので
$A|_W$ は対角化可能であり、$W$ は $A$ の固有ベクトルからなる基底を持つ。
各固有空間は1次元だから、その基底は相異なる二つの固有値に属する。従って
\[
\langle v_1,v_2\rangle,\qquad
\langle v_1,v_3\rangle,\qquad
\langle v_2,v_3\rangle
\]
が全てである。実際、各空間は固有ベクトルで生成されるので $A$ で不変である。
\end{proof}

\begin{problem}{第2問｜近似恒等核}{2017-b-2}
$f$ は $[0,\infty)$ 上の有界連続関数とし
$I_n=\int_0^\infty n f(x)e^{-nx}\,\d x$ とおく。
\begin{enumerate}
  \item[(1)] $I_n$ が有限であることを示せ。
  \item[(2)] 任意の $\delta>0$ に対し $\displaystyle\int_\delta^\infty nf(x)e^{-nx}\,\d x\to0$ を示せ。
  \item[(3)] $I_n\to f(0)$ を示せ。
\end{enumerate}
\end{problem}
\begin{proof}[解答]
$|f(x)|\le M$ とする。(1) は
\[
\int_0^\infty|nf(x)e^{-nx}|\,\d x
\le M\int_0^\infty ne^{-nx}\,\d x
=M[-e^{-nx}]_0^\infty=M
\]
より絶対収束する。(2) も同様に
\[
\left|\int_\delta^\infty nf(x)e^{-nx}\,\d x\right|
\le M\int_\delta^\infty ne^{-nx}\,\d x
=Me^{-n\delta}\longrightarrow0.
\]

(3) $\int_0^\infty ne^{-nx}\,\d x=1$ より
\[
I_n-f(0)=\int_0^\infty n(f(x)-f(0))e^{-nx}\,\d x.
\]
任意の $\varepsilon>0$ を取る。$f$ の0での連続性から、ある $\delta>0$ があり
$0\le x\le\delta$ なら $|f(x)-f(0)|<\varepsilon/2$ となる。一方
$|f(x)-f(0)|\le2M$ なので、積分を $[0,\delta]$ と $[\delta,\infty)$ に分けると
\begin{align*}
|I_n-f(0)|
&\le\frac\varepsilon2\int_0^\delta ne^{-nx}\,\d x
+2M\int_\delta^\infty ne^{-nx}\,\d x\\
&\le\frac\varepsilon2+2Me^{-n\delta}.
\end{align*}
$n$ を十分大きくすれば右辺は $\varepsilon$ 未満になるので $I_n\to f(0)$ である。
\end{proof}

\begin{problem}{第3問｜有理数の部分空間}{2017-b-3}
$\Q$ に通常の距離を入れ，$A=\{x\in\Q\mid1\le x<2\}$ とする。$A$ は閉，コンパクト，連結か。
\end{problem}
\begin{proof}[解答]
閉でもコンパクトでも連結でもない。

まず $n\ge2$ に対して $x_n=2-1/n$ とおくと
\[
x_n\in A,\qquad d_{\Q}(x_n,2)=|x_n-2|=\frac1n\longrightarrow0
\]
である。一方、$2\in\Q$ であるが $2\notin A$ なので、$A$ は $\Q$ の閉集合ではない。
距離空間は Hausdorff 空間であり、そのコンパクト部分集合は閉集合である。
従って、閉集合でない $A$ はコンパクトでもない。

次に、無理数 $c\in(1,2)$ を一つ取り
\[
U=A\cap(-\infty,c),\qquad V=A\cap(c,\infty)
\]
とおく。ここで区間との共通部分は $\Q$ の相対位相で開であるから、
$U,V$ は $A$ の相対位相で開である。また有理数の稠密性より $U,V$ はともに空でなく、
$c\notin\Q$ だから
\[
A=U\sqcup V
\]
となる。よって $U,V$ は $A$ の非自明な開分離であり、$A$ は連結でない。
\end{proof}

\begin{problem}{第1問｜群の交換子}{2017-s-1}
$G$ を有限群，$K\subset H$ を部分群とし
$X=\{x\in G\mid[x,h]\in K\ (\forall h\in H)\}$，$[x,y]=x^{-1}y^{-1}xy$ とおく。
\begin{enumerate}
  \item[(1)] $x\in X$ なら $x^{-1}Kx=K$ を示せ。
  \item[(2)] $X$ が $G$ の部分群であることを示せ。
  \item[(3)] $K\triangleleft H$ かつ $H/K$ が可換であることと $H\subset X$ が同値であることを示せ。
\end{enumerate}
\end{problem}
\begin{proof}[解答]
(1) $k\in K\subset H$ とすると $[x,k]=x^{-1}k^{-1}xk\in K$ である。
従って $x^{-1}k^{-1}x=[x,k]k^{-1}\in K$、逆元を取って $x^{-1}kx\in K$。
よって $x^{-1}Kx\subset K$ であり、両者は有限集合で同じ位数を持つので等号となる。

(2) まず $1\in X$ である。$x,y\in X$, $h\in H$ に対し交換子公式から
\[
[xy,h]=y^{-1}[x,h]y[y,h].
\]
(1) より $y^{-1}Ky=K$ であり、$[x,h],[y,h]\in K$ だから右辺は $K$ に入る。
従って $xy\in X$。さらに
\[
[x^{-1},h]=\bigl(x[x,h]x^{-1}\bigr)^{-1}
\]
であり、$[x,h]\in K$、(1) より $xKx^{-1}=K$ なので $x^{-1}\in X$。
よって $X$ は部分群である。

(3) $H\subset X$ とする。(1) を $x\in H$ に適用すれば $K\triangleleft H$。
また全ての $h_1,h_2\in H$ について $[h_1,h_2]\in K$ だから、商群では
$(h_1K)(h_2K)=(h_2K)(h_1K)$ となり $H/K$ は可換である。
逆に $K\triangleleft H$ かつ $H/K$ が可換なら、任意の $h_1,h_2\in H$ の交換子は
$K$ に入る。従って各 $h_1\in H$ は $X$ に属し、$H\subset X$ となる。
\end{proof}

\begin{problem}{第2問｜剰余環と局所化}{2017-s-2}
$R$ を可換環，$a\in R$，$S=\{a^n\mid n\ge0\}$ とする。自然な写像 $\pi:R\to R/(a)$，$\tau:R\to S^{-1}R$ を考える。
\begin{enumerate}
  \item[(1)] $x\in\ker\pi\cap\ker\tau$ なら，ある $n$ で $x^n=0$ となることを示せ。
  \item[(2)] $\tau$ が全射なら $\ker\pi+\ker\tau=R$ を示せ。
\end{enumerate}
\end{problem}
\begin{proof}[解答]
\begin{enumerate}
  \item[(1)] $x\in\ker\pi$ であることは $x\in(a)$ を意味するので、ある $r\in R$ により
  \[
  x=ar
  \]
  と書ける。また $x\in\ker\tau$ だから $x/1=0/1$ である。
  局所化における分数の等号の定義より、ある $m\ge0$ が存在して
  \[
  a^m(x\cdot1-0\cdot1)=a^mx=0
  \]
  となる。従って
  \[
  x^{m+1}=(ar)^m x=a^m r^m x=r^m(a^mx)=0
  \]
  であり、$x$ は冪零である。

  \item[(2)] $\tau$ が全射なので、$S^{-1}R$ の元 $1/a$ に対して、ある $b\in R$ が存在し
  \[
  \frac b1=\frac1a
  \]
  となる。再び局所化における分数の等号の定義を使うと、ある $n\ge0$ に対して
  \[
  a^n(ab-1)=0
  \]
  である。従って
  \[
  \tau(ab-1)=\frac{ab-1}{1}=0,
  \qquad ab-1\in\ker\tau.
  \]
  一方、$ab\in(a)=\ker\pi$ であるから
  \[
  1=ab-(ab-1)\in\ker\pi+\ker\tau.
  \]
  二つの核の和は $R$ のイデアルであり、$1$ を含むので
  $\ker\pi+\ker\tau=R$ である。
\end{enumerate}
\end{proof}

\begin{problem}{第3問｜被覆空間と Betti 数}{2017-s-3}
\begin{enumerate}
  \item[(1)] $F:\R\to S^1$，$F(\theta)=(\cos\theta,\sin\theta)$ が普遍被覆であることを示し，被覆変換群を求めよ。
  \item[(2)] 四面体の境界のうち，三つの面からなる複体を $K_1$，残り一面を $K_2$ とする。$K_1\cap K_2,K_1,K_2,K_1\cup K_2$ の Betti 数を求めよ。
\end{enumerate}
\end{problem}
\begin{proof}[解答]
\begin{enumerate}
\item[(1)] $z_0=F(\theta_0)\in S^1$ を任意に取り、$0<\varepsilon<\pi$ とする。
円弧
\[
U=F((\theta_0-\varepsilon,\theta_0+\varepsilon))
\]
に対して
\[
F^{-1}(U)
=\bigsqcup_{k\in\Z}
(\theta_0-\varepsilon+2\pi k,\theta_0+\varepsilon+2\pi k)
\]
であり、各区間への $F$ の制限は $U$ への同相写像である。
従って $F$ は被覆写像である。さらに $\R$ は単連結なので、これは普遍被覆である。

$k\in\Z$ に対して
\[
T_k(\theta)=\theta+2\pi k
\]
とおけば $F\circ T_k=F$ なので、$T_k$ は被覆変換である。逆に被覆変換 $T$ に対して
\[
F(T(\theta))=F(\theta)
\quad\Longrightarrow\quad
T(\theta)-\theta\in2\pi\Z.
\]
左辺は $\theta$ の連続関数であり、$\R$ は連結だから、この整数値は一定である。
従って $T=T_k$ となる。合成について $T_kT_\ell=T_{k+\ell}$ なので、被覆変換群は $\Z$ と同型である。

\item[(2)] $K_2$ は三角形一枚なので $D^2$ と同相であり、$K_1$ も残りの三面を合わせた円板なので $D^2$ と同相である。
両者の共通部分は $K_2$ の境界であるから
\[
K_1\simeq D^2,\qquad K_2\simeq D^2,
\qquad K_1\cap K_2\simeq S^1,
\qquad K_1\cup K_2\simeq S^2.
\]
$D^2,S^1,S^2$ のホモロジー群から、Betti 数は
\[
\begin{array}{c|ccc}
&b_0&b_1&b_2\\\hline
K_1\cap K_2&1&1&0\\
K_1&1&0&0\\
K_2&1&0&0\\
K_1\cup K_2&1&0&1
\end{array}
\]
であり、高次の Betti 数はいずれも0である。
\end{enumerate}
\end{proof}

\begin{problem}{第4問｜管状曲面}{2017-s-4}
$0<a<1$ とし
\[
p(u,v)=(u,u^2/2,0)+\frac{a\cos v}{\sqrt{u^2+1}}(-u,1,0)+a\sin v(0,0,1).
\]
これが曲面であることを示し，Gauss 曲率・平均曲率を求めよ。また $q(s)=p(u_0,s/a)$ が測地線であることを示せ。
\end{problem}
\begin{proof}[解答]
中心曲線を $c(u)=(u,u^2/2,0)$ とし、
\[
w=\norm{c'(u)}=\sqrt{1+u^2},\quad
T=\frac{(1,u,0)}w,\quad
N=\frac{(-u,1,0)}w,\quad B=(0,0,1)
\]
とおく。中心曲線の曲率は
\[
\kappa=\frac{\norm{c'\times c''}}{\norm{c'}^3}
=\frac1{(1+u^2)^{3/2}}
\]
であり、Frenet の公式から $N_u=-\kappa wT$ である。問題の写像は
\[
p(u,v)=c(u)+a\cos v\,N(u)+a\sin v\,B
\]
と書けるので
\[
p_u=w(1-a\kappa\cos v)T,
\qquad p_v=a(-\sin v\,N+\cos v\,B).
\]
$0<a<1$ かつ $0<\kappa\le1$ だから $1-a\kappa\cos v\ge1-a>0$。
従って $p_u,p_v$ はともに0でなく互いに直交し、$p_u\times p_v\ne0$ なので正則な曲面である。

外向き単位法線を $\nu=\cos v\,N+\sin v\,B$ と取ると
\[
\nu_u=-\kappa w\cos v\,T,
\qquad \nu_v=-\sin v\,N+\cos v\,B=\frac1a p_v.
\]
形作用素 $S(X)=-\d\nu(X)$ を $p_u,p_v$ に作用させると
\[
S(p_u)=\frac{\kappa\cos v}{1-a\kappa\cos v}p_u,
\qquad S(p_v)=-\frac1a p_v.
\]
従って座標方向は主方向で、主曲率は
\[
k_1=\frac{\kappa\cos v}{1-a\kappa\cos v},\qquad k_2=-\frac1a.
\]
よって積と平均を取れば
\[
K=-\frac{\kappa\cos v}{a(1-a\kappa\cos v)},\qquad
H=\frac12\left(\frac{\kappa\cos v}{1-a\kappa\cos v}-\frac1a\right).
\]
$q(s)=p(u_0,s/a)$ について連鎖律から
\[
q'(s)=\frac1a p_v(u_0,s/a),\qquad
q''(s)=\frac1{a^2}p_{vv}(u_0,s/a).
\]
$\norm{p_v}=a$ なので $q$ は単位速度であり、
$p_{vv}=-a(\cos v\,N+\sin v\,B)=-a\nu$ だから
\[
q''(s)=-\frac1a\nu(u_0,s/a).
\]
加速度が曲面法線方向だけで接成分が0なので、$q$ は測地線である。
\end{proof}

\begin{problem}{第5問｜扇形上の正則関数}{2017-s-5}
$S=\{z\in\mathbb C\mid0<|z|<r,\ \alpha<\arg z<\beta\}$，$\beta<\alpha+2\pi$ とする。
\begin{enumerate}
  \item[(1)] $f(z)$ は $z\to0$ で極限を持つが $f'(z)$ は持たない例を挙げよ。
  \item[(2)] $0<r'<r$，$\alpha<\alpha'<\beta'<\beta$ で切り取った閉じた小扇形の各点を中心に，$S$ に含まれる円板があることを示せ。
  \item[(3)] $f$ が $S$ 上有界正則なら，境界から $\varepsilon$ だけ離した小扇形 $T_\varepsilon$ 上で $|f'(z)|\le M_\varepsilon/|z|$ となることを示せ。
\end{enumerate}
\end{problem}
\begin{proof}[解答]
\begin{enumerate}
\item[(1)] 扇形の角度は $2\pi$ 未満なので、$S$ 上で正則な対数の枝を取れる。
\[
f(z)=\exp\left(\frac12\log z\right)
\]
とおけば $|f(z)|=|z|^{1/2}\to0$ である。一方
\[
f'(z)=\frac12\exp\left(-\frac12\log z\right),\qquad
|f'(z)|=\frac1{2\sqrt{|z|}}\longrightarrow\infty,
\]
従って $f'$ は有限な極限を持たない。

\item[(2)] 切り取った小扇形の点 $z$ は $0<|z|<r$ かつ
$\alpha<\arg z<\beta$ を満たすので $z\in S$ である。$S$ は開集合だから、
開集合の定義により、各 $z$ に対してある $\rho_z>0$ が存在し
$B(z,\rho_z)\subset S$ となる。

\item[(3)] $T_\varepsilon$ では角度が二つの境界半直線から少なくとも
$\varepsilon$ 離れ、外周も $r'<r$ に切り取られているとする。
\[
0<c_\varepsilon<\min\left\{1,\sin\varepsilon,\frac r{r'}-1\right\}
\]
を取れば、任意の $z\in T_\varepsilon$ に対し
\[
B(z,c_\varepsilon|z|)\subset S.
\]
実際、半径が $|z|$ 未満なので円板は0を含まず、境界半直線までの距離
$|z|\sin\varepsilon$ より小さく、さらに円板内の $w$ は
$|w|<(1+c_\varepsilon)r'<r$ を満たす。

$|f|\le B$ とし、円 $|\zeta-z|=c_\varepsilon|z|$ 上で Cauchy の微分公式を使うと
\[
f'(z)=\frac1{2\pi i}\int_{|\zeta-z|=c_\varepsilon|z|}
\frac{f(\zeta)}{(\zeta-z)^2}\,\d\zeta.
\]
従って円周の長さも含めて評価すると
\[
|f'(z)|\le
\frac1{2\pi}\frac{B}{(c_\varepsilon|z|)^2}
2\pi c_\varepsilon|z|
=\frac{B}{c_\varepsilon|z|}.
\]
$M_\varepsilon=B/c_\varepsilon$ とおけば求める評価を得る。
\end{enumerate}
\end{proof}

\begin{problem}{第6問｜線形微分方程式}{2017-s-6}
\begin{enumerate}
  \item[(1)] $y''+2y'+2y=0$，$y(0)=1,y'(0)=0$ を解き，零点を全て求めよ。
  \item[(2)] $y''+2y=\cos(\omega x)$，$y(0)=0,y'(0)=1$ の解が $x>0$ で非有界となる $\omega$ と，その解を求めよ。
\end{enumerate}
\end{problem}
\begin{proof}[解答]
\begin{enumerate}
\item[(1)] 特性方程式は
\[
r^2+2r+2=0,\qquad r=-1\pm i
\]
なので一般解は $y=e^{-x}(C_1\cos x+C_2\sin x)$。
$y(0)=1$ から $C_1=1$、
\[
y'(0)=-C_1+C_2=0
\]
から $C_2=1$。従って
\[
y=e^{-x}(\cos x+\sin x),\qquad x=-\frac\pi4+k\pi\quad(k\in\Z).
\]
最後の零点は $e^{-x}\ne0$ と
$\cos x+\sin x=\sqrt2\sin(x+\pi/4)$ から得られる。
\item[(2)] $\omega^2\ne2$ なら特殊解 $\cos(\omega x)/(2-\omega^2)$ が取れ、
同次解も三角関数なので全ての解は有界である。
$\omega=\pm\sqrt2$ のときは共振し、$a=\sqrt2$ とおけば
\[
y_p=\frac{x}{2a}\sin ax
\]
と置く。このとき
\[
y_p'=\frac1{2a}\sin ax+\frac x2\cos ax,
\]
\[
y_p''=\frac12\cos ax+\frac12\cos ax-\frac{ax}{2}\sin ax
=\cos ax-\frac{ax}{2}\sin ax.
\]
従って
\[
y_p''+a^2y_p
=\cos ax-\frac{ax}{2}\sin ax+\frac{ax}{2}\sin ax
=\cos ax.
\]
初期条件を満たす同次解 $(1/a)\sin ax$ を加えて
\[
y(x)=\frac{x+2}{2\sqrt2}\sin(\sqrt2x).
\]
これは $x>0$ で非有界である。従って求める $\omega$ は $\omega=\pm\sqrt2$。
\end{enumerate}
\end{proof}

\begin{problem}{第7問｜パラメータ積分}{2017-s-7}
$f$ は $(0,\infty)$ 上可測で $\int_0^\infty x|f(x)|\,\d x<\infty$ を満たす。次を求めよ。
\[
\lim_{a\downarrow0}\frac1a\int_0^\infty(\log(1+ax)-ax)f(x)\,\d x.
\]
\end{problem}
\begin{proof}[解答]
$u=ax\ge0$ とおく。$a\downarrow0$ のとき $u\to0$ であり、
$\log(1+u)=u+o(u)$ だから、各 $x>0$ について
\[
\frac{\log(1+ax)-ax}{a}
=x\frac{\log(1+u)-u}{u}\longrightarrow0.
\]
また $\log(1+u)\le u$ であり、$\log(1+u)\ge0$ でもあるので
\[
0\le u-\log(1+u)\le u.
\]
従って
\[
\left|\frac{\log(1+ax)-ax}{a}f(x)\right|
\le x|f(x)|.
\]
仮定より右辺は $(0,\infty)$ 上可積分である。よって優収束定理により積分と極限を交換でき、求める極限は $0$ である。
\end{proof}

\begin{problem}{第8問｜確率収束}{2017-s-8}
実確率変数列 $X_n$ と実確率変数 $X$ について答えよ。
\begin{enumerate}
  \item[(1)] $\lim_{N\to\infty}P(|X|>N)=0$ を示せ。
  \item[(2)] $X_n\to X$ が確率収束し，$f:\R\to\R$ が連続なら $f(X_n)\to f(X)$ も確率収束することを示せ。
\end{enumerate}
\end{problem}
\begin{proof}[解答]
\begin{enumerate}
\item[(1)] $A_N=\{|X|>N\}$ とおくと
\[
A_1\supset A_2\supset\cdots,
\qquad \bigcap_{N=1}^{\infty}A_N=\varnothing
\]
である。確率測度の上からの連続性より
\[
\lim_{N\to\infty}P(|X|>N)
=\lim_{N\to\infty}P(A_N)
=P\left(\bigcap_{N=1}^{\infty}A_N\right)=0.
\]

\item[(2)] 任意の $\varepsilon>0$ と $\eta>0$ を取る。(1) より、十分大きな $M$ を取れば
\[
P(|X|>M)<\frac\eta2
\]
となる。$f$ はコンパクト区間 $[-M-1,M+1]$ 上で一様連続なので、ある $0<\delta\le1$ が存在して
\[
x,y\in[-M-1,M+1],\quad |x-y|<\delta
\quad\Longrightarrow\quad |f(x)-f(y)|<\varepsilon
\]
となる。

$|X|\le M$ かつ $|X_n-X|<\delta$ なら $|X_n|\le M+1$ であり、上の一様連続性が使える。従って事象の包含
\[
\{|f(X_n)-f(X)|\ge\varepsilon\}
\subset
\{|X|>M\}\cup\{|X_n-X|\ge\delta\}
\]
が成り立つので
\[
P(|f(X_n)-f(X)|\ge\varepsilon)
\le P(|X|>M)+P(|X_n-X|\ge\delta),
\]
を得る。$X_n\to X$ は確率収束するから、十分大きな $n$ では第2項も $\eta/2$ 未満である。よって左辺は $\eta$ 未満となり、$f(X_n)\to f(X)$ は確率収束する。
\end{enumerate}
\end{proof}
