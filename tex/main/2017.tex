\section{専門基礎科目（3題必修）}

\begin{problem}{第1問｜不変部分空間}{2017-b-1}
$f:\R^3\to\R^3$ を
\[
f(x)=\begin{pmatrix}1&0&0\\1&3&-3\\-2&2&-4\end{pmatrix}x
\]
で定める。$f(W)\subset W$ となる1次元および2次元の部分空間 $W$ を全て求めよ。
\end{problem}
\begin{memo*}
\textbf{解答の見通し。}
$f(W)\subset W$ は、$W$ のベクトルへ行列 $A$ を掛けても結果が $W$ の外へ出ない、という意味である。
まず $|\lambda I-A|=0$ から三つの固有値を求め、各連立方程式
$(A-\lambda I)x=0$ を解いて固有方向を得る。
1次元不変部分空間では、生成ベクトル $w$ に対して $Aw$ が $w$ の定数倍になるため、
$w$ は固有ベクトルでなければならない。

2次元の場合に「固有方向二つの組」以外がないことは、固有値ごとの成分を取り出す
多項式 $\Pi_j(A)$ を具体的に作って示す。$W$ は $A$ で不変なら $A$ の多項式でも不変なので、
$w\in W$ の各固有方向成分も $W$ に入る。
\end{memo*}
\begin{proof}[解答]
行列を $A$ と書く。数 $\lambda$ が固有値であるとは、$(\lambda I-A)x=0$ が
$x\ne0$ の解を持つこと、すなわち $|\lambda I-A|=0$ となることである。まず
\[
\lambda I-A
=\begin{pmatrix}
\lambda-1&0&0\\
-1&\lambda-3&3\\
2&-2&\lambda+4
\end{pmatrix}
\]
である。第1行は $(1,1)$ 成分以外が0なので、第1行に沿って余因子展開すると
\begin{align*}
|\lambda I-A|
&=(\lambda-1)\begin{vmatrix}\lambda-3&3\\-2&\lambda+4\end{vmatrix}\\
&=(\lambda-1)\{(\lambda-3)(\lambda+4)+6\}\\
&=(\lambda-1)(\lambda^2+\lambda-6)\\
&=(\lambda-1)(\lambda-2)(\lambda+3).
\end{align*}
従って固有値は $1,2,-3$ の三つで、互いに相異なる。

各固有空間を、連立方程式 $(A-\lambda I)x=0$ を解いて求める。
$\lambda=1$ では
\[
A-I=\begin{pmatrix}0&0&0\\1&2&-3\\-2&2&-5\end{pmatrix}
\]
なので、方程式は
\[
x+2y-3z=0,\qquad -2x+2y-5z=0
\]
の二本である。第1式を $x=3z-2y$ と解いて第2式へ代入すると
\[
-2(3z-2y)+2y-5z=6y-11z=0
\]
となる。分数を避けるため $z=6t$ と取れば $y=11t$、$x=18t-22t=-4t$ である。
$\lambda=2$ では
\[
A-2I=\begin{pmatrix}-1&0&0\\1&1&-3\\-2&2&-6\end{pmatrix}
\]
であり、第1行から $x=0$、第2行から $y=3z$ が従う。第3行はこの二つから
自動的に満たされるので、新しい条件を与えない。
$\lambda=-3$ では
\[
A+3I=\begin{pmatrix}4&0&0\\1&6&-3\\-2&2&-1\end{pmatrix}
\]
であり、第1行から $x=0$、第2行から $z=2y$ が従う。第3行 $-2x+2y-z=0$ は
同じ条件である。以上より、各固有空間は1次元で、固有ベクトルを
\[
v_1=(-4,11,6)^{\mathsf T},\quad v_2=(0,3,1)^{\mathsf T},\quad
v_3=(0,1,2)^{\mathsf T}
\]
と取れる。相異なる固有値に属する固有ベクトルは一次独立なので、
$v_1,v_2,v_3$ は $\R^3$ の基底である。

1次元部分空間 $W=\langle w\rangle$ が不変なら $Aw\in\langle w\rangle$ だから、
$Aw=\lambda w$ となり、$w$ は固有ベクトルである。各固有空間は1次元なので、
1次元不変部分空間は
$\langle v_1\rangle,\langle v_2\rangle,\langle v_3\rangle$ に限る。

2次元不変部分空間 $W$ を考える。$v_1,v_2,v_3$ は基底なので、任意の $w\in\R^3$ は
\[
w=c_1v_1+c_2v_2+c_3v_3
\]
と一意に書ける。この分解の各項を取り出すため、固有値 $\lambda_1,\lambda_2,\lambda_3=1,2,-3$ に対して
\[
\Pi_1=\frac{(A-2I)(A+3I)}{(1-2)(1+3)},\qquad
\Pi_2=\frac{(A-I)(A+3I)}{(2-1)(2+3)},\qquad
\Pi_3=\frac{(A-I)(A-2I)}{(-3-1)(-3-2)}
\]
とおく。各 $\Pi_k$ は、分子で $\lambda_k$ 以外の固有値の因子を掛け、
分母でその値を1に正規化した行列である。実際、$Av_j=\lambda_jv_j$ より
$(A-\mu I)v_j=(\lambda_j-\mu)v_j$ なので
\[
\Pi_1v_j=\frac{(\lambda_j-2)(\lambda_j+3)}{(1-2)(1+3)}\,v_j
\]
となる。$j=1$ では係数は $((1-2)(1+3))/((1-2)(1+3))=1$、
$j=2,3$ では分子に因子 $\lambda_j-2$ または $\lambda_j+3$ が0として現れるため係数は0である。従って
\[
\Pi_1v_1=v_1,\qquad \Pi_1v_2=\Pi_1v_3=0,
\qquad\text{すなわち}\qquad
\Pi_1w=c_1v_1.
\]
同じ計算により $\Pi_2w=c_2v_2$、$\Pi_3w=c_3v_3$ である。

$W$ が $A$ で不変なら、$w\in W$ に対し $Aw\in W$、これを繰り返して $A^2w\in W$ であり、
部分空間は和と定数倍で閉じているので、$A$ の多項式で書けた $\Pi_k$ についても
$\Pi_kw\in W$ となる。従って $w$ の各成分 $c_kv_k$ は $W$ に属する。
すなわち $W$ は、$W$ の元に現れる固有方向 $v_k$ たちで生成される。
$\dim W=2$ であり各 $\langle v_k\rangle$ は1次元だから、現れる固有方向はちょうど二つである。よって
\[
\langle v_1,v_2\rangle,\qquad
\langle v_1,v_3\rangle,\qquad
\langle v_2,v_3\rangle
\]
のいずれかに限る。逆にこれら三つは固有ベクトルで生成されるので、
$A(av_i+bv_j)=\lambda_iav_i+\lambda_jbv_j$ より $A$ で不変である。
従って2次元不変部分空間はこの三つで全てである。
\end{proof}

\begin{problem}{第2問｜近似恒等核}{2017-b-2}
$f$ は $[0,\infty)$ 上の有界連続関数とし
$I_n=\int_0^\infty n f(x)e^{-nx}\,\d x$ とおく。
\begin{enumerate}
  \item[(1)] $I_n$ が有限であることを示せ。
  \item[(2)] 任意の $\delta>0$ に対し $\displaystyle\int_\delta^\infty nf(x)e^{-nx}\,\d x\to0$ を示せ。
  \item[(3)] $I_n\to f(0)$ を示せ。
\end{enumerate}
\end{problem}
\begin{memo*}
\textbf{解答の見通し。}
有界とは、ある定数 $M$ が存在して全ての $x\ge0$ で $|f(x)|\le M$ となることである。
重み $ne^{-nx}$ は非負で、$[0,\infty)$ 上の積分が1になる。
さらに固定した $\delta>0$ より右側の積分は $e^{-n\delta}$ なので、$n\to\infty$ で0になる。

$I_n-f(0)$ を一つの積分として書き、$[0,\delta]$ と $[\delta,\infty)$ に分ける。
前半では0における連続性により $f(x)-f(0)$ が小さく、後半では $f$ の有界性と
指数関数の減衰により積分全体が小さくなる。この二つを別々に $\varepsilon/2$ 未満へ抑える。
\end{memo*}
\begin{proof}[解答]
$f$ が有界であるとは、ある定数 $M\ge0$ が存在して、全ての $x\ge0$ で
$|f(x)|\le M$ となることである。以下で繰り返し使うため、重み $ne^{-nx}$ の積分を先に計算しておく。
$a\ge0$ に対して
\[
\int_a^\infty ne^{-nx}\,\d x
=\lim_{R\to\infty}\left[-e^{-nx}\right]_{x=a}^{x=R}
=\lim_{R\to\infty}\left(e^{-na}-e^{-nR}\right)
=e^{-na}.
\]
特に $a=0$ とすれば
\[
\int_0^\infty ne^{-nx}\,\d x=1
\]
であり、この重みは全体で1になる。
\begin{enumerate}
  \item[(1)] 全ての $x\ge0$ で
  \[
  |nf(x)e^{-nx}|=n|f(x)|e^{-nx}\le Mne^{-nx}
  \]
  であるから、上で計算した積分を使うと
  \[
  \int_0^\infty|nf(x)e^{-nx}|\,\d x
  \le M\int_0^\infty ne^{-nx}\,\d x=M<\infty.
  \]
  よって積分は絶対収束し、$I_n$ は有限の値を取る。

  \item[(2)] 同じ評価を区間 $[\delta,\infty)$ で行うと
  \[
  \left|\int_\delta^\infty nf(x)e^{-nx}\,\d x\right|
  \le M\int_\delta^\infty ne^{-nx}\,\d x
  =Me^{-n\delta}.
  \]
  $\delta>0$ は $n$ によらず固定されているので、$n\to\infty$ のとき
  $e^{-n\delta}\to0$ である。従って左辺も0へ収束する。

  \item[(3)] 重みの積分が1であることから
  $f(0)=\int_0^\infty nf(0)e^{-nx}\,\d x$ なので、差を一つの積分にまとめて
  \[
  I_n-f(0)=\int_0^\infty n\{f(x)-f(0)\}e^{-nx}\,\d x
  \]
  と書ける。任意の $\varepsilon>0$ を取る。$f$ は0で連続だから、ある $\delta>0$ が存在して
  \[
  0\le x\le\delta\quad\Longrightarrow\quad
  |f(x)-f(0)|<\frac\varepsilon2
  \]
  となる。一方、全ての $x\ge0$ で
  $|f(x)-f(0)|\le|f(x)|+|f(0)|\le2M$ である。
  積分区間を $[0,\delta]$ と $[\delta,\infty)$ に分け、前半には連続性から得た評価、
  後半には有界性から得た評価を使うと
  \begin{align*}
  |I_n-f(0)|
  &\le\int_0^\delta n|f(x)-f(0)|e^{-nx}\,\d x
    +\int_\delta^\infty n|f(x)-f(0)|e^{-nx}\,\d x\\
  &\le\frac\varepsilon2\int_0^\delta ne^{-nx}\,\d x
    +2M\int_\delta^\infty ne^{-nx}\,\d x\\
  &\le\frac\varepsilon2+2Me^{-n\delta}.
  \end{align*}
  最後の行では、被積分関数が非負であることから
  $\int_0^\delta ne^{-nx}\,\d x\le\int_0^\infty ne^{-nx}\,\d x=1$ を使った。
  ここで $\delta$ は $n$ によらず定まっているので、
  $2Me^{-n\delta}<\varepsilon/2$ となる $N$ を取れる。$n\ge N$ なら
  $|I_n-f(0)|<\varepsilon$ である。$\varepsilon>0$ は任意だったから $I_n\to f(0)$ である。
\end{enumerate}
\end{proof}

\begin{problem}{第3問｜有理数の部分空間}{2017-b-3}
$\Q$ に通常の距離を入れ，$A=\{x\in\Q\mid1\le x\le2\}$ とする。$A$ は閉，コンパクト，連結か。
\end{problem}
\begin{memo*}
\textbf{解答の見通し。}
ここでの全体空間は $\R$ ではなく $\Q$ なので、閉・コンパクト・連結は全て $\Q$ の中で判定する。
$A$ は $\R$ の閉集合 $[1,2]$ と $\Q$ の共通部分なので、相対位相の定義から $\Q$ で閉である。
つまり「閉である」ことと「コンパクトである」ことは、ここでは連動しない。

$\R$ なら Heine--Borel の定理で $[1,2]$ はコンパクトだが、$\Q$ では $A$ の内部に
無理数の「穴」が空いている。そこで $1<c<2$ となる無理数 $c$ を一つ取り、
$c$ からの距離が $1/n$ より大きい点全体という開集合の増大列で $A$ を覆う。
有限個を選べばその和は一つの $W_N$ に等しく、$c$ のすぐ右にある有理数を覆い残す。

連結でないことを示すには、同じ $c$ で $A$ を交わらない二つの非空な相対開集合へ分ける。
$c$ 自身は $\Q$ に存在しないため、左側と右側の間に点を残さず分割できる。
\end{memo*}
\begin{proof}[解答]
閉集合ではあるが、コンパクトでも連結でもない。全体空間は $\R$ ではなく $\Q$ なので、
以下の判定は全て距離空間 $\Q$ の中で行う。
以下では $c=\sqrt2$ とおく。$1<c<2$ であり、$c$ は無理数だから $c\notin\Q$ である。

まず $A$ が $\Q$ の閉集合であることを示す。補集合を取ると
\[
\Q\setminus A
=\{x\in\Q\mid x<1\}\cup\{x\in\Q\mid x>2\}
\]
である。$\{x\in\R\mid x<1\}$ と $\{x\in\R\mid x>2\}$ は $\R$ の開集合であり、
$\Q$ の位相は $\R$ からの相対位相だから、これらと $\Q$ の共通部分は $\Q$ の開集合である。
開集合の合併は開集合なので $\Q\setminus A$ は $\Q$ で開であり、$A$ は $\Q$ の閉集合である。

次に $A$ がコンパクトでないことを示す。$n\ge1$ に対して
\[
W_n=\left\{x\in\Q\ \middle|\ |x-c|>\frac1n\right\}
\]
とおく。$W_n$ は $\R$ の開集合 $\{x\in\R\mid|x-c|>1/n\}$ と $\Q$ の共通部分だから
$\Q$ の開集合である。また $1/n$ は $n$ について減少するので
\[
W_1\subset W_2\subset\cdots
\]
と増大する。$A$ の点はすべて有理数で $c\notin\Q$ だから $|x-c|>0$ であり、
$1/n<|x-c|$ となる $n$ を取れば $x\in W_n$ となる。従って $\{W_n\}_{n\ge1}$ は $A$ の開被覆である。

ここで有限個 $W_{n_1},\ldots,W_{n_k}$ を選び、$N=\max\{n_1,\ldots,n_k\}$ とおくと、
増大列であることから
\[
W_{n_1}\cup\cdots\cup W_{n_k}=W_N
\]
である。$c<2$ なので $\min\{c+1/N,\,2\}>c$ であり、有理数の稠密性から
\[
c<x<\min\left\{c+\frac1N,\,2\right\}
\]
を満たす有理数 $x$ が取れる。この $x$ は $1<c<x<2$ より $x\in A$ であるが、
$0<x-c<1/N$ すなわち $|x-c|<1/N$ なので $x\notin W_N$ である。
よってどの有限部分族も $A$ を覆えず、$A$ はコンパクトでない。

最後に連結性を見る。
\[
U=A\cap\{x\in\Q\mid x<c\},\qquad
V=A\cap\{x\in\Q\mid x>c\}
\]
とおく。$\{x\in\Q\mid x<c\}$ と $\{x\in\Q\mid x>c\}$ は $\Q$ の開集合なので、
$U,V$ は $A$ の相対位相で開である。
$1\in U$ であり、有理数の稠密性から $c<x<2$ を満たす有理数 $x$ が取れて $x\in V$ だから、
$U,V$ はともに空でない。さらに $c\notin\Q$ なので、$A$ の各点は $x<c$ と $x>c$ の
ちょうど一方だけを満たす。従って
\[
A=U\sqcup V
\]
となる。よって $U,V$ は $A$ の非自明な開分離であり、$A$ は連結でない。
\end{proof}

\begin{problem}{第1問｜群の交換子}{2017-s-1}
$G$ を有限群，$K\subset H$ を部分群とし
$X=\{x\in G\mid[x,h]\in K\ (\forall h\in H)\}$，$[x,y]=x^{-1}y^{-1}xy$ とおく。
\begin{enumerate}
  \item[(1)] $x\in X$ なら $x^{-1}Kx=K$ を示せ。
  \item[(2)] $X$ が $G$ の部分群であることを示せ。
  \item[(3)] $K\triangleleft H$ かつ $H/K$ が可換であることと $H\subset X$ が同値であることを示せ。
\end{enumerate}
\end{problem}
\begin{memo*}
\textbf{解答の見通し。}
$[x,k]\in K$ を変形すると、$x$ が共役によって $K$ を保つことが分かる。
この正規化の事実と積・逆元に対する交換子公式を組み合わせて $X$ の部分群性を示す。
最後は「商群で可換」と「全ての交換子が $K$ に入る」が同値であることを使う。
\end{memo*}
\begin{proof}[解答]
(1) $k\in K\subset H$ とすると $[x,k]=x^{-1}k^{-1}xk\in K$ である。
従って $x^{-1}k^{-1}x=[x,k]k^{-1}\in K$、逆元を取って $x^{-1}kx\in K$。
よって $x^{-1}Kx\subset K$ であり、両者は有限集合で同じ位数を持つので等号となる。

(2) まず $1\in X$ である。$x,y\in X$, $h\in H$ に対し交換子公式から
\[
[xy,h]=y^{-1}[x,h]y[y,h].
\]
(1) より $y^{-1}Ky=K$ であり、$[x,h],[y,h]\in K$ だから右辺は $K$ に入る。
従って $xy\in X$。さらに
\[
[x^{-1},h]=\bigl(x[x,h]x^{-1}\bigr)^{-1}
\]
であり、$[x,h]\in K$、(1) より $xKx^{-1}=K$ なので $x^{-1}\in X$。
よって $X$ は部分群である。

(3) $H\subset X$ とする。(1) を $x\in H$ に適用すれば $K\triangleleft H$。
また全ての $h_1,h_2\in H$ について $[h_1,h_2]\in K$ だから、商群では
$(h_1K)(h_2K)=(h_2K)(h_1K)$ となり $H/K$ は可換である。
逆に $K\triangleleft H$ かつ $H/K$ が可換なら、任意の $h_1,h_2\in H$ の交換子は
$K$ に入る。従って各 $h_1\in H$ は $X$ に属し、$H\subset X$ となる。
\end{proof}

\begin{problem}{第2問｜剰余環と局所化}{2017-s-2}
$R$ を可換環，$a\in R$，$S=\{a^n\mid n\ge0\}$ とする。自然な写像 $\pi:R\to R/(a)$，$\tau:R\to S^{-1}R$ を考える。
\begin{enumerate}
  \item[(1)] $x\in\ker\pi\cap\ker\tau$ なら，ある $n$ で $x^n=0$ となることを示せ。
  \item[(2)] $\tau$ が全射なら $\ker\pi+\ker\tau=R$ を示せ。
\end{enumerate}
\end{problem}
\begin{memo*}
\textbf{解答の見通し。}
局所化で $x/1=0$ とは、ある $a^m$ を掛けると $x$ が0になることを意味する。
(1) ではさらに $x=ar$ を使って $x$ 自身の冪を0にする。
(2) では全射性により局所化の元 $1/a$ を $R$ の元で表し、
そこから $1$ を二つの核の元の和として書く。
\end{memo*}
\begin{proof}[解答]
\begin{enumerate}
  \item[(1)] $x\in\ker\pi$ であることは $x\in(a)$ を意味するので、ある $r\in R$ により
  \[
  x=ar
  \]
  と書ける。また $x\in\ker\tau$ だから $x/1=0/1$ である。
  局所化における分数の等号の定義より、ある $m\ge0$ が存在して
  \[
  a^m(x\cdot1-0\cdot1)=a^mx=0
  \]
  となる。従って
  \[
  x^{m+1}=(ar)^m x=a^m r^m x=r^m(a^mx)=0
  \]
  であり、$x$ は冪零である。

  \item[(2)] $\tau$ が全射なので、$S^{-1}R$ の元 $1/a$ に対して、ある $b\in R$ が存在し
  \[
  \frac b1=\frac1a
  \]
  となる。再び局所化における分数の等号の定義を使うと、ある $n\ge0$ に対して
  \[
  a^n(ab-1)=0
  \]
  である。従って
  \[
  \tau(ab-1)=\frac{ab-1}{1}=0,
  \qquad ab-1\in\ker\tau.
  \]
  一方、$ab\in(a)=\ker\pi$ であるから
  \[
  1=ab-(ab-1)\in\ker\pi+\ker\tau.
  \]
  二つの核の和は $R$ のイデアルであり、$1$ を含むので
  $\ker\pi+\ker\tau=R$ である。
\end{enumerate}
\end{proof}

\begin{problem}{第3問｜被覆空間と Betti 数}{2017-s-3}
\begin{enumerate}
  \item[(1)] $F:\R\to S^1$，$F(\theta)=(\cos\theta,\sin\theta)$ が普遍被覆であることを示し，被覆変換群を求めよ。
  \item[(2)] 四面体の境界のうち，三つの面からなる複体を $K_1$，残り一面を $K_2$ とする。$K_1\cap K_2,K_1,K_2,K_1\cup K_2$ の Betti 数を求めよ。
\end{enumerate}
\end{problem}
\begin{memo*}
\textbf{解答の見通し。}
被覆写像の確認では短い円弧の逆像が $2\pi$ ごとの区間へ分かれることを明示する。
被覆変換は $F(T(\theta))=F(\theta)$ から差が $2\pi$ の整数倍と分かり、
連結性によってその整数が一定になる。
複体はそれぞれ $D^2,S^1,S^2$ のどれと同相かを見抜けば Betti 数を読める。
\end{memo*}
\begin{proof}[解答]
\begin{enumerate}
\item[(1)] $z_0=F(\theta_0)\in S^1$ を任意に取り、$0<\varepsilon<\pi$ とする。
円弧
\[
U=F((\theta_0-\varepsilon,\theta_0+\varepsilon))
\]
に対して
\[
F^{-1}(U)
=\bigsqcup_{k\in\Z}
(\theta_0-\varepsilon+2\pi k,\theta_0+\varepsilon+2\pi k)
\]
であり、各区間への $F$ の制限は $U$ への同相写像である。
従って $F$ は被覆写像である。さらに $\R$ は単連結なので、これは普遍被覆である。

$k\in\Z$ に対して
\[
T_k(\theta)=\theta+2\pi k
\]
とおけば $F\circ T_k=F$ なので、$T_k$ は被覆変換である。逆に被覆変換 $T$ に対して
\[
F(T(\theta))=F(\theta)
\quad\Longrightarrow\quad
T(\theta)-\theta\in2\pi\Z.
\]
左辺は $\theta$ の連続関数であり、$\R$ は連結だから、この整数値は一定である。
従って $T=T_k$ となる。合成について $T_kT_\ell=T_{k+\ell}$ なので、被覆変換群は $\Z$ と同型である。

\item[(2)] $K_2$ は三角形一枚なので $D^2$ と同相であり、$K_1$ も残りの三面を合わせた円板なので $D^2$ と同相である。
両者の共通部分は $K_2$ の境界であるから
\[
K_1\simeq D^2,\qquad K_2\simeq D^2,
\qquad K_1\cap K_2\simeq S^1,
\qquad K_1\cup K_2\simeq S^2.
\]
$D^2,S^1,S^2$ のホモロジー群から、Betti 数は
\[
\begin{array}{c|ccc}
&b_0&b_1&b_2\\\hline
K_1\cap K_2&1&1&0\\
K_1&1&0&0\\
K_2&1&0&0\\
K_1\cup K_2&1&0&1
\end{array}
\]
であり、高次の Betti 数はいずれも0である。
\end{enumerate}
\end{proof}

\begin{problem}{第4問｜管状曲面}{2017-s-4}
$0<a<1$ とし
\[
p(u,v)=(u,u^2/2,0)+\frac{a\cos v}{\sqrt{u^2+1}}(-u,1,0)+a\sin v(0,0,1).
\]
これが曲面であることを示し，Gauss 曲率・平均曲率を求めよ。また $q(s)=p(u_0,s/a)$ が測地線であることを示せ。
\end{problem}
\begin{memo*}
\textbf{解答の見通し。}
与式を中心曲線 $c$ の法平面内に半径 $a$ の円を動かした管状曲面として読む。
Frenet 標構で $p_u,p_v$ を表せば、正則性と主方向が同時に分かる。
単位法線の微分から二つの主曲率を求め、その積と平均を取る。
$u=u_0$ の円では、単位速度曲線の加速度が曲面法線方向だけになることを示す。
\end{memo*}
\begin{proof}[解答]
中心曲線を $c(u)=(u,u^2/2,0)$ とし、
\[
w=\norm{c'(u)}=\sqrt{1+u^2},\quad
T=\frac{(1,u,0)}w,\quad
N=\frac{(-u,1,0)}w,\quad B=(0,0,1)
\]
とおく。中心曲線の曲率は
\[
\kappa=\frac{\norm{c'\times c''}}{\norm{c'}^3}
=\frac1{(1+u^2)^{3/2}}
\]
であり、Frenet の公式から $N_u=-\kappa wT$ である。問題の写像は
\[
p(u,v)=c(u)+a\cos v\,N(u)+a\sin v\,B
\]
と書けるので
\[
p_u=w(1-a\kappa\cos v)T,
\qquad p_v=a(-\sin v\,N+\cos v\,B).
\]
$0<a<1$ かつ $0<\kappa\le1$ だから $1-a\kappa\cos v\ge1-a>0$。
従って $p_u,p_v$ はともに0でなく互いに直交し、$p_u\times p_v\ne0$ なので正則な曲面である。

外向き単位法線を $\nu=\cos v\,N+\sin v\,B$ と取ると
\[
\nu_u=-\kappa w\cos v\,T,
\qquad \nu_v=-\sin v\,N+\cos v\,B=\frac1a p_v.
\]
形作用素 $S(X)=-\d\nu(X)$ を $p_u,p_v$ に作用させると
\[
S(p_u)=\frac{\kappa\cos v}{1-a\kappa\cos v}p_u,
\qquad S(p_v)=-\frac1a p_v.
\]
従って座標方向は主方向で、主曲率は
\[
k_1=\frac{\kappa\cos v}{1-a\kappa\cos v},\qquad k_2=-\frac1a.
\]
よって積と平均を取れば
\[
K=-\frac{\kappa\cos v}{a(1-a\kappa\cos v)},\qquad
H=\frac12\left(\frac{\kappa\cos v}{1-a\kappa\cos v}-\frac1a\right).
\]
$q(s)=p(u_0,s/a)$ について連鎖律から
\[
q'(s)=\frac1a p_v(u_0,s/a),\qquad
q''(s)=\frac1{a^2}p_{vv}(u_0,s/a).
\]
$\norm{p_v}=a$ なので $q$ は単位速度であり、
$p_{vv}=-a(\cos v\,N+\sin v\,B)=-a\nu$ だから
\[
q''(s)=-\frac1a\nu(u_0,s/a).
\]
加速度が曲面法線方向だけで接成分が0なので、$q$ は測地線である。
\end{proof}

\begin{problem}{第5問｜扇形上の正則関数}{2017-s-5}
$S=\{z\in\mathbb C\mid0<|z|<r,\ \alpha<\arg z<\beta\}$，$\beta<\alpha+2\pi$ とする。
\begin{enumerate}
  \item[(1)] $f(z)$ は $z\to0$ で極限を持つが $f'(z)$ は持たない例を挙げよ。
  \item[(2)] $0<r'<r$，$\alpha<\alpha'<\beta'<\beta$ で切り取った閉じた小扇形の各点を中心に，$S$ に含まれる円板があることを示せ。
  \item[(3)] $f$ が $S$ 上有界正則なら，境界から $\varepsilon$ だけ離した小扇形 $T_\varepsilon$ 上で $|f'(z)|\le M_\varepsilon/|z|$ となることを示せ。
\end{enumerate}
\end{problem}
\begin{memo*}
\textbf{解答の見通し。}
角度が $2\pi$ 未満なので対数の枝を取り、平方根で (1) の例を作る。
境界から角度 $\varepsilon$ だけ離れた点 $z$ では、
半径を $c_\varepsilon|z|$ とする円板を扇形内に一様に取れる。
その円周上で Cauchy の微分公式を使うと、半径の逆数 $1/|z|$ が現れる。
\end{memo*}
\begin{proof}[解答]
\begin{enumerate}
\item[(1)] 扇形の角度は $2\pi$ 未満なので、$S$ 上で正則な対数の枝を取れる。
\[
f(z)=\exp\left(\frac12\log z\right)
\]
とおけば $|f(z)|=|z|^{1/2}\to0$ である。一方
\[
f'(z)=\frac12\exp\left(-\frac12\log z\right),\qquad
|f'(z)|=\frac1{2\sqrt{|z|}}\longrightarrow\infty,
\]
従って $f'$ は有限な極限を持たない。

\item[(2)] 切り取った小扇形の点 $z$ は $0<|z|<r$ かつ
$\alpha<\arg z<\beta$ を満たすので $z\in S$ である。$S$ は開集合だから、
開集合の定義により、各 $z$ に対してある $\rho_z>0$ が存在し
$B(z,\rho_z)\subset S$ となる。

\item[(3)] $T_\varepsilon$ では角度が二つの境界半直線から少なくとも
$\varepsilon$ 離れ、外周も $r'<r$ に切り取られているとする。
\[
0<c_\varepsilon<\min\left\{1,\sin\varepsilon,\frac r{r'}-1\right\}
\]
を取れば、任意の $z\in T_\varepsilon$ に対し
\[
B(z,c_\varepsilon|z|)\subset S.
\]
実際、半径が $|z|$ 未満なので円板は0を含まず、境界半直線までの距離
$|z|\sin\varepsilon$ より小さく、さらに円板内の $w$ は
$|w|<(1+c_\varepsilon)r'<r$ を満たす。

$|f|\le B$ とし、円 $|\zeta-z|=c_\varepsilon|z|$ 上で Cauchy の微分公式を使うと
\[
f'(z)=\frac1{2\pi i}\int_{|\zeta-z|=c_\varepsilon|z|}
\frac{f(\zeta)}{(\zeta-z)^2}\,\d\zeta.
\]
従って円周の長さも含めて評価すると
\[
|f'(z)|\le
\frac1{2\pi}\frac{B}{(c_\varepsilon|z|)^2}
2\pi c_\varepsilon|z|
=\frac{B}{c_\varepsilon|z|}.
\]
$M_\varepsilon=B/c_\varepsilon$ とおけば求める評価を得る。
\end{enumerate}
\end{proof}

\begin{problem}{第6問｜線形微分方程式}{2017-s-6}
\begin{enumerate}
  \item[(1)] $y''+2y'+2y=0$，$y(0)=1,y'(0)=0$ を解き，零点を全て求めよ。
  \item[(2)] $y''+2y=\cos(\omega x)$，$y(0)=0,y'(0)=1$ の解が $x>0$ で非有界となる $\omega$ と，その解を求めよ。
\end{enumerate}
\end{problem}
\begin{memo*}
\textbf{解答の見通し。}
(1) は複素特性根から減衰振動の形を作り、三角関数の位相をずらして零点を読む。
(2) は外力の角振動数 $|\omega|$ が固有角振動数 $\sqrt2$ と一致するときだけ共振する。
共振時は通常の三角関数に $x$ を掛けた特殊解を選び、初期条件を合わせる。
\end{memo*}
\begin{proof}[解答]
\begin{enumerate}
\item[(1)] 特性方程式は
\[
r^2+2r+2=0,\qquad r=-1\pm i
\]
なので一般解は $y=e^{-x}(C_1\cos x+C_2\sin x)$。
$y(0)=1$ から $C_1=1$、
\[
y'(0)=-C_1+C_2=0
\]
から $C_2=1$。従って
\[
y=e^{-x}(\cos x+\sin x),\qquad x=-\frac\pi4+k\pi\quad(k\in\Z).
\]
最後の零点は $e^{-x}\ne0$ と
$\cos x+\sin x=\sqrt2\sin(x+\pi/4)$ から得られる。
\item[(2)] $\omega^2\ne2$ なら特殊解 $\cos(\omega x)/(2-\omega^2)$ が取れ、
同次解も三角関数なので全ての解は有界である。
$\omega=\pm\sqrt2$ のときは共振し、$a=\sqrt2$ とおけば
\[
y_p=\frac{x}{2a}\sin ax
\]
と置く。このとき
\[
y_p'=\frac1{2a}\sin ax+\frac x2\cos ax,
\]
\[
y_p''=\frac12\cos ax+\frac12\cos ax-\frac{ax}{2}\sin ax
=\cos ax-\frac{ax}{2}\sin ax.
\]
従って
\[
y_p''+a^2y_p
=\cos ax-\frac{ax}{2}\sin ax+\frac{ax}{2}\sin ax
=\cos ax.
\]
初期条件を満たす同次解 $(1/a)\sin ax$ を加えて
\[
y(x)=\frac{x+2}{2\sqrt2}\sin(\sqrt2x).
\]
これは $x>0$ で非有界である。従って求める $\omega$ は $\omega=\pm\sqrt2$。
\end{enumerate}
\end{proof}

\begin{problem}{第7問｜パラメータ積分}{2017-s-7}
$f$ は $(0,\infty)$ 上可測で $\int_0^\infty x|f(x)|\,\d x<\infty$ を満たす。次を求めよ。
\[
\lim_{a\downarrow0}\frac1a\int_0^\infty(\log(1+ax)-ax)f(x)\,\d x.
\]
\end{problem}
\begin{memo*}
\textbf{解答の見通し。}
固定した $x$ では $\log(1+ax)-ax=o(a)$ なので被積分関数は0へ収束する。
積分と極限を交換するには、$0\le u-\log(1+u)\le u$ を使って
絶対値を仮定中の可積分関数 $x|f(x)|$ で抑える。
\end{memo*}
\begin{proof}[解答]
$u=ax\ge0$ とおく。$a\downarrow0$ のとき $u\to0$ であり、
$\log(1+u)=u+o(u)$ だから、各 $x>0$ について
\[
\frac{\log(1+ax)-ax}{a}
=x\frac{\log(1+u)-u}{u}\longrightarrow0.
\]
また $\log(1+u)\le u$ であり、$\log(1+u)\ge0$ でもあるので
\[
0\le u-\log(1+u)\le u.
\]
従って
\[
\left|\frac{\log(1+ax)-ax}{a}f(x)\right|
\le x|f(x)|.
\]
仮定より右辺は $(0,\infty)$ 上可積分である。よって優収束定理により積分と極限を交換でき、求める極限は $0$ である。
\end{proof}

\begin{problem}{第8問｜確率収束}{2017-s-8}
実確率変数列 $X_n$ と実確率変数 $X$ について答えよ。
\begin{enumerate}
  \item[(1)] $\lim_{N\to\infty}P(|X|>N)=0$ を示せ。
  \item[(2)] $X_n\to X$ が確率収束し，$f:\R\to\R$ が連続なら $f(X_n)\to f(X)$ も確率収束することを示せ。
\end{enumerate}
\end{problem}
\begin{memo*}
\textbf{解答の見通し。}
まず実確率変数 $X$ が大きなコンパクト区間の外へ出る確率を小さくできることを示す。
その区間を少し広げた上で連続関数 $f$ の一様連続性を使う。
$f(X_n)$ の誤差事象を「$X$ が区間外」と「$X_n-X$ が大きい」の和集合に含めれば、
二つの確率を別々に小さくできる。
\end{memo*}
\begin{proof}[解答]
\begin{enumerate}
\item[(1)] $A_N=\{|X|>N\}$ とおくと
\[
A_1\supset A_2\supset\cdots,
\qquad \bigcap_{N=1}^{\infty}A_N=\varnothing
\]
である。確率測度の上からの連続性より
\[
\lim_{N\to\infty}P(|X|>N)
=\lim_{N\to\infty}P(A_N)
=P\left(\bigcap_{N=1}^{\infty}A_N\right)=0.
\]

\item[(2)] 任意の $\varepsilon>0$ と $\eta>0$ を取る。(1) より、十分大きな $M$ を取れば
\[
P(|X|>M)<\frac\eta2
\]
となる。$f$ はコンパクト区間 $[-M-1,M+1]$ 上で一様連続なので、ある $0<\delta\le1$ が存在して
\[
x,y\in[-M-1,M+1],\quad |x-y|<\delta
\quad\Longrightarrow\quad |f(x)-f(y)|<\varepsilon
\]
となる。

$|X|\le M$ かつ $|X_n-X|<\delta$ なら $|X_n|\le M+1$ であり、上の一様連続性が使える。従って事象の包含
\[
\{|f(X_n)-f(X)|\ge\varepsilon\}
\subset
\{|X|>M\}\cup\{|X_n-X|\ge\delta\}
\]
が成り立つので
\[
P(|f(X_n)-f(X)|\ge\varepsilon)
\le P(|X|>M)+P(|X_n-X|\ge\delta),
\]
を得る。$X_n\to X$ は確率収束するから、十分大きな $n$ では第2項も $\eta/2$ 未満である。よって左辺は $\eta$ 未満となり、$f(X_n)\to f(X)$ は確率収束する。
\end{enumerate}
\end{proof}
