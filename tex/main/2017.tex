\begin{problem}{第1問｜不変部分空間}{2017-b-1}
実ベクトル空間 $\R^3$ 上の線形写像 $f:\R^3\to\R^3$ を
\[
f(x):=\begin{pmatrix}1&0&0\\1&3&-3\\-2&2&-4\end{pmatrix}x
\qquad(x\in\R^3)
\]
で定める。次の問いに答えよ。
\begin{enumerate}
  \item[(1)] $f(W)\subset W$ かつ $\dim W=1$ となる部分空間 $W\subset\R^3$ をすべて求めよ。
  \item[(2)] $f(W)\subset W$ かつ $\dim W=2$ となる部分空間 $W\subset\R^3$ をすべて求めよ。
\end{enumerate}
\end{problem}
\begin{memo*}
\textbf{解答の見通し。}
$f(W)\subset W$ となる部分空間 $W$ を $f$ の\textbf{不変部分空間}という。
$f$ は行列 $A$ を掛ける写像なので、これは $W$ のベクトルに $A$ を掛けても
$W$ の外へ出ない、という条件である。

固有値が三つとも相異なるので、固有ベクトル $v_1,v_2,v_3$ は $\R^3$ の基底になる。
不変部分空間はこの三方向から $\dim W$ 個を選んで生成したものに限られるので、
(1) は3つ、(2) も3つ出る。固有値と固有ベクトルの計算は両方で共通なので先にまとめておく。

「組み合わせに限る」ことは、$w\in W$ に $(A-\lambda_iI)(A-\lambda_jI)$ を掛けると
残り一つの方向 $v_k$ だけが生き残ることから従う。
$W$ は $A$ で不変だから、掛けた結果も $W$ に入る。
\end{memo*}
\begin{proof}[解答]
係数行列を $A$ とする。
\[
\det(A-\lambda I)
=(1-\lambda)(\lambda+3)(\lambda-2).
\]
対応する固有ベクトルとして
\[
v_1=\begin{pmatrix}-4\\11\\6\end{pmatrix}\ (\lambda=1),\qquad
v_2=\begin{pmatrix}0\\3\\1\end{pmatrix}\ (\lambda=2),\qquad
v_3=\begin{pmatrix}0\\1\\2\end{pmatrix}\ (\lambda=-3)
\]
を取れる。固有値が相異なるので $v_1,v_2,v_3$ は基底をなす。

$A$-不変部分空間 $W$ は、これらの固有ベクトルの一部で生成される。
実際、$w=\sum c_kv_k\in W$ に対し
\[
\prod_{j\ne k}\frac{A-\lambda_jI}{\lambda_k-\lambda_j}\,w=c_kv_k\in W
\]
となる。逆に、それらの任意の組の生成する空間は明らかに $A$-不変である。
従って
\begin{enumerate}
  \item[(1)]
  \[
  W=\langle v_1\rangle,\qquad\langle v_2\rangle,\qquad\langle v_3\rangle.
  \]

  \item[(2)]
  \[
  W=\langle v_1,v_2\rangle,\qquad
  \langle v_1,v_3\rangle,\qquad
  \langle v_2,v_3\rangle.
  \]
\end{enumerate}
\end{proof}

\begin{problem}{第2問｜近似恒等核}{2017-b-2}
$f(x)$ を区間 $[0,\infty)$ で定義された有界かつ連続な関数とするとき，次の問いに答えよ。
\begin{enumerate}
  \item[(1)] 任意の自然数 $n$ に対して $\displaystyle\int_0^\infty nf(x)e^{-nx}\,\d x$ は有限値であることを示せ。この積分値を $I_n$ とおく。
  \item[(2)] 任意に選んだ $\delta>0$ に対して
  \[
  \lim_{n\to\infty}\int_\delta^\infty nf(x)e^{-nx}\,\d x=0
  \]
  を示せ。
  \item[(3)] $\displaystyle\lim_{n\to\infty}I_n=f(0)$ を示せ。
\end{enumerate}
\end{problem}
\begin{memo*}
\textbf{解答の見通し。}
$f$ の有界性から定数 $M$ を一つ取っておき、以後の評価は全てこの $M$ に帰着させる。
重み $ne^{-nx}$ は非負で $[0,\infty)$ 上の積分が1、$\delta>0$ より右側の積分は $e^{-n\delta}$ なので、
$n\to\infty$ では右側だけが消える。これが (1) (2) の内容である。

(3) では重みの積分が1であることを使って $f(0)$ を積分の中へ入れ、$I_n-f(0)$ を一つの積分にまとめる。
$[0,\delta]$ では0における連続性、$[\delta,\infty)$ では有界性で抑えると
$|I_n-f(0)|\le\varepsilon+2Me^{-n\delta}$ となる。
$\delta$ は $\varepsilon$ だけで決まり $n$ に依らないので、$n\to\infty$ として $\limsup$ を取れば
$\varepsilon$ だけが残り、$\varepsilon$ の任意性で終わる。
$\varepsilon/2$ ずつに割り振って $N$ を取る必要はない。
\end{memo*}
\begin{proof}[解答]
$|f(x)|\le M$ $(x\ge0)$ となる $M>0$ を取る。
\begin{enumerate}
  \item[(1)]
  \[
  |I_n|\le M\int_0^\infty ne^{-nx}\,\d x=M<\infty.
  \]

  \item[(2)] 任意の $\delta>0$ に対して
  \[
  \left|\int_\delta^\infty nf(x)e^{-nx}\,\d x\right|
  \le M\int_\delta^\infty ne^{-nx}\,\d x
  =Me^{-n\delta}\longrightarrow0.
  \]

  \item[(3)] $\int_0^\infty ne^{-nx}\,\d x=1$ より
  \[
  I_n-f(0)=\int_0^\infty n\{f(x)-f(0)\}e^{-nx}\,\d x.
  \]
  任意の $\varepsilon>0$ に対し、連続性から
  $0\le x\le\delta$ なら $|f(x)-f(0)|<\varepsilon$ となる $\delta>0$ を取ると
  \[
  |I_n-f(0)|
  \le\varepsilon\int_0^\delta ne^{-nx}\,\d x
  +2M\int_\delta^\infty ne^{-nx}\,\d x
  \le\varepsilon+2Me^{-n\delta}.
  \]
  $n\to\infty$ とし、$\varepsilon>0$ の任意性を用いれば $I_n\to f(0)$。
\end{enumerate}
\end{proof}

\begin{problem}{第3問｜有理数の部分空間}{2017-b-3}
$\Q$ を有理数全体の集合とする。$\Q$ の2点 $x,y$ に対して $d(x,y):=|x-y|$ と定義することにより，
$\Q$ は $d$ を距離とする距離空間となる。$\Q$ の部分集合 $A=\{x\in\Q\mid1\le x\le2\}$ について
次の各命題は正しいか？ 理由をつけて答えよ。
\begin{enumerate}
  \item[(1)] $A$ は $\Q$ の閉集合である。
  \item[(2)] $A$ はコンパクト集合である。
  \item[(3)] $A$ は連結集合である。
\end{enumerate}
\end{problem}
\begin{memo*}
\textbf{解答の見通し。}
全体空間は $\R$ ではなく $\Q$ なので、閉・コンパクト・連結は全て $\Q$ の中で判定する。
$\R$ なら Heine--Borel の定理で $[1,2]$ はコンパクトだが、$\Q$ では $A$ に無理数の
「穴」が空いており、閉であってもコンパクト性は従わない。穴を一つ $\sqrt2$ と選べば、
コンパクト性も連結性も同じ点で崩せる。

連結とは「空でない開集合 $U,V$ で $X=U\cup V$，$U\cap V=\varnothing$ と分けられないこと」
だったから、そのような $U,V$ を一組作れば連結でないと言える。
\end{memo*}
\begin{proof}[解答]
\begin{enumerate}
  \item[(1)] 正しい。
  \[
  \Q\setminus A=\Q\cap\{(-\infty,1)\cup(2,\infty)\}
  \]
  は $\Q$ で開だから、$A$ は閉である。

  \item[(2)] 正しくない。$x_n\in A$, $x_n\to\sqrt2$ となる有理数列を取る。
  その任意の部分列も $\R$ では $\sqrt2\notin\Q$ に収束するので、
  $A$ 内で収束する部分列をもたない。従って $A$ はコンパクトでない。

  \item[(3)] 正しくない。
  \[
  U=A\cap(-\infty,\sqrt2),\qquad
  V=A\cap(\sqrt2,\infty)
  \]
  は空でない互いに素な $A$ の開集合で、$\sqrt2\notin\Q$ より
  $U\cup V=A$。従って $A$ は連結でない。
\end{enumerate}
\end{proof}

\begin{problem}{第1問｜群の交換子}{2017-s-1}
$G$ を有限群とする。$H,K$ は $G$ の部分群で $K\subset H$ であるとする。
\[
X=\{x\in G\mid[x,h]\in K\ (\forall h\in H)\}
\]
とおく。ここで，$x,y\in G$ に対して $[x,y]=x^{-1}y^{-1}xy$ とする。次の問いに答えよ。
\begin{enumerate}
  \item[(1)] $x\in X$ に対して $x^{-1}Kx=K$ が成り立つことを示せ。
  \item[(2)] $X$ は $G$ の部分群であることを示せ。
  \item[(3)] $K\triangleleft H$ かつ $H/K$ が可換群であることは，$H\subset X$ となるための必要十分条件であることを示せ。
\end{enumerate}
\end{problem}
\begin{memo*}
\textbf{解答の見通し。}
$[x,k]\in K$ を変形すると、$x$ が共役によって $K$ を保つことが分かる。
この正規化の事実と積・逆元に対する交換子公式を組み合わせて $X$ の部分群性を示す。
最後は「商群で可換」と「全ての交換子が $K$ に入る」が同値であることを使う。
\end{memo*}
\begin{proof}[解答]
(1) $k\in K\subset H$ とすると $[x,k]=x^{-1}k^{-1}xk\in K$ である。
従って $x^{-1}k^{-1}x=[x,k]k^{-1}\in K$、逆元を取って $x^{-1}kx\in K$。
よって $x^{-1}Kx\subset K$ であり、両者は有限集合で同じ位数を持つので等号となる。

(2) まず $1\in X$ である。$x,y\in X$, $h\in H$ に対し交換子公式から
\[
[xy,h]=y^{-1}[x,h]y[y,h].
\]
(1) より $y^{-1}Ky=K$ であり、$[x,h],[y,h]\in K$ だから右辺は $K$ に入る。
従って $xy\in X$。さらに
\[
[x^{-1},h]=\bigl(x[x,h]x^{-1}\bigr)^{-1}
\]
であり、$[x,h]\in K$、(1) より $xKx^{-1}=K$ なので $x^{-1}\in X$。
よって $X$ は部分群である。

(3) $H\subset X$ とする。(1) を $x\in H$ に適用すれば $K\triangleleft H$。
また全ての $h_1,h_2\in H$ について $[h_1,h_2]\in K$ だから、商群では
$(h_1K)(h_2K)=(h_2K)(h_1K)$ となり $H/K$ は可換である。
逆に $K\triangleleft H$ かつ $H/K$ が可換なら、任意の $h_1,h_2\in H$ の交換子は
$K$ に入る。従って各 $h_1\in H$ は $X$ に属し、$H\subset X$ となる。
\end{proof}

\begin{problem}{第2問｜剰余環と局所化}{2017-s-2}
$R$ を可換環とし，$a$ を $R$ の元とする。$R$ の積閉集合 $S$ を $S=\{a^n\mid n\ge0\}$ で定める。
ただし，$a^0$ は $R$ の乗法的単位元とする。$\pi:R\to R/(a)$ および $\tau:R\to S^{-1}R$ を
自然な環準同型写像とする。次の問いに答えよ。
\begin{enumerate}
  \item[(1)] $R$ の元 $x$ が $\ker\pi\cap\ker\tau$ に属するなら，$x^n=0$ となる自然数 $n$ が存在することを示せ。
  \item[(2)] $\tau$ が全射なら $\ker\pi+\ker\tau=R$ となることを示せ。
\end{enumerate}
\end{problem}
\begin{memo*}
\textbf{解答の見通し。}
局所化で $x/1=0$ とは、ある $a^m$ を掛けると $x$ が0になることを意味する。
(1) ではさらに $x=ar$ を使って $x$ 自身の冪を0にする。
(2) では全射性により局所化の元 $1/a$ を $R$ の元で表し、
そこから $1$ を二つの核の元の和として書く。
\end{memo*}
\begin{proof}[解答]
\begin{enumerate}
  \item[(1)] $x\in\ker\pi$ であることは $x\in(a)$ を意味するので、ある $r\in R$ により
  \[
  x=ar
  \]
  と書ける。また $x\in\ker\tau$ だから $x/1=0/1$ である。
  局所化における分数の等号の定義より、ある $m\ge0$ が存在して
  \[
  a^m(x\cdot1-0\cdot1)=a^mx=0
  \]
  となる。従って
  \[
  x^{m+1}=(ar)^m x=a^m r^m x=r^m(a^mx)=0
  \]
  であり、$x$ は冪零である。

  \item[(2)] $\tau$ が全射なので、$S^{-1}R$ の元 $1/a$ に対して、ある $b\in R$ が存在し
  \[
  \frac b1=\frac1a
  \]
  となる。再び局所化における分数の等号の定義を使うと、ある $n\ge0$ に対して
  \[
  a^n(ab-1)=0
  \]
  である。従って
  \[
  \tau(ab-1)=\frac{ab-1}{1}=0,
  \qquad ab-1\in\ker\tau.
  \]
  一方、$ab\in(a)=\ker\pi$ であるから
  \[
  1=ab-(ab-1)\in\ker\pi+\ker\tau.
  \]
  二つの核の和は $R$ のイデアルであり、$1$ を含むので
  $\ker\pi+\ker\tau=R$ である。
\end{enumerate}
\end{proof}

\begin{problem}{第3問｜被覆空間と Betti 数}{2017-s-3}
\begin{enumerate}
  \item[(1)] $F:\R\to S^1$，$F(\theta)=(\cos\theta,\sin\theta)$ が普遍被覆であることを示し，被覆変換群を求めよ。
  \item[(2)] $\R^4$ の点 $a_1,a_2,a_3,a_4$ を $a_1=(1,0,0,0)$, $a_2=(0,1,0,0)$, $a_3=(0,0,1,0)$, $a_4=(0,0,0,1)$ で定める。3つの2単体 $|a_1a_3a_4|$, $|a_1a_2a_4|$, $|a_1a_2a_3|$ およびこれらの全ての辺単体からなる単体的複体を $K_1$ で表し，2単体 $|a_2a_3a_4|$ およびこの全ての辺単体からなる単体的複体を $K_2$ で表す。このとき $K_1\cap K_2$ が単体的複体であることを示せ。また，非負整数 $p$ に対し $K_1\cap K_2$, $K_1$ および $K_2$ の $p$ 次元 Betti 数を求め，さらに Mayer--Vietoris 完全系列を用いて単体的複体 $K=K_1\cup K_2$ の $p$ 次元 Betti 数を求めよ。
\end{enumerate}
\end{problem}
\begin{memo*}
\textbf{解答の見通し。}
被覆写像の確認では短い円弧の逆像が $2\pi$ ごとの区間へ分かれることを明示する。
被覆変換は $F(T(\theta))=F(\theta)$ から差が $2\pi$ の整数倍と分かり、
連結性によってその整数が一定になる。
複体は $a_1,\ldots,a_4$ を頂点とする4面体の境界を三面と一面に割ったもので、
$K_1,K_2$ はそれぞれ $D^2$、共通部分は $S^1$、和は $S^2$ と同相である。
$K=K_1\cup K_2$ の Betti 数だけは、指示どおり Mayer--Vietoris 完全系列から求める。
\end{memo*}
\begin{proof}[解答]
\begin{enumerate}
\item[(1)] $z_0=F(\theta_0)\in S^1$ を任意に取り、$0<\varepsilon<\pi$ とする。
円弧
\[
U=F((\theta_0-\varepsilon,\theta_0+\varepsilon))
\]
に対して
\[
F^{-1}(U)
=\bigsqcup_{k\in\Z}
(\theta_0-\varepsilon+2\pi k,\theta_0+\varepsilon+2\pi k)
\]
であり、各区間への $F$ の制限は $U$ への同相写像である。
従って $F$ は被覆写像である。さらに $\R$ は単連結なので、これは普遍被覆である。

$k\in\Z$ に対して
\[
T_k(\theta)=\theta+2\pi k
\]
とおけば $F\circ T_k=F$ なので、$T_k$ は被覆変換である。逆に被覆変換 $T$ に対して
\[
F(T(\theta))=F(\theta)
\quad\Longrightarrow\quad
T(\theta)-\theta\in2\pi\Z.
\]
左辺は $\theta$ の連続関数であり、$\R$ は連結だから、この整数値は一定である。
従って $T=T_k$ となる。合成について $T_kT_\ell=T_{k+\ell}$ なので、被覆変換群は $\Z$ と同型である。

\item[(2)] まず $K_1\cap K_2$ が単体的複体であることを見る。
二つの複体の共通部分とは、両方に属する単体全体のことである。
$K_1$ には $a_1,\ldots,a_4$ の全ての頂点と6本の辺が属し、
$K_2$ には $a_2,a_3,a_4$ とその3本の辺、および2単体 $|a_2a_3a_4|$ が属する。
$|a_2a_3a_4|$ は $K_1$ に属さないので
\[
K_1\cap K_2=\{|a_2|,|a_3|,|a_4|,|a_2a_3|,|a_3a_4|,|a_2a_4|\},
\]
すなわち2単体 $|a_2a_3a_4|$ の境界複体である。
一般に、単体の面はもとの単体が属する複体に属するから、
共通部分も面を取る操作で閉じている。また二つの単体の共通部分は
$K_1$ でも $K_2$ でも共通の面だから、$K_1\cap K_2$ でもそうである。
従って $K_1\cap K_2$ は単体的複体である。

多面体としては、$K_2$ は三角形一枚なので $D^2$ と同相であり、
$K_1$ は4面体の残り三面を合わせたものでやはり $D^2$ と同相、
$K_1\cap K_2$ は三角形の境界なので $S^1$ と同相である。従って
\[
b_p(K_1)=b_p(K_2)=
\begin{cases}1&p=0\\0&p\ge1\end{cases},
\qquad
b_p(K_1\cap K_2)=
\begin{cases}1&p=0,1\\0&p\ge2.\end{cases}
\]

$K=K_1\cup K_2$ の Betti 数は Mayer--Vietoris 完全系列
\[
\cdots\to H_p(K_1\cap K_2)\to H_p(K_1)\oplus H_p(K_2)
\to H_p(K)\to H_{p-1}(K_1\cap K_2)\to\cdots
\]
から求める。以下 $H_p$ は整数係数のホモロジー群を表す。
$p\ge2$ では $H_p(K_1)\oplus H_p(K_2)=0$ かつ $H_{p-1}(K_1\cap K_2)=0$ $(p\ge3)$ なので
$H_p(K)=0$、すなわち $b_p(K)=0$ $(p\ge3)$ である。
$p=2$ の部分は
\[
0=H_2(K_1)\oplus H_2(K_2)\to H_2(K)\to H_1(K_1\cap K_2)\to H_1(K_1)\oplus H_1(K_2)=0
\]
なので $H_2(K)\cong H_1(K_1\cap K_2)\cong\Z$、よって $b_2(K)=1$ である。
$p=1$ の部分は
\[
0=H_1(K_1)\oplus H_1(K_2)\to H_1(K)\to
H_0(K_1\cap K_2)\xrightarrow{\ \iota\ }H_0(K_1)\oplus H_0(K_2)
\]
であり、$K_1\cap K_2$ は連結だから $\iota$ は $\Z\to\Z\oplus\Z$, $1\mapsto(1,-1)$ で単射である。
従って $H_1(K)=0$、すなわち $b_1(K)=0$ である。
最後に $K$ は連結なので $b_0(K)=1$ である。以上をまとめると
\[
\begin{array}{c|ccc}
&b_0&b_1&b_2\\\hline
K_1\cap K_2&1&1&0\\
K_1&1&0&0\\
K_2&1&0&0\\
K=K_1\cup K_2&1&0&1
\end{array}
\]
であり、$p\ge3$ の Betti 数はいずれも0である。
これは $K$ が $S^2$ と同相であることとも整合する。
\end{enumerate}
\end{proof}

\begin{problem}{第4問｜管状曲面}{2017-s-4}
$a$ は $0<a<1$ を満たす定数とし，写像 $p:\R^2\to\R^3$ を
\[
p(u,v):=\left(u,\ \frac12u^2,\ 0\right)
+a\frac{\cos v}{\sqrt{u^2+1}}(-u,\ 1,\ 0)
+a(\sin v)(0,\ 0,\ 1)
\]
で定める。次の問いに答えよ。
\begin{enumerate}
  \item[(1)] $p$ は曲面であることを示せ。
  \item[(2)] 曲面 $p$ の Gauss 曲率および平均曲率を求めよ。
  \item[(3)] 固定された $u_0\in\R$ に対し $q(s)=p(u_0,s/a)$ で定義される曲線 $q$ は $p$ の測地線であることを示せ。
\end{enumerate}
\end{problem}
\begin{memo*}
\textbf{解答の見通し。}
与式を中心曲線 $c$ の法平面内に半径 $a$ の円を動かした管状曲面として読む。
Frenet 標構で $\partial p/\partial u$, $\partial p/\partial v$ を表せば、正則性と主方向が同時に分かる。
単位法線の微分から二つの主曲率を求め、その積と平均を取る。
$u=u_0$ の円では、単位速度曲線の加速度が曲面法線方向だけになることを示す。
\end{memo*}
\begin{proof}[解答]
(1) 中心曲線を $c(u)=(u,u^2/2,0)$ とし、
\[
w=\norm{c'(u)}=\sqrt{1+u^2},\quad
T=\frac{(1,u,0)}w,\quad
N=\frac{(-u,1,0)}w,\quad B=(0,0,1)
\]
とおく。中心曲線の曲率は
\[
\kappa=\frac{\norm{c'\times c''}}{\norm{c'}^3}
=\frac1{(1+u^2)^{3/2}}
\]
であり、Frenet の公式から $\dfrac{\d N}{\d u}=-\kappa wT$ である。問題の写像は
\[
p(u,v)=c(u)+a\cos v\,N(u)+a\sin v\,B
\]
と書けるので
\[
\frac{\partial p}{\partial u}=w(1-a\kappa\cos v)T,
\qquad \frac{\partial p}{\partial v}=a(-\sin v\,N+\cos v\,B).
\]
$0<a<1$ かつ $0<\kappa\le1$ だから $1-a\kappa\cos v\ge1-a>0$。
従って $\partial p/\partial u$, $\partial p/\partial v$ はともに0でなく互いに直交し、
外積も0でないので正則な曲面である。

(2) 外向き単位法線を $\nu=\cos v\,N+\sin v\,B$ と取ると
\[
\nu_u=-\kappa w\cos v\,T,
\qquad \frac{\partial\nu}{\partial v}=-\sin v\,N+\cos v\,B=\frac1a\frac{\partial p}{\partial v}.
\]
形作用素 $S(X)=-\d\nu(X)$ を $\partial p/\partial u$, $\partial p/\partial v$ に作用させると
\[
S\!\left(\frac{\partial p}{\partial u}\right)
=\frac{\kappa\cos v}{1-a\kappa\cos v}\frac{\partial p}{\partial u},
\qquad S\!\left(\frac{\partial p}{\partial v}\right)=-\frac1a\frac{\partial p}{\partial v}.
\]
従って座標方向は主方向で、主曲率は
\[
k_1=\frac{\kappa\cos v}{1-a\kappa\cos v},\qquad k_2=-\frac1a.
\]
よって積と平均を取れば
\[
K=-\frac{\kappa\cos v}{a(1-a\kappa\cos v)},\qquad
H=\frac12\left(\frac{\kappa\cos v}{1-a\kappa\cos v}-\frac1a\right).
\]
(3) $q(s)=p(u_0,s/a)$ について連鎖律から
\[
q'(s)=\frac1a\frac{\partial p}{\partial v}(u_0,s/a),\qquad
q''(s)=\frac1{a^2}\frac{\partial^2p}{\partial v^2}(u_0,s/a).
\]
$\left\|\partial p/\partial v\right\|=a$ なので $q$ は単位速度であり、
$\dfrac{\partial^2p}{\partial v^2}=-a(\cos v\,N+\sin v\,B)=-a\nu$ だから
\[
q''(s)=-\frac1a\nu(u_0,s/a).
\]
加速度が曲面法線方向だけで接成分が0なので、$q$ は測地線である。
\end{proof}

\begin{problem}{第5問｜扇形上の正則関数}{2017-s-5}
正数 $r$ と $\alpha<\beta<\alpha+2\pi$ を満たす実数 $\alpha,\beta$ に対し，領域 $S$ を
\[
S=\{z\in\mathbb C\mid0<|z|<r,\ \alpha<\arg z<\beta\}
\]
で定める。次の問いに答えよ。
\begin{enumerate}
  \item[(1)] $S$ 上の正則関数 $f(z)$ で，$\displaystyle\lim_{\substack{z\to0\\z\in S}}f(z)$ は存在するが $\displaystyle\lim_{\substack{z\to0\\z\in S}}f'(z)$ は存在しないものの例を挙げよ。ただし $S$ 上定義された関数 $g(z)$ に対し，$z\in S$ を保ちながら $z\to0$ としたときの $g(z)$ の極限値を $\displaystyle\lim_{\substack{z\to0\\z\in S}}g(z)$ で表す。
  \item[(2)] $r'<r$ を満たす正数 $r'$ と $\alpha<\alpha'<\beta'<\beta$ を満たす実数 $\alpha',\beta'$ に対し，
  \[
  T=\{z\in\mathbb C\mid0<|z|\le r',\ \alpha'\le\arg z\le\beta'\}
  \]
  という $S$ の部分集合を考える。$T$ の任意の点 $z$ に対して，$z$ を中心とする円で $S$ に含まれるものが存在することを示せ。
  \item[(3)] $f(z)$ を $S$ 上正則かつ有界な関数とする。$r-\varepsilon>0$, $\alpha+\varepsilon<\beta-\varepsilon$ を満たす任意の正数 $\varepsilon$ に対して，ある正数 $M_\varepsilon$ が存在して，
  \[
  |f'(z)|\le\frac{M_\varepsilon}{|z|}
  \]
  がすべての $z\in T_\varepsilon$ に対して成立することを示せ。ただし $T_\varepsilon$ は
  \[
  T_\varepsilon=\{z\in\mathbb C\mid0<|z|\le r-\varepsilon,\ \alpha+\varepsilon\le\arg z\le\beta-\varepsilon\}
  \]
  で定まる $S$ の部分集合である。
\end{enumerate}
\end{problem}
\begin{memo*}
\textbf{解答の見通し。}
角度が $2\pi$ 未満なので対数の枝を取り、平方根で (1) の例を作る。
境界から角度 $\varepsilon$ だけ離れた点 $z$ では、
半径を $c_\varepsilon|z|$ とする円板を扇形内に一様に取れる。
その円周上で Cauchy の微分公式を使うと、半径の逆数 $1/|z|$ が現れる。
\end{memo*}
\begin{proof}[解答]
\begin{enumerate}
\item[(1)] 扇形の角度は $2\pi$ 未満なので、$S$ 上で正則な対数の枝を取れる。
\[
f(z)=\exp\left(\frac12\log z\right)
\]
とおけば $|f(z)|=|z|^{1/2}\to0$ である。一方
\[
f'(z)=\frac12\exp\left(-\frac12\log z\right),\qquad
|f'(z)|=\frac1{2\sqrt{|z|}}\longrightarrow\infty,
\]
従って $f'$ は有限な極限を持たない。

\item[(2)] $z\in T$ とすると $0<|z|\le r'<r$ であり、
$\alpha<\alpha'\le\arg z\le\beta'<\beta$ だから $z\in S$ である。
$S$ は開集合なので、開集合の定義により各 $z$ に対してある $\rho_z>0$ が存在して
$z$ を中心とする円 $B(z,\rho_z)$ が $S$ に含まれる。

\item[(3)] $\varepsilon'\le\varepsilon$ なら $T_\varepsilon\subset T_{\varepsilon'}$ なので、
$\varepsilon$ を小さく取り替えて $\varepsilon\le\pi/2$ としてよい。
$T_\varepsilon$ の点は二つの境界半直線から角度で少なくとも $\varepsilon$ 離れ、
動径も $r-\varepsilon$ 以下である。そこで
\[
0<c_\varepsilon<\min\left\{1,\ \sin\varepsilon,\ \frac{\varepsilon}{r-\varepsilon}\right\}
\]
を取ると、任意の $z\in T_\varepsilon$ に対し
\[
B(z,c_\varepsilon|z|)\subset S
\]
が成り立つ。実際、半径 $c_\varepsilon|z|$ は $|z|$ 未満なので円板は0を含まない。
また境界半直線までの距離は $|z|\sin\varepsilon$ 以上なので円板はそれらを越えず、
円板内の $w$ は
\[
|w|<(1+c_\varepsilon)|z|\le(1+c_\varepsilon)(r-\varepsilon)<r
\]
を満たす。

$|f|\le B$ とし、円 $|\zeta-z|=c_\varepsilon|z|$ 上で Cauchy の微分公式を使うと
\[
f'(z)=\frac1{2\pi i}\int_{|\zeta-z|=c_\varepsilon|z|}
\frac{f(\zeta)}{(\zeta-z)^2}\,\d\zeta.
\]
従って円周の長さも含めて評価すると
\[
|f'(z)|\le
\frac1{2\pi}\frac{B}{(c_\varepsilon|z|)^2}
2\pi c_\varepsilon|z|
=\frac{B}{c_\varepsilon|z|}.
\]
$M_\varepsilon=B/c_\varepsilon$ とおけば求める評価を得る。
\end{enumerate}
\end{proof}

\begin{problem}{第6問｜線形微分方程式}{2017-s-6}
次の問いに答えよ。
\begin{enumerate}
  \item[(1)] 微分方程式の初期値問題
  \[
  \begin{cases}
  \dfrac{\d^2y}{\d x^2}+2\dfrac{\d y}{\d x}+2y=0,\\[1mm]
  y(0)=1,\quad\dfrac{\d y}{\d x}(0)=0
  \end{cases}
  \]
  の解 $y(x)$ を求めよ。また，$y(x)=0$ を満たす $x$ をすべて求めよ。
  \item[(2)] 微分方程式の初期値問題
  \[
  \begin{cases}
  \dfrac{\d^2y}{\d x^2}+2y=\cos(\omega x),\\[1mm]
  y(0)=0,\quad\dfrac{\d y}{\d x}(0)=1
  \end{cases}
  \]
  の解 $y(x)$ が $\displaystyle\sup_{x>0}|y(x)|=\infty$ を満たすように定数 $\omega$ を定め，またそのときの解 $y(x)$ を求めよ。
\end{enumerate}
\end{problem}
\begin{memo*}
\textbf{解答の見通し。}
(1) は複素特性根から減衰振動の形を作り、三角関数の位相をずらして零点を読む。
(2) は外力の角振動数 $|\omega|$ が固有角振動数 $\sqrt2$ と一致するときだけ共振する。
共振時は通常の三角関数に $x$ を掛けた特殊解を選び、初期条件を合わせる。
\end{memo*}
\begin{proof}[解答]
\begin{enumerate}
\item[(1)] 特性方程式は
\[
r^2+2r+2=0,\qquad r=-1\pm i
\]
なので一般解は $y=e^{-x}(C_1\cos x+C_2\sin x)$。
$y(0)=1$ から $C_1=1$、
\[
y'(0)=-C_1+C_2=0
\]
から $C_2=1$。従って
\[
y=e^{-x}(\cos x+\sin x),\qquad x=-\frac\pi4+k\pi\quad(k\in\Z).
\]
最後の零点は $e^{-x}\ne0$ と
$\cos x+\sin x=\sqrt2\sin(x+\pi/4)$ から得られる。
\item[(2)] $\omega^2\ne2$ なら特殊解 $\cos(\omega x)/(2-\omega^2)$ が取れ、
同次解も三角関数なので全ての解は有界である。
$\omega=\pm\sqrt2$ のときは共振し、$a=\sqrt2$ とおけば
\[
y_p=\frac{x}{2a}\sin ax
\]
と置く。このとき
\[
y_p'=\frac1{2a}\sin ax+\frac x2\cos ax,
\]
\[
y_p''=\frac12\cos ax+\frac12\cos ax-\frac{ax}{2}\sin ax
=\cos ax-\frac{ax}{2}\sin ax.
\]
従って
\[
y_p''+a^2y_p
=\cos ax-\frac{ax}{2}\sin ax+\frac{ax}{2}\sin ax
=\cos ax.
\]
初期条件を満たす同次解 $(1/a)\sin ax$ を加えて
\[
y(x)=\frac{x+2}{2\sqrt2}\sin(\sqrt2x).
\]
これは $x>0$ で非有界である。従って求める $\omega$ は $\omega=\pm\sqrt2$。
\end{enumerate}
\end{proof}

\begin{problem}{第7問｜パラメータ積分}{2017-s-7}
$f$ は $(0,\infty)$ 上の Lebesgue 可測関数で，
\[
\int_0^\infty x|f(x)|\,\d x<\infty
\]
を満たしているとする。このとき
\[
\lim_{a\to+0}\frac1a\int_0^\infty(\log(ax+1)-ax)f(x)\,\d x
\]
を求めよ。
\end{problem}
\begin{memo*}
\textbf{解答の見通し。}
固定した $x$ では $\log(1+ax)-ax=o(a)$ なので被積分関数は0へ収束する。
積分と極限を交換するには、$0\le u-\log(1+u)\le u$ を使って
絶対値を仮定中の可積分関数 $x|f(x)|$ で抑える。
\end{memo*}
\begin{proof}[解答]
$u=ax\ge0$ とおく。$a\downarrow0$ のとき $u\to0$ であり、
$\log(1+u)=u+o(u)$ だから、各 $x>0$ について
\[
\frac{\log(1+ax)-ax}{a}
=x\frac{\log(1+u)-u}{u}\longrightarrow0.
\]
また $\log(1+u)\le u$ であり、$\log(1+u)\ge0$ でもあるので
\[
0\le u-\log(1+u)\le u.
\]
従って
\[
\left|\frac{\log(1+ax)-ax}{a}f(x)\right|
\le x|f(x)|.
\]
仮定より右辺は $(0,\infty)$ 上可積分である。よって優収束定理により積分と極限を交換でき、求める極限は $0$ である。
\end{proof}

\begin{problem}{第8問｜確率収束}{2017-s-8}
$\{X_n\}_{n=1}^\infty$ を実数値確率変数列，$X$ を実数値確率変数とする。次の問いに答えよ。
\begin{enumerate}
  \item[(1)] $\displaystyle\lim_{N\to\infty}P(|X|>N)=0$ が成り立つことを示せ。
  \item[(2)] $\{X_n\}_{n=1}^\infty$ が $X$ に確率収束するとき，$\R$ 上の任意の連続関数 $f$ に対して，$\{f(X_n)\}_{n=1}^\infty$ は $f(X)$ に確率収束することを示せ。
\end{enumerate}
\end{problem}
\begin{memo*}
\textbf{解答の見通し。}
まず実確率変数 $X$ が大きなコンパクト区間の外へ出る確率を小さくできることを示す。
その区間を少し広げた上で連続関数 $f$ の一様連続性を使う。
$f(X_n)$ の誤差事象を「$X$ が区間外」と「$X_n-X$ が大きい」の和集合に含めれば、
二つの確率を別々に小さくできる。
\end{memo*}
\begin{proof}[解答]
\begin{enumerate}
\item[(1)] $A_N=\{|X|>N\}$ とおくと
\[
A_1\supset A_2\supset\cdots,
\qquad \bigcap_{N=1}^{\infty}A_N=\varnothing
\]
である。確率測度の上からの連続性より
\[
\lim_{N\to\infty}P(|X|>N)
=\lim_{N\to\infty}P(A_N)
=P\left(\bigcap_{N=1}^{\infty}A_N\right)=0.
\]

\item[(2)] 任意の $\varepsilon>0$ と $\eta>0$ を取る。(1) より、十分大きな $M$ を取れば
\[
P(|X|>M)<\frac\eta2
\]
となる。$f$ はコンパクト区間 $[-M-1,M+1]$ 上で一様連続なので、ある $0<\delta\le1$ が存在して
\[
x,y\in[-M-1,M+1],\quad |x-y|<\delta
\quad\Longrightarrow\quad |f(x)-f(y)|<\varepsilon
\]
となる。

$|X|\le M$ かつ $|X_n-X|<\delta$ なら $|X_n|\le M+1$ であり、上の一様連続性が使える。従って事象の包含
\[
\{|f(X_n)-f(X)|\ge\varepsilon\}
\subset
\{|X|>M\}\cup\{|X_n-X|\ge\delta\}
\]
が成り立つので
\[
P(|f(X_n)-f(X)|\ge\varepsilon)
\le P(|X|>M)+P(|X_n-X|\ge\delta),
\]
を得る。$X_n\to X$ は確率収束するから、十分大きな $n$ では第2項も $\eta/2$ 未満である。よって左辺は $\eta$ 未満となり、$f(X_n)\to f(X)$ は確率収束する。
\end{enumerate}
\end{proof}
