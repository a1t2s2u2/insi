\begin{problem}{第1問｜実行列}{2022-1-1}
$n$ 次単位行列を $E_n$ とする。
\begin{enumerate}
  \item[(1)] 実行列 $A$ が $A^2=-E_n$ を満たすなら $n$ は偶数であることを示し，$n=2$ の例を挙げよ。
  \item[(2)] ${}^{\mathsf T}AA=-E_n$ を満たす実行列は存在しないことを示せ。
  \item[(3)] $A\ne O$ が ${}^{\mathsf T}A=-A$ を満たすとき，$P^{-1}AP$ を対角行列にする実正則行列 $P$ は存在しないことを示せ。
\end{enumerate}
\end{problem}
\begin{proof}[解答]
\begin{enumerate}
  \item[(1)] $A^2=-E_n$ より $A$ は正則である。両辺の行列式を取ると
  \[
  (\det A)^2=\det(A^2)=\det(-E_n)=(-1)^n.
  \]
  $A$ は正則だから $(\det A)^2>0$ であり、従って $(-1)^n=1$ でなければならない。
  よって $n$ は偶数である。$n=2$ では
  \[
  J=\begin{pmatrix}0&-1\\1&0\end{pmatrix},\qquad
  J^2=
  \begin{pmatrix}0&-1\\1&0\end{pmatrix}
  \begin{pmatrix}0&-1\\1&0\end{pmatrix}
  =\begin{pmatrix}-1&0\\0&-1\end{pmatrix}=-E_2
  \]
  が一例である。

  \item[(2)] $A^{\mathsf T}A=-E_n$ を満たす実行列 $A$ が存在すると仮定する。
  $0\ne x\in\R^n$ を取って両辺を二次形式で評価すると
  \[
  x^{\mathsf T}A^{\mathsf T}Ax
  =(Ax)^{\mathsf T}(Ax)=\norm{Ax}^2\ge0
  \]
  である一方、
  \[
  x^{\mathsf T}(-E_n)x=-x^{\mathsf T}x=-\norm{x}^2<0
  \]
  となる。これは矛盾であるから、そのような実行列は存在しない。

  \item[(3)] $A^{\mathsf T}=-A$ とする。任意の $v\in\R^n$ に対して
  $v^{\mathsf T}Av$ は $1\times1$ 行列、すなわち実数なので、転置しても値は変わらない。従って
  \[
  v^{\mathsf T}Av
  =(v^{\mathsf T}Av)^{\mathsf T}
  =v^{\mathsf T}A^{\mathsf T}v
  =-v^{\mathsf T}Av,
  \]
  よって $v^{\mathsf T}Av=0$ である。

  もし実正則行列 $P$ により $P^{-1}AP=D$ が実対角行列になるなら、$D$ の各対角成分は
  $A$ の実固有値である。実固有値 $\lambda$ と対応する実固有ベクトル $0\ne v$ に対して
  \[
  0=v^{\mathsf T}Av=v^{\mathsf T}(\lambda v)
  =\lambda\norm v^2
  \]
  だから $\lambda=0$ である。従って $D=O$ となり
  \[
  A=PDP^{-1}=O
  \]
  となるが、これは $A\ne O$ に矛盾する。よってそのような $P$ は存在しない。
\end{enumerate}
\end{proof}

\begin{problem}{第2問｜Cauchy の関数方程式}{2022-1-2}
$f:\R\to\R$ が $f(x+y)=f(x)+f(y)$ を満たす。
\begin{enumerate}
  \item[(1)] $f(0)=0$ と $f(-x)=-f(x)$ を示せ。
  \item[(2)] $r\in\Q$ に対して $f(rx)=rf(x)$ を示せ。
  \item[(3)] $f$ が $\R$ 上連続なら，ある $c\in\R$ により $f(x)=cx$ と書けることを示せ。
\end{enumerate}
\end{problem}
\begin{proof}[解答]
\begin{enumerate}
  \item[(1)] $f(0)=f(0+0)=2f(0)$ より $f(0)=0$。
  また $0=f(0)=f(x+(-x))=f(x)+f(-x)$ より $f(-x)=-f(x)$。
  \item[(2)] $n\in\N$ なら帰納法で $f(nx)=nf(x)$。負の整数については (1) を使う。
  $n\ge1$ に対し $f(x)=f(n(x/n))=nf(x/n)$ だから $f(x/n)=f(x)/n$。
  従って $r=m/n\in\Q$ に対し
  \[
  f(rx)=f\left(m\frac{x}{n}\right)=m f(x/n)=\frac mn f(x)=rf(x).
  \]
  \item[(3)] $c=f(1)$ とおくと (2) より $f(r)=cr$ $(r\in\Q)$。
  任意の $x\in\R$ に対し $r_n\in\Q$, $r_n\to x$ を取る。
  連続性から
  \[
  f(x)=f\left(\lim_{n\to\infty}r_n\right)
  =\lim_{n\to\infty}f(r_n)
  =\lim_{n\to\infty}cr_n=cx.
  \]
\end{enumerate}
\end{proof}

\begin{problem}{第3問｜連続写像の反例}{2022-1-3}
連続写像 $f:X\to Y$ に関する次の三命題が成り立たないことを，反例で示せ。
\begin{enumerate}
  \item[(1)] 開集合 $U\subset X$ の像 $f(U)$ は $Y$ で開である。
  \item[(2)] コンパクト集合 $K\subset Y$ の逆像 $f^{-1}(K)$ は $X$ でコンパクトである。
  \item[(3)] $f$ が全射なら，稠密集合 $V\subset Y$ の逆像 $f^{-1}(V)$ は $X$ で稠密である。
\end{enumerate}
\end{problem}
\begin{proof}[解答]
\begin{enumerate}
  \item[(1)] $X=Y=\R$ とし、定数写像
  \[
  f:\R\to\R,qquad f(x)=0
  \]
  を考える。任意の開集合 $O\subset\R$ に対して
  \[
  f^{-1}(O)=
  \begin{cases}
  \R &(0\in O),\\
  \varnothing &(0\notin O)
  \end{cases}
  \]
  だから $f$ は連続である。しかし $U=\R$ は $X$ の開集合である一方、
  $f(U)=\{0\}$ は $Y=\R$ の開集合ではない。

  \item[(2)] (1) と同じ定数写像を使う。$K=\{0\}\subset\R$ は有限集合なのでコンパクトであるが
  \[
  f^{-1}(K)=f^{-1}(\{0\})=\R
  \]
  はコンパクトでない。実際、開被覆 $\{(-n,n)\}_{n\ge1}$ は有限部分被覆を持たない。

  \item[(3)] $X=\R\sqcup\{p\}$ に直和位相を入れ、$Y=\R$ とする。写像 $f:X\to Y$ を
  \[
  f(x)=x\quad(x\in\R),\qquad f(p)=0
  \]
  と定める。$f|_{\R}$ が恒等写像なので $f$ は全射である。
  $Y$ の開集合 $O$ に対して
  \[
  f^{-1}(O)=
  \begin{cases}
  O\sqcup\{p\} &(0\in O),\\
  O &(0\notin O)
  \end{cases}
  \]
  であり、どちらも直和位相で開だから $f$ は連続である。

  $V=\R\setminus\{0\}$ とすると $\overline V=\R$ なので $V$ は $Y$ で稠密である。しかし
  \[
  f^{-1}(V)=\R\setminus\{0\}\subset X
  \]
  は $X$ の非空開集合 $\{p\}$ と交わらない。従って
  $\overline{f^{-1}(V)}\ne X$ であり、$f^{-1}(V)$ は $X$ で稠密でない。
\end{enumerate}
\end{proof}

\begin{problem}{第1問｜有限群}{2022-s-1}
$G=S_5$，$P=\langle(1\ 2\ 3\ 4\ 5)\rangle$ とする。また $\Omega$ を $G$ の Sylow $5$ 部分群全体とする。
\begin{enumerate}
  \item[(1)] 位数5の巡回群の自己同型群の位数を求めよ。
  \item[(2)] 正規化群 $N_G(P)$ の Sylow $2$ 部分群を一つ求めよ。
  \item[(3)] 共役作用 $\varphi:G\to\operatorname{Sym}(\Omega)$ が準同型であることを示せ。
  \item[(4)] $S_6$ に，指数6で互いに共役でない二つの部分群があることを示せ。
\end{enumerate}
\end{problem}
\begin{proof}[解答]
\begin{enumerate}
  \item[(1)] 生成元の像は4通りなので $|\operatorname{Aut}(C_5)|=4$。
  \item[(2)] $a=(1\ 2\ 3\ 4\ 5)$，$b=(2\ 3\ 5\ 4)$ とすれば $bab^{-1}=a^2$。よって $\langle b\rangle$ は位数4で，$|N_G(P)|=5\cdot4=20$ の Sylow $2$ 部分群。
  \item[(3)] $\varphi(gh)(Q)=ghQ(gh)^{-1}=\varphi(g)(\varphi(h)(Q))$。
  \item[(4)] Sylow の定理から $|\Omega|=6$。作用の核は $N_G(P)$ に含まれる。$S_5$ の正規部分群のうち $N_G(P)$ に入るのは $\{1\}$ だけなので，作用は忠実。従って $\varphi(S_5)\subset S_6$ は指数6で，6点に推移的に作用する。一方，通常の点固定部分群 $S_5\subset S_6$ は固定点を持つ。共役で軌道構造は変わらないから，両者は共役でない。
\end{enumerate}
\end{proof}

\begin{problem}{第2問｜球面の面積形式}{2022-s-2}
$S^2=\{p\in\R^3\mid\norm p=1\}$ とし，接ベクトル $u,v\in T_pS^2$ に対して
$\omega_p(u,v)=\inner p{u\times v}$ と定める。
\begin{enumerate}
  \item[(1)] $\omega_p(u,v)=-\omega_p(v,u)$ を示せ。また $\omega_p(u,v)=0$ が全ての $v$ に対して成り立つなら $u=0$ であることを示せ。
  \item[(2)] $U=S^2\setminus\{(x,0,z)\in S^2\mid x\ge0\}$ とする。$0<\theta<2\pi$，$-1<h<1$ に対し $(r\cos\theta,r\sin\theta,h)\in U$ となる $r$ を求めよ。
  \item[(3)] $T(\theta,h)=(r\cos\theta,r\sin\theta,h)$ とおくとき，$\omega_{T(\theta,h)}(T_\theta,T_h)$ を求めよ。
\end{enumerate}
\end{problem}
\begin{proof}[解答]
\begin{enumerate}
  \item[(1)] 外積の反対称性 $v\times u=-(u\times v)$ より
  \[
  \omega_p(v,u)=\inner p{v\times u}
  =-\inner p{u\times v}=-\omega_p(u,v).
  \]
  次に $\omega_p(u,v)=0$ が全ての $v\in T_pS^2$ に対して成り立つと仮定する。
  $u\in T_pS^2$ だから $\inner pu=0$ であり、$v=p\times u$ も $p$ と直交するので
  $v\in T_pS^2$ である。ベクトル三重積の公式を用いると
  \[
  u\times(p\times u)
  =p\inner uu-u\inner up
  =\norm u^2p.
  \]
  従って
  \[
  0=\omega_p(u,p\times u)
  =\inner p{\norm u^2p}
  =\norm u^2\norm p^2=\norm u^2.
  \]
  よって $u=0$ である。

  \item[(2)] 点 $(r\cos\theta,r\sin\theta,h)$ が $S^2$ 上にある条件は
  \[
  r^2\cos^2\theta+r^2\sin^2\theta+h^2=1,
  \]
  すなわち $r^2=1-h^2$ である。$-1<h<1$ では右辺は正であり、
  $r$ を緯線の半径として正に取るので
  \[
  r(\theta,h)=\sqrt{1-h^2}.
  \]
  $r$ は $\theta$ に依存しない。また $0<\theta<2\pi$ としたことで、除いた半大円上の点は現れない。

  \item[(3)] $r=\sqrt{1-h^2}$ だから $\partial r/\partial h=-h/r$ であり
  \[
  T_\theta=(-r\sin\theta,r\cos\theta,0),\qquad
  T_h=\left(-\frac h r\cos\theta,-\frac h r\sin\theta,1\right).
  \]
  外積を成分ごとに計算すると
  \begin{align*}
  T_\theta\times T_h
  &=\left(r\cos\theta,r\sin\theta,
    h(\sin^2\theta+\cos^2\theta)\right)\\
  &=(r\cos\theta,r\sin\theta,h)=T(\theta,h).
  \end{align*}
  従って $\omega_T(T_\theta,T_h)=\inner{T}{T}=1$。
\end{enumerate}
\end{proof}

\begin{problem}{第3問｜微分方程式}{2022-s-3}
$f$ を $\R$ 上有界連続とし，$y''-y=f(x)$ を考える。
\begin{enumerate}
  \item[(1)] $f=0$ のときの一般解を求めよ。
  \item[(2)] $e^{-x}f(x)$ が任意の $[a,\infty)$ 上で広義積分可能であることを示せ。
  \item[(3)] 次で定める $g$ を考える：
  \[
  g(x)=-\frac{e^x}{2}\int_x^\infty e^{-t}f(t)\,\d t
  -\frac{e^{-x}}2\int_{-\infty}^x e^tf(t)\,\d t
  \]
  が方程式の解であることを示せ。
  \item[(4)] $f(x)\to1$ $(x\to\infty)$ なら $g(x)\to-1$ を示せ。
  \item[(5)] $f(x)=1-e^{-x}$ $(x\ge0)$，$f(x)=0$ $(x<0)$ のとき，$y(x)\to-1$ $(x\to\infty)$ を満たす解を全て求めよ。
\end{enumerate}
\end{problem}
\begin{proof}[解答]
\begin{enumerate}
  \item[(1)] $y=C_1e^x+C_2e^{-x}$。
  \item[(2)] $|f|\le M$ とすれば $\int_a^\infty|e^{-x}f(x)|\,\d x\le Me^{-a}$。
  \item[(3)]
  \[
  A(x)=\int_x^\infty e^{-t}f(t)\,\d t,\qquad
  B(x)=\int_{-\infty}^x e^tf(t)\,\d t
  \]
  とおく。微積分学の基本定理より
  \[
  A'(x)=-e^{-x}f(x),\qquad B'(x)=e^xf(x).
  \]
  $g=-(e^xA+e^{-x}B)/2$ を積の微分法で微分すると
  \begin{align*}
  g'
  &=-\frac12\{e^xA+e^xA'-e^{-x}B+e^{-x}B'\}\\
  &=-\frac12e^xA+\frac12e^{-x}B,
  \end{align*}
  ここでは $e^xA'=-f$ と $e^{-x}B'=f$ が打ち消し合った。もう一度微分して
  \begin{align*}
  g''
  &=-\frac12(e^xA+e^xA')
    +\frac12(-e^{-x}B+e^{-x}B')\\
  &=-\frac12(e^xA+e^{-x}B)+f(x)\\
  &=g(x)+f(x).
  \end{align*}
  従って $g''-g=f$ である。
  \item[(4)] $h=f-1$ とおくと $h(x)\to0$ で、$h$ は有界である。
  定数関数 $1$ を上の式に代入して得られる解は
  \[
  -\frac{e^x}{2}\int_x^\infty e^{-t}\,\d t
  -\frac{e^{-x}}2\int_{-\infty}^x e^t\,\d t=-1.
  \]
  一方、$h$ が作る第1項は変数変換 $t=x+s$ により
  \[
  e^x\int_x^\infty e^{-t}h(t)\,\d t
  =\int_0^\infty e^{-s}h(x+s)\,\d s\longrightarrow0,
  \]
  第2項も $t=x-s$ により
  \[
  e^{-x}\int_{-\infty}^x e^th(t)\,\d t
  =\int_0^\infty e^{-s}h(x-s)\,\d s\longrightarrow0.
  \]
  いずれも有界性から $Ce^{-s}$ を優関数にして優収束定理を使った。
  従って $g(x)\to-1$。
  \item[(5)] $x\ge0$ では、定数項 $1$ の特殊解は $-1$。
  また $-e^{-x}$ は同次解 $e^{-x}$ と共振するので
  $y_p=Cxe^{-x}$ とおくと
  \[
  y_p''-y_p=-2Ce^{-x}.
  \]
  よって $C=1/2$。$x\to\infty$ で $-1$ に収束するには $e^x$ の項があってはならないから
  \[
  y(x)=-1+\left(D+\frac x2\right)e^{-x}\qquad(x\ge0).
  \]
  $x<0$ では $f=0$ なので $y=Be^x+Ce^{-x}$。
  $x=0$ で $y,y'$ を一致させると
  \[
  B+C=-1+D,\qquad B-C=\frac12-D,
  \]
  従って $B=-1/4$, $C=D-3/4$。以上より、任意の $D\in\R$ に対し
  \[
  y(x)=\begin{cases}
  -\frac14e^x+(D-\frac34)e^{-x},&x<0,\\
  -1+(D+\frac x2)e^{-x},&x\ge0
  \end{cases}
  \]
  であり、これが全てである。
\end{enumerate}
\end{proof}

\begin{problem}{第4問｜Dirichlet 積分}{2022-s-4}
$t>0$ とする。
\begin{enumerate}
  \item[(1)] $\sin x/x$ が $(0,t)$ 上でルベーグ積分可能であることを示せ。
  \item[(2)] $y>0$ に対し
  \[
  \int_0^t e^{-xy}\sin x\,\d x
  =\frac1{1+y^2}-\frac{e^{-ty}}{1+y^2}(y\sin t+\cos t)
  \]
  を示せ。
  \item[(3)] 次の等式を示せ：
  \[
  \int_0^t\frac{\sin x}{x}\,\d x
  =\int_0^\infty\frac{\d y}{1+y^2}
  -\int_0^\infty\frac{e^{-ty}}{1+y^2}(y\sin t+\cos t)\,\d y
  \]
  \item[(4)] $\displaystyle\lim_{t\to\infty}\int_0^t\frac{\sin x}{x}\,\d x$ を求めよ。
\end{enumerate}
\end{problem}
\begin{proof}[解答]
\begin{enumerate}
  \item[(1)] $\lim_{x\to0}\sin x/x=1$ なので、$x=0$ で値を1と定めれば $[0,t]$ 上連続になる。
  従ってコンパクト区間上でルベーグ積分可能である。
  \item[(2)]
  \[
  F(x)=\frac{e^{-xy}(-y\sin x-\cos x)}{1+y^2}
  \]
  とおく。積の微分法により
  \begin{align*}
  F'(x)
  &=\frac{e^{-xy}}{1+y^2}
  \{(-y)(-y\sin x-\cos x)-y\cos x+\sin x\}\\
  &=e^{-xy}\sin x.
  \end{align*}
  よって
  \begin{align*}
  \int_0^t e^{-xy}\sin x\,\d x
  &=F(t)-F(0)\\
  &=\frac1{1+y^2}
   -\frac{e^{-ty}}{1+y^2}(y\sin t+\cos t).
  \end{align*}
  \item[(3)] $x>0$ では
  \[
  \frac1x=\int_0^\infty e^{-xy}\,\d y.
  \]
  また
  \[
  \int_0^t\int_0^\infty|e^{-xy}\sin x|\,\d y\d x
  =\int_0^t\frac{|\sin x|}{x}\,\d x<\infty
  \]
  なので Fubini の定理で積分順序を交換できる。(2) を用いて
  \begin{align*}
  \int_0^t\frac{\sin x}{x}\,\d x
  &=\int_0^\infty\int_0^t e^{-xy}\sin x\,\d x\d y\\
  &=\int_0^\infty\frac{\d y}{1+y^2}
  -\int_0^\infty\frac{e^{-ty}}{1+y^2}(y\sin t+\cos t)\,\d y.
  \end{align*}
  \item[(4)] 第2積分の絶対値は
  \[
  \int_0^\infty e^{-ty}(y+1)\,\d y
  =\frac1{t^2}+\frac1t\longrightarrow0
  \]
  以下である。一方
  \[
  \int_0^\infty\frac{\d y}{1+y^2}
  =[\arctan y]_0^\infty=\frac\pi2.
  \]
  従って求める極限は $\pi/2$。
\end{enumerate}
\end{proof}

\section{第2期・専門基礎科目（3題必修）}

\begin{problem}{第1問｜位相空間}{2022-2-1}
\begin{enumerate}
  \item[(1)] 位相空間 $X$ の部分集合 $A,B$ に対し $\overline{A\cup B}=\overline A\cup\overline B$ を示せ。
  \item[(2)] Hausdorff 空間のコンパクト部分集合は閉集合であることを示せ。
  \item[(3)] 全射連続写像 $F:S\to T$ について，$S$ が連結なら $T$ も連結であることを示せ。
\end{enumerate}
\end{problem}
\begin{proof}[解答]
\begin{enumerate}
\item[(1)] 閉包 $\overline{A\cup B}$ は $A\cup B$ を含む最小の閉集合である。
$\overline A\cup\overline B$ は閉集合であり $A\cup B$ を含むので
\[
\overline{A\cup B}\subset\overline A\cup\overline B.
\]
逆に $A\subset A\cup B$ と $B\subset A\cup B$ に閉包の単調性を用いると
\[
\overline A\subset\overline{A\cup B},\qquad
\overline B\subset\overline{A\cup B}.
\]
従って $\overline A\cup\overline B\subset\overline{A\cup B}$ であり、両方の包含から
\[
\overline{A\cup B}=\overline A\cup\overline B
\]
を得る。

\item[(2)] $K\subset X$ をコンパクト部分集合とし、$x\in X\setminus K$ を取る。
$X$ は Hausdorff 空間なので、各 $y\in K$ に対して互いに交わらない開近傍
$U_y\ni x$ と $V_y\ni y$ を取れる。$\{V_y\}_{y\in K}$ は $K$ の開被覆だから、
コンパクト性より $y_1,\ldots,y_m\in K$ が存在して
\[
K\subset V_{y_1}\cup\cdots\cup V_{y_m}
\]
となる。ここで
\[
U=U_{y_1}\cap\cdots\cap U_{y_m}
\]
とおくと、$U$ は $x$ の開近傍であり $U\cap K=\varnothing$ である。
従って $X\setminus K$ は開集合であり、$K$ は閉集合である。

\item[(3)] $T$ が連結でないと仮定すると、非空な開集合 $U,V\subset T$ により
\[
T=U\sqcup V
\]
と書ける。$F$ は連続なので $F^{-1}(U),F^{-1}(V)$ は $S$ の開集合であり、全射なのでどちらも非空である。さらに
\[
S=F^{-1}(T)=F^{-1}(U)\sqcup F^{-1}(V)
\]
となり、$S$ が連結であることに矛盾する。従って $T$ は連結である。
\end{enumerate}
\end{proof}

\begin{problem}{第2問｜微分積分}{2022-2-2}
$a<b$，$u$ は $[a,b]^2$ 上連続とする。
\begin{enumerate}
  \item[(1)] $\displaystyle f(x)=\int_a^x\d s\int_a^s u(s,t)\,\d t$ が $(a,b)$ で微分可能で，$\displaystyle f'(x)=\int_a^x u(x,t)\,\d t$ となることを示せ。
  \item[(2)] $x>0$，$D_x=\{(s,t)\mid0\le s\le t\le x\}$ とし
  $\displaystyle g(x)=\iint_{D_x}s(1-2st)\,\d s\d t$ とおく。$x>0$ における最大値を求めよ。
\end{enumerate}
\end{problem}
\begin{proof}[解答]
\begin{enumerate}
\item[(1)] $F(s)=\int_a^s u(s,t)\,\d t$ とおく。
$u$ の連続性から $F$ も連続であり $f(x)=\int_a^xF(s)\,\d s$ だから、微積分学の基本定理より
\[
f'(x)=F(x)=\int_a^x u(x,t)\,\d t.
\]
\item[(2)] 積分領域は $0\le s\le t\le x$ なので
\begin{align*}
g(x)
&=\int_0^x\int_0^t (s-2s^2t)\,\d s\d t\\
&=\int_0^x\left[\frac{s^2}{2}-\frac{2ts^3}{3}\right]_{s=0}^{s=t}\d t\\
&=\int_0^x\left(\frac{t^2}{2}-\frac{2t^4}{3}\right)\d t\\
&=\frac{x^3}{6}-\frac{2x^5}{15}.
\end{align*}
従って
\[
g'(x)=\frac{x^2}{2}-\frac{2x^4}{3}
=x^2\left(\frac12-\frac{2x^2}{3}\right).
\]
$x>0$ では $x=\sqrt3/2$ の前で正、後で負であり、$g(x)\to-\infty$ $(x\to\infty)$ だからここで最大になる。
\[
g\left(\frac{\sqrt3}{2}\right)
=\frac{\sqrt3}{16}-\frac{3\sqrt3}{80}
=\frac{\sqrt3}{40}.
\]
\end{enumerate}
\end{proof}

\begin{problem}{第3問｜線形空間}{2022-2-3}
\begin{enumerate}
  \item[(1)] 部分空間 $W_1,W_2\subset V$ に対し $W_1\cap W_2$ は部分空間であること，また $W_1\cup W_2$ が部分空間なら $W_1\subset W_2$ または $W_2\subset W_1$ であることを示せ。
  \item[(2)] 線形写像 $f:U\to V$ に対し $\ker f$ は部分空間であり，$f$ が単射であることと $\ker f=\{0\}$ が同値であることを示せ。
  \item[(3)] $M_3(\R)$ の交代行列全体 $W=\{A\mid A^{\mathsf T}=-A\}$ の基底と次元を求めよ。
\end{enumerate}
\end{problem}
\begin{proof}[解答]
\begin{enumerate}
\item[(1)] $W_1,W_2$ は部分空間なので $0\in W_1\cap W_2$ である。また
$x,y\in W_1\cap W_2$、$a,b$ をスカラーとすると、$x,y$ は $W_1,W_2$ の両方に属するから
\[
ax+by\in W_1,\qquad ax+by\in W_2.
\]
従って $ax+by\in W_1\cap W_2$ であり、$W_1\cap W_2$ は部分空間である。

次に $W_1\cup W_2$ が部分空間であると仮定する。
$W_1\not\subset W_2$ かつ $W_2\not\subset W_1$ なら、
\[
u\in W_1\setminus W_2,
\qquad v\in W_2\setminus W_1
\]
を取れる。合併が部分空間なら $u+v\in W_1\cup W_2$ である。
もし $u+v\in W_1$ なら $v=(u+v)-u\in W_1$ となって矛盾し、
もし $u+v\in W_2$ なら $u=(u+v)-v\in W_2$ となって矛盾する。
従って、必ず $W_1\subset W_2$ または $W_2\subset W_1$ である。

\item[(2)] $x,y\in\ker f$ とスカラー $a,b$ に対し
\[
f(ax+by)=af(x)+bf(y)=0
\]
なので $ax+by\in\ker f$ である。また $f(0)=0$ だから $0\in\ker f$ であり、$\ker f$ は部分空間である。
また
\[
f(x)=f(y)\Longleftrightarrow f(x-y)=0
\Longleftrightarrow x-y\in\ker f.
\]
従って、$\ker f=\{0\}$ なら $f(x)=f(y)$ から $x=y$ が従い、$f$ は単射である。
逆に $f$ が単射なら、$x\in\ker f$ に対して $f(x)=0=f(0)$ だから $x=0$ であり、
$\ker f=\{0\}$ となる。

\item[(3)] $A=(a_{ij})\in W$ とする。条件 $A^{\mathsf T}=-A$ は
\[
a_{ji}=-a_{ij}\qquad(1\le i,j\le3)
\]
を意味する。$i=j$ とすれば $a_{ii}=-a_{ii}$ より $a_{ii}=0$ であるから、$A$ は
\[
A=\begin{pmatrix}
0&a&b\\
-a&0&c\\
-b&-c&0
\end{pmatrix}
=a(E_{12}-E_{21})+b(E_{13}-E_{31})+c(E_{23}-E_{32})
\]
と一意的に書ける。従って
\[
E_{12}-E_{21},\quad E_{13}-E_{31},\quad E_{23}-E_{32}
\]
が基底で，$\dim W=3$。
\end{enumerate}
\end{proof}
