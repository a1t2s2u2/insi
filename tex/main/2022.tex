\section{第1期・専門基礎科目（3題必修）}

\begin{problem}{第1問｜実行列}{2022-1-1}
$n$ 次単位行列を $E_n$ とする。
\begin{enumerate}
  \item[(1)] 実行列 $A$ が $A^2=-E_n$ を満たすなら $n$ は偶数であることを示し，$n=2$ の例を挙げよ。
  \item[(2)] ${}^{\mathsf T}AA=-E_n$ を満たす実行列は存在しないことを示せ。
  \item[(3)] $A\ne O$ が ${}^{\mathsf T}A=-A$ を満たすとき，$P^{-1}AP$ を対角行列にする実正則行列 $P$ は存在しないことを示せ。
\end{enumerate}
\end{problem}
\begin{memo*}
\textbf{解答の見通し。}
行列の等式は成分が多く、そのままでは比較しにくい。そこで、行列式や
$x^{\mathsf T}(\cdot)x$ を使って、行列の等式を一つの実数の等式へ変える。
(1) では $|BC|=|B|\,|C|$ により $A^2=-E_n$ を
$|A|^2=(-1)^n$ へ変え、左右の符号を比べる。
(2) では $x^{\mathsf T}A^{\mathsf T}Ax=\norm{Ax}^2\ge0$ なのに、
右辺は $-\norm{x}^2<0$ になることを使う。
(3) の $A^{\mathsf T}=-A$ は交代行列の条件であり、まず全ての実固有値が0だと示す。
もし実数上で対角化できれば、対角成分は全て0となって $A=O$ になり、仮定と矛盾する。
\end{memo*}
\begin{proof}[解答]
\begin{enumerate}
  \item[(1)] $A^2=-E_n$ なら
  \[
  A(-A)=(-A)A=E_n
  \]
  だから、$-A$ が $A$ の逆行列であり、$A$ は正則である。両辺の行列式を取ると
  \[
  |A|^2=|A^2|=|-E_n|=(-1)^n.
  \]
  $A$ は正則だから $|A|\ne0$、従って $|A|^2>0$ である。
  よって $(-1)^n=1$ でなければならない。
  よって $n$ は偶数である。$n=2$ では
  \[
  J=\begin{pmatrix}0&-1\\1&0\end{pmatrix},\qquad
  J^2=
  \begin{pmatrix}0&-1\\1&0\end{pmatrix}
  \begin{pmatrix}0&-1\\1&0\end{pmatrix}
  =\begin{pmatrix}-1&0\\0&-1\end{pmatrix}=-E_2
  \]
  が一例である。

  \item[(2)] $A^{\mathsf T}A=-E_n$ を満たす実行列 $A$ が存在すると仮定する。
  $0\ne x\in\R^n$ を一つ取る。等式の左から $x^{\mathsf T}$、右から $x$ を掛けると
  \[
  x^{\mathsf T}A^{\mathsf T}Ax
  =(Ax)^{\mathsf T}(Ax)=\norm{Ax}^2\ge0
  \]
  である一方、
  \[
  x^{\mathsf T}(-E_n)x=-x^{\mathsf T}x=-\norm{x}^2<0
  \]
  となる。これは矛盾であるから、そのような実行列は存在しない。

  \item[(3)] $A^{\mathsf T}=-A$ とする。任意の $v\in\R^n$ に対して
  $v^{\mathsf T}Av$ は $1\times1$ 行列、すなわち実数なので、転置しても値は変わらない。従って
  \[
  v^{\mathsf T}Av
  =(v^{\mathsf T}Av)^{\mathsf T}
  =v^{\mathsf T}A^{\mathsf T}v
  =-v^{\mathsf T}Av,
  \]
  よって $v^{\mathsf T}Av=0$ である。

  もし実正則行列 $P$ により $P^{-1}AP=D$ が実対角行列になるなら、
  $P$ の各列は $A$ の実固有ベクトルであり、$D$ の各対角成分は対応する実固有値である。
  実固有値 $\lambda$ と対応する実固有ベクトル $0\ne v$ に対して
  \[
  0=v^{\mathsf T}Av=v^{\mathsf T}(\lambda v)
  =\lambda\norm v^2
  \]
  だから $\lambda=0$ である。従って $D=O$ となり
  \[
  A=PDP^{-1}=O
  \]
  となるが、これは $A\ne O$ に矛盾する。よってそのような $P$ は存在しない。
\end{enumerate}
\end{proof}

\begin{problem}{第2問｜Cauchy の関数方程式}{2022-1-2}
$f:\R\to\R$ が $f(x+y)=f(x)+f(y)$ を満たす。
\begin{enumerate}
  \item[(1)] $f(0)=0$ と $f(-x)=-f(x)$ を示せ。
  \item[(2)] $r\in\Q$ に対して $f(rx)=rf(x)$ を示せ。
  \item[(3)] $f$ が $\R$ 上連続なら，ある $c\in\R$ により $f(x)=cx$ と書けることを示せ。
\end{enumerate}
\end{problem}
\begin{memo*}
\textbf{解答の見通し。}
条件 $f(x+y)=f(x)+f(y)$ は、入力の足し算が出力の足し算へ移るという意味である。
まず式へ $x=y=0$ を入れて $f(0)$ を決め、次に $y=-x$ を入れて負号の扱いを決める。
同じ $x$ を繰り返し足せば整数倍、$x=n(x/n)$ と書けば $1/n$ 倍、両者を組み合わせれば
有理数倍に対する式が得られる。最後に、任意の実数 $x$ へ近づく有理数列を取り、
連続性によって有理数上の公式 $f(r)=f(1)r$ を実数全体へ移す。
\end{memo*}
\begin{proof}[解答]
\begin{enumerate}
  \item[(1)] 与えられた式へ $x=y=0$ を代入すると
  \[
  f(0)=f(0+0)=f(0)+f(0)=2f(0)
  \]
  である。両辺から $f(0)$ を引けば $f(0)=0$ を得る。
  次に $y=-x$ を代入すると
  \[
  0=f(0)=f(x+(-x))=f(x)+f(-x)
  \]
  なので $f(-x)=-f(x)$ である。

  \item[(2)] まず $n\in\N$ に対する $f(nx)=nf(x)$ を、$n$ についての帰納法で示す。
  $n=1$ では両辺とも $f(x)$ である。$n$ で成り立つと仮定すると
  \[
  f((n+1)x)=f(nx+x)=f(nx)+f(x)=nf(x)+f(x)=(n+1)f(x)
  \]
  なので $n+1$ でも成り立つ。従って全ての $n\in\N$ で成り立つ。
  負の整数 $-n$ については、(1) の $f(-u)=-f(u)$ を $u=nx$ に使って
  \[
  f((-n)x)=f(-(nx))=-f(nx)=-nf(x)
  \]
  となる。$n=0$ の場合は (1) の $f(0)=0$ である。よって全ての整数 $m$ で
  $f(mx)=mf(x)$ が成り立つ。

  次に、$n\ge1$ に対して $x=n\cdot(x/n)$ と書き、いま示した式を $x/n$ へ使うと
  \[
  f(x)=f\left(n\cdot\frac xn\right)=nf\left(\frac xn\right),
  \qquad\text{すなわち}\qquad
  f\left(\frac xn\right)=\frac{f(x)}n.
  \]
  従って $r=m/n$（$m\in\Z$, $n\in\N$）に対し
  \[
  f(rx)=f\left(m\cdot\frac{x}{n}\right)
  =m f\left(\frac xn\right)
  =\frac mn f(x)=rf(x).
  \]
  \item[(3)] $c=f(1)$ とおくと (2) より $f(r)=cr$ $(r\in\Q)$。
  任意の $x\in\R$ に対し $r_n\in\Q$, $r_n\to x$ を取る。
  連続性から
  \[
  f(x)=f\left(\lim_{n\to\infty}r_n\right)
  =\lim_{n\to\infty}f(r_n)
  =\lim_{n\to\infty}cr_n=cx.
  \]
\end{enumerate}
\end{proof}

\begin{problem}{第3問｜連続写像の反例}{2022-1-3}
連続写像 $f:X\to Y$ に関する次の三命題が成り立たないことを，反例で示せ。
\begin{enumerate}
  \item[(1)] 開集合 $U\subset X$ の像 $f(U)$ は $Y$ で開である。
  \item[(2)] コンパクト集合 $K\subset Y$ の逆像 $f^{-1}(K)$ は $X$ でコンパクトである。
  \item[(3)] $f$ が全射なら，稠密集合 $V\subset Y$ の逆像 $f^{-1}(V)$ は $X$ で稠密である。
\end{enumerate}
\end{problem}
\begin{memo*}
\textbf{解答の見通し。}
反例とは、仮定を全て満たすのに結論だけが失敗する具体例である。
連続性が直接保証するのは「開集合の\emph{逆像}が開」であり、開集合の像や
コンパクト集合の逆像ではない。(1), (2) は全ての点を0へ送る定数写像で、この違いを示す。
(3) では全射性も必要なので、実数直線はそのまま恒等的に写し、別に加えた孤立点 $p$ だけを0へ送る。
孤立点とは一元集合 $\{p\}$ 自身が開になる点であり、これが逆像の稠密性を壊す。
\end{memo*}
\begin{proof}[解答]
\begin{enumerate}
  \item[(1)] $X=Y=\R$ とし、定数写像
  \[
  f:\R\to\R,\qquad f(x)=0
  \]
  を考える。任意の開集合 $O\subset\R$ に対して
  \[
  f^{-1}(O)=
  \begin{cases}
  \R &(0\in O),\\
  \varnothing &(0\notin O)
  \end{cases}
  \]
  だから $f$ は連続である。しかし $U=\R$ は $X$ の開集合である一方、
  $f(U)=\{0\}$ は $Y=\R$ の開集合ではない。

  \item[(2)] (1) と同じ定数写像を使う。$K=\{0\}\subset\R$ は有限集合なのでコンパクトであるが
  \[
  f^{-1}(K)=f^{-1}(\{0\})=\R
  \]
  はコンパクトでない。実際、開被覆 $\{(-n,n)\}_{n\ge1}$ は有限部分被覆を持たない。

  \item[(3)] $X=\R\sqcup\{p\}$ とする。$X$ の部分集合を、
  $\R$ の開集合 $O$ と $E=\varnothing$ または $E=\{p\}$ を用いて
  $O\sqcup E$ と書けるときに開と定める。これがここでいう直和位相であり、特に $\{p\}$ は開である。
  $Y=\R$ とし、写像 $f:X\to Y$ を
  \[
  f(x)=x\quad(x\in\R),\qquad f(p)=0
  \]
  と定める。$f|_{\R}$ が恒等写像なので $f$ は全射である。
  $Y$ の開集合 $O$ に対して
  \[
  f^{-1}(O)=
  \begin{cases}
  O\sqcup\{p\} &(0\in O),\\
  O &(0\notin O)
  \end{cases}
  \]
  であり、どちらも直和位相で開だから $f$ は連続である。

  $V=\R\setminus\{0\}$ とすると $\overline V=\R$ なので $V$ は $Y$ で稠密である。しかし
  \[
  f^{-1}(V)=\R\setminus\{0\}\subset X
  \]
  は $X$ の非空開集合 $\{p\}$ と交わらない。従って
  $\overline{f^{-1}(V)}\ne X$ であり、$f^{-1}(V)$ は $X$ で稠密でない。
\end{enumerate}
\end{proof}

\begin{problem}{第1問｜有限群}{2022-s-1}
$G=S_5$，$P=\langle(1\ 2\ 3\ 4\ 5)\rangle$ とする。また $\Omega$ を $G$ の Sylow $5$ 部分群全体とする。
\begin{enumerate}
  \item[(1)] 位数5の巡回群の自己同型群の位数を求めよ。
  \item[(2)] 正規化群 $N_G(P)$ の Sylow $2$ 部分群を一つ求めよ。
  \item[(3)] 共役作用 $\varphi:G\to\operatorname{Sym}(\Omega)$ が準同型であることを示せ。
  \item[(4)] $S_6$ に，指数6で互いに共役でない二つの部分群があることを示せ。
\end{enumerate}
\end{problem}
\begin{memo*}
\textbf{解答の見通し。}
位数5の部分群を数えると、$S_5$ が6個の Sylow $5$ 部分群へ作用することが分かる。
この共役作用から通常とは異なる $S_5\hookrightarrow S_6$ を作り、
最後に「6点へ推移的に作用するか、1点を固定するか」という共役で変わらない性質で
二つの指数6部分群を区別する。
\end{memo*}
\begin{proof}[解答]
\begin{enumerate}
  \item[(1)] $C_5=\langle a\rangle$ とする。自己同型は生成元 $a$ の像によって一意に決まり、
  自己同型は生成元を生成元へ写す。$C_5$ の生成元は
  \[
  a,a^2,a^3,a^4
  \]
  の4個である。逆に $k=1,2,3,4$ に対して $a\mapsto a^k$ は自己同型を定めるので
  \[
  |\operatorname{Aut}(C_5)|=4.
  \]

  \item[(2)] $S_5$ の5-cycle の個数は、最初の文字を1に固定して残りを並べることで
  \[
  (5-1)!=24
  \]
  個である。一つの位数5の部分群には単位元でない5-cycle が4個あり、異なる位数5の部分群は
  単位元以外では交わらない。従って Sylow $5$ 部分群の個数は
  \[
  |\Omega|=\frac{24}{4}=6.
  \]
  共役作用における $P$ の安定化群は $N_G(P)$ であり、Sylow の定理により作用は推移的だから、
  軌道・安定化群定理から
  \[
  |N_G(P)|=\frac{|S_5|}{|\Omega|}
  =\frac{120}{6}=20=2^2\cdot5.
  \]

  ここで
  \[
  a=(1\ 2\ 3\ 4\ 5),
  \qquad b=(2\ 3\ 5\ 4)
  \]
  とおく。$b$ は4-cycle なので位数は4である。また巡回置換の共役公式から
  \[
  bab^{-1}
  =(b(1)\ b(2)\ b(3)\ b(4)\ b(5))
  =(1\ 3\ 5\ 2\ 4)=a^2.
  \]
  従って $bPb^{-1}=P$、すなわち $b\in N_G(P)$ である。
  よって $\langle b\rangle$ は $N_G(P)$ の位数4の部分群であり、
  $|N_G(P)|$ に現れる2の最高冪が $2^2=4$ だから Sylow $2$ 部分群である。

  \item[(3)] $g\in G$ と $Q\in\Omega$ に対して
  \[
  \varphi(g)(Q)=gQg^{-1}
  \]
  と定める。$Q$ が位数5なら $gQg^{-1}$ も位数5なので、$\varphi(g)$ は $\Omega$ の置換を定める。
  $g,h\in G$ に対して
  \begin{align*}
  \varphi(gh)(Q)
  &=(gh)Q(gh)^{-1}\\
  &=g(hQh^{-1})g^{-1}\\
  &=\varphi(g)(\varphi(h)(Q)).
  \end{align*}
  従って $\varphi(gh)=\varphi(g)\varphi(h)$ であり、
  $\varphi:G\to\operatorname{Sym}(\Omega)$ は群準同型である。

  \item[(4)] (2) より $|\Omega|=6$ なので、$\Omega$ と $\{1,\ldots,6\}$ を同一視すれば
  \[
  \varphi:S_5\longrightarrow S_6
  \]
  とみなせる。$K=\ker\varphi$ とおくと $K\triangleleft S_5$ である。
  $K$ の元は全ての $Q\in\Omega$ を固定し、特に $P$ を固定するから
  \[
  K\subset N_G(P).
  \]
  $S_5$ の正規部分群は $\{1\},A_5,S_5$ の三つである。
  一方 $|N_G(P)|=20$ なので、位数60の $A_5$ も位数120の $S_5$ も
  $N_G(P)$ に含まれない。従って $K=\{1\}$ であり、$\varphi$ は単射である。

  よって
  \[
  H_1:=\varphi(S_5)\subset S_6,
  \qquad |H_1|=120,
  \qquad [S_6:H_1]=\frac{720}{120}=6.
  \]
  Sylow 部分群は互いに共役なので、$H_1$ の6点集合 $\Omega$ への作用は推移的である。

  一方、通常の点固定部分群
  \[
  H_2:=\{\tau\in S_6\mid\tau(6)=6\}
  \]
  は $S_5$ と同型で、$|H_2|=120$、従って $[S_6:H_2]=6$ である。
  $H_2$ は点6を固定するが、$H_1$ は6点に推移的に作用するため固定点を持たない。
  もし $H_1=sH_2s^{-1}$ となる $s\in S_6$ が存在すれば、$H_1$ は点 $s(6)$ を固定することになり矛盾する。
  従って $H_1,H_2$ は指数6で互いに共役でない二つの部分群である。
\end{enumerate}
\end{proof}

\begin{problem}{第2問｜球面の面積形式}{2022-s-2}
$S^2=\{p\in\R^3\mid\norm p=1\}$ とし，接ベクトル $u,v\in T_pS^2$ に対して
$\omega_p(u,v)=\inner p{u\times v}$ と定める。
\begin{enumerate}
  \item[(1)] $\omega_p(u,v)=-\omega_p(v,u)$ を示せ。また $\omega_p(u,v)=0$ が全ての $v$ に対して成り立つなら $u=0$ であることを示せ。
  \item[(2)] $U=S^2\setminus\{(x,0,z)\in S^2\mid x\ge0\}$ とする。$0<\theta<2\pi$，$-1<h<1$ に対し $(r\cos\theta,r\sin\theta,h)\in U$ となる $r$ を求めよ。
  \item[(3)] $T(\theta,h)=(r\cos\theta,r\sin\theta,h)$ とおくとき，$\omega_{T(\theta,h)}(T_\theta,T_h)$ を求めよ。
\end{enumerate}
\end{problem}
\begin{memo*}
\textbf{解答の見通し。}
(1) の反対称性は外積の順序交換から従う。非退化性では
$v=p\times u$ と選ぶと、三重積が $\norm{u}^2p$ になって $u$ の長さを取り出せる。
(2), (3) では高さ $h$ の緯線の半径を球面の方程式から決め、
座標ベクトルの外積が外向き法線 $T$ に一致することを確かめる。
\end{memo*}
\begin{proof}[解答]
\begin{enumerate}
  \item[(1)] 外積の反対称性 $v\times u=-(u\times v)$ より
  \[
  \omega_p(v,u)=\inner p{v\times u}
  =-\inner p{u\times v}=-\omega_p(u,v).
  \]
  次に $\omega_p(u,v)=0$ が全ての $v\in T_pS^2$ に対して成り立つと仮定する。
  $u\in T_pS^2$ だから $\inner pu=0$ であり、$v=p\times u$ も $p$ と直交するので
  $v\in T_pS^2$ である。ベクトル三重積の公式を用いると
  \[
  u\times(p\times u)
  =p\inner uu-u\inner up
  =\norm u^2p.
  \]
  従って
  \[
  0=\omega_p(u,p\times u)
  =\inner p{\norm u^2p}
  =\norm u^2\norm p^2=\norm u^2.
  \]
  よって $u=0$ である。

  \item[(2)] 点 $(r\cos\theta,r\sin\theta,h)$ が $S^2$ 上にある条件は
  \[
  r^2\cos^2\theta+r^2\sin^2\theta+h^2=1,
  \]
  すなわち $r^2=1-h^2$ である。$-1<h<1$ では右辺は正であり、
  $r$ を緯線の半径として正に取るので
  \[
  r(\theta,h)=\sqrt{1-h^2}.
  \]
  $r$ は $\theta$ に依存しない。また $0<\theta<2\pi$ としたことで、除いた半大円上の点は現れない。

  \item[(3)] $r=\sqrt{1-h^2}$ だから $\partial r/\partial h=-h/r$ であり
  \[
  T_\theta=(-r\sin\theta,r\cos\theta,0),\qquad
  T_h=\left(-\frac h r\cos\theta,-\frac h r\sin\theta,1\right).
  \]
  外積を成分ごとに計算すると
  \begin{align*}
  T_\theta\times T_h
  &=\left(r\cos\theta,r\sin\theta,
    h(\sin^2\theta+\cos^2\theta)\right)\\
  &=(r\cos\theta,r\sin\theta,h)=T(\theta,h).
  \end{align*}
  従って $\omega_T(T_\theta,T_h)=\inner{T}{T}=1$。
\end{enumerate}
\end{proof}

\begin{problem}{第3問｜微分方程式}{2022-s-3}
$f$ を $\R$ 上有界連続とし，$y''-y=f(x)$ を考える。
\begin{enumerate}
  \item[(1)] $f=0$ のときの一般解を求めよ。
  \item[(2)] $e^{-x}f(x)$ が任意の $[a,\infty)$ 上で広義積分可能であることを示せ。
  \item[(3)] 次で定める $g$ を考える：
  \[
  g(x)=-\frac{e^x}{2}\int_x^\infty e^{-t}f(t)\,\d t
  -\frac{e^{-x}}2\int_{-\infty}^x e^tf(t)\,\d t
  \]
  が方程式の解であることを示せ。
  \item[(4)] $f(x)\to1$ $(x\to\infty)$ なら $g(x)\to-1$ を示せ。
  \item[(5)] $f(x)=1-e^{-x}$ $(x\ge0)$，$f(x)=0$ $(x<0)$ のとき，$y(x)\to-1$ $(x\to\infty)$ を満たす解を全て求めよ。
\end{enumerate}
\end{problem}
\begin{memo*}
\textbf{解答の見通し。}
最初に同次方程式の二つの基本解 $e^x,e^{-x}$ を確認する。
(3) の積分表示は二つの積分を別々に微分すると交差項が消えるよう設計されている。
(4) は $f=1+(f-1)$ と分けて優収束定理を使い、
(5) は $x<0$ と $x\ge0$ で解いた後、$x=0$ で $y,y'$ を接続する。
\end{memo*}
\begin{proof}[解答]
\begin{enumerate}
  \item[(1)] 特性方程式は
  \[
  \lambda^2-1=(\lambda-1)(\lambda+1)=0
  \]
  であり、相異なる特性根は $1,-1$ である。従って一般解は
  \[
  y(x)=C_1e^x+C_2e^{-x}\qquad(C_1,C_2\in\R)
  \]
  である。
  \item[(2)] $f$ は有界だから、ある $M\ge0$ が存在して
  $|f(x)|\le M$ が全ての $x$ で成り立つ。任意の $a\in\R$ に対して
  \[
  \int_a^\infty|e^{-x}f(x)|\,\d x
  \le M\int_a^\infty e^{-x}\,\d x
  =Me^{-a}<\infty.
  \]
  よって $e^{-x}f(x)$ は $[a,\infty)$ 上で絶対可積分であり、広義積分可能である。
  \item[(3)]
  \[
  A(x)=\int_x^\infty e^{-t}f(t)\,\d t,\qquad
  B(x)=\int_{-\infty}^x e^tf(t)\,\d t
  \]
  とおく。微積分学の基本定理より
  \[
  A'(x)=-e^{-x}f(x),\qquad B'(x)=e^xf(x).
  \]
  $g=-(e^xA+e^{-x}B)/2$ を積の微分法で微分すると
  \begin{align*}
  g'
  &=-\frac12\{e^xA+e^xA'-e^{-x}B+e^{-x}B'\}\\
  &=-\frac12e^xA+\frac12e^{-x}B,
  \end{align*}
  ここでは $e^xA'=-f$ と $e^{-x}B'=f$ が打ち消し合った。もう一度微分して
  \begin{align*}
  g''
  &=-\frac12(e^xA+e^xA')
    +\frac12(-e^{-x}B+e^{-x}B')\\
  &=-\frac12(e^xA+e^{-x}B)+f(x)\\
  &=g(x)+f(x).
  \end{align*}
  従って $g''-g=f$ である。
  \item[(4)] $h=f-1$ とおくと $h(x)\to0$ で、$h$ は有界である。
  定数関数 $1$ を上の式に代入して得られる解は
  \[
  -\frac{e^x}{2}\int_x^\infty e^{-t}\,\d t
  -\frac{e^{-x}}2\int_{-\infty}^x e^t\,\d t=-1.
  \]
  一方、$h$ が作る第1項は変数変換 $t=x+s$ により
  \[
  e^x\int_x^\infty e^{-t}h(t)\,\d t
  =\int_0^\infty e^{-s}h(x+s)\,\d s\longrightarrow0,
  \]
  第2項も $t=x-s$ により
  \[
  e^{-x}\int_{-\infty}^x e^th(t)\,\d t
  =\int_0^\infty e^{-s}h(x-s)\,\d s\longrightarrow0.
  \]
  いずれも有界性から $Ce^{-s}$ を優関数にして優収束定理を使った。
  従って $g(x)\to-1$。
  \item[(5)] $x\ge0$ では、定数項 $1$ の特殊解は $-1$。
  また $-e^{-x}$ は同次解 $e^{-x}$ と共振するので
  $y_p=Cxe^{-x}$ とおくと
  \[
  y_p''-y_p=-2Ce^{-x}.
  \]
  よって $C=1/2$。$x\to\infty$ で $-1$ に収束するには $e^x$ の項があってはならないから
  \[
  y(x)=-1+\left(D+\frac x2\right)e^{-x}\qquad(x\ge0).
  \]
  $x<0$ では $f=0$ なので $y=Be^x+Ce^{-x}$。
  $x=0$ で $y,y'$ を一致させると
  \[
  B+C=-1+D,\qquad B-C=\frac12-D,
  \]
  従って $B=-1/4$, $C=D-3/4$。以上より、任意の $D\in\R$ に対し
  \[
  y(x)=\begin{cases}
  -\frac14e^x+(D-\frac34)e^{-x},&x<0,\\
  -1+(D+\frac x2)e^{-x},&x\ge0
  \end{cases}
  \]
  であり、これが全てである。
\end{enumerate}
\end{proof}

\begin{problem}{第4問｜Dirichlet 積分}{2022-s-4}
$t>0$ とする。
\begin{enumerate}
  \item[(1)] $\sin x/x$ が $(0,t)$ 上でルベーグ積分可能であることを示せ。
  \item[(2)] $y>0$ に対し
  \[
  \int_0^t e^{-xy}\sin x\,\d x
  =\frac1{1+y^2}-\frac{e^{-ty}}{1+y^2}(y\sin t+\cos t)
  \]
  を示せ。
  \item[(3)] 次の等式を示せ：
  \[
  \int_0^t\frac{\sin x}{x}\,\d x
  =\int_0^\infty\frac{\d y}{1+y^2}
  -\int_0^\infty\frac{e^{-ty}}{1+y^2}(y\sin t+\cos t)\,\d y
  \]
  \item[(4)] $\displaystyle\lim_{t\to\infty}\int_0^t\frac{\sin x}{x}\,\d x$ を求めよ。
\end{enumerate}
\end{problem}
\begin{memo*}
\textbf{解答の見通し。}
$\sin x/x$ は原点の除去可能特異点を埋めれば連続になる。
次に $1/x=\int_0^\infty e^{-xy}\,\d y$ を使って二重積分へ移し、
Fubini の定理で積分順序を交換する。
最後は明示式の誤差項が $1/t+1/t^2$ 以下になることを示せばよい。
\end{memo*}
\begin{proof}[解答]
\begin{enumerate}
  \item[(1)] $\lim_{x\to0}\sin x/x=1$ なので、$x=0$ で値を1と定めれば $[0,t]$ 上連続になる。
  従ってコンパクト区間上でルベーグ積分可能である。
  \item[(2)]
  \[
  F(x)=\frac{e^{-xy}(-y\sin x-\cos x)}{1+y^2}
  \]
  とおく。積の微分法により
  \begin{align*}
  F'(x)
  &=\frac{e^{-xy}}{1+y^2}
  \{(-y)(-y\sin x-\cos x)-y\cos x+\sin x\}\\
  &=e^{-xy}\sin x.
  \end{align*}
  よって
  \begin{align*}
  \int_0^t e^{-xy}\sin x\,\d x
  &=F(t)-F(0)\\
  &=\frac1{1+y^2}
   -\frac{e^{-ty}}{1+y^2}(y\sin t+\cos t).
  \end{align*}
  \item[(3)] $x>0$ では
  \[
  \frac1x=\int_0^\infty e^{-xy}\,\d y.
  \]
  また
  \[
  \int_0^t\int_0^\infty|e^{-xy}\sin x|\,\d y\d x
  =\int_0^t\frac{|\sin x|}{x}\,\d x<\infty
  \]
  なので Fubini の定理で積分順序を交換できる。(2) を用いて
  \begin{align*}
  \int_0^t\frac{\sin x}{x}\,\d x
  &=\int_0^\infty\int_0^t e^{-xy}\sin x\,\d x\d y\\
  &=\int_0^\infty\frac{\d y}{1+y^2}
  -\int_0^\infty\frac{e^{-ty}}{1+y^2}(y\sin t+\cos t)\,\d y.
  \end{align*}
  \item[(4)] 第2積分の絶対値は
  \[
  \int_0^\infty e^{-ty}(y+1)\,\d y
  =\frac1{t^2}+\frac1t\longrightarrow0
  \]
  以下である。一方
  \[
  \int_0^\infty\frac{\d y}{1+y^2}
  =[\arctan y]_0^\infty=\frac\pi2.
  \]
  従って求める極限は $\pi/2$。
\end{enumerate}
\end{proof}

\section{第2期・専門基礎科目（3題必修）}

\begin{problem}{第1問｜線形空間}{2022-2-1}
以下，$U,V$ は $\R$ 上の線形空間とする。
\begin{enumerate}
  \item[(1)] $V$ の部分空間 $W_1,W_2\subset V$ に対し $W_1\cap W_2$ は部分空間であること，また $W_1\cup W_2$ が部分空間なら $W_1\subset W_2$ または $W_2\subset W_1$ であることを示せ。
  \item[(2)] 線形写像 $f:U\to V$ に対し $\ker f$ は部分空間であり，$f$ が単射であることと $\ker f=\{0\}$ が同値であることを示せ。
  \item[(3)] $M_3(\R)$ の交代行列全体 $W=\{A\mid A^{\mathsf T}=-A\}$ の基底と次元を求めよ。
\end{enumerate}
\end{problem}
\begin{memo*}
\textbf{解答の見通し。}
部分空間であることは、零ベクトルを含み、集合の任意の元 $x,y$ と任意の $a,b\in\R$ に対して
$ax+by$ も集合内に残ることを確認すればよい。
合併の問題では、どちらにも含まれないベクトルを一つずつ取ると、その和が矛盾を生む。
核 $\ker f$ は0へ写るベクトル全体である。単射の条件 $f(x)=f(y)\Rightarrow x=y$ を、
線形性により $f(x-y)=0$ へ直すと核との関係が見える。
交代行列では $A^{\mathsf T}=-A$ を各成分の式 $a_{ji}=-a_{ij}$ に直し、
対角成分と下三角成分が上三角成分から決まることを読む。
\end{memo*}
\begin{proof}[解答]
\begin{enumerate}
\item[(1)] $W_1,W_2$ は部分空間なので $0\in W_1\cap W_2$ である。また
$x,y\in W_1\cap W_2$ と $a,b\in\R$ を取ると、$x,y$ は $W_1,W_2$ の両方に属するから
\[
ax+by\in W_1,\qquad ax+by\in W_2.
\]
従って $ax+by\in W_1\cap W_2$ であり、$W_1\cap W_2$ は部分空間である。

次に $W_1\cup W_2$ が部分空間であると仮定する。
$W_1\not\subset W_2$ かつ $W_2\not\subset W_1$ なら、
\[
u\in W_1\setminus W_2,
\qquad v\in W_2\setminus W_1
\]
を取れる。合併が部分空間なら $u+v\in W_1\cup W_2$ である。
もし $u+v\in W_1$ なら $v=(u+v)-u\in W_1$ となって矛盾し、
もし $u+v\in W_2$ なら $u=(u+v)-v\in W_2$ となって矛盾する。
従って、必ず $W_1\subset W_2$ または $W_2\subset W_1$ である。

\item[(2)] $x,y\in\ker f$ と $a,b\in\R$ に対し
\[
f(ax+by)=af(x)+bf(y)=0
\]
なので $ax+by\in\ker f$ である。また $f(0)=0$ だから $0\in\ker f$ であり、$\ker f$ は部分空間である。
また
\[
f(x)=f(y)\Longleftrightarrow f(x-y)=0
\Longleftrightarrow x-y\in\ker f.
\]
従って、$\ker f=\{0\}$ なら $f(x)=f(y)$ から $x=y$ が従い、$f$ は単射である。
逆に $f$ が単射なら、$x\in\ker f$ に対して $f(x)=0=f(0)$ だから $x=0$ であり、
$\ker f=\{0\}$ となる。

\item[(3)] $A=(a_{ij})\in W$ とする。条件 $A^{\mathsf T}=-A$ は
\[
a_{ji}=-a_{ij}\qquad(1\le i,j\le3)
\]
を意味する。$i=j$ とすれば $a_{ii}=-a_{ii}$、すなわち $2a_{ii}=0$ より $a_{ii}=0$ である。
また $i<j$ の成分を $a_{12}=a$, $a_{13}=b$, $a_{23}=c$ とおけば、
残りの成分は $a_{21}=-a$, $a_{31}=-b$, $a_{32}=-c$ と決まる。従って $A$ は
\[
A=\begin{pmatrix}
0&a&b\\
-a&0&c\\
-b&-c&0
\end{pmatrix}
=aF_{12}+bF_{13}+cF_{23},
\qquad
F_{ij}=E_{ij}-E_{ji}
\]
と書ける。ここで $E_{ij}$ は $(i,j)$ 成分だけが1で、他の成分が0の行列である。
逆に $F_{12},F_{13},F_{23}$ はいずれも $F_{ij}^{\mathsf T}=-F_{ij}$ を満たすので $W$ に属し、
その一次結合も $W$ に属する。よって $F_{12},F_{13},F_{23}$ は $W$ を生成する。

一次独立性を見る。$aF_{12}+bF_{13}+cF_{23}=O$ とすると、上の行列表示の
$(1,2),(1,3),(2,3)$ 成分を読めばそれぞれ $a=0$, $b=0$, $c=0$ である。
従って $F_{12},F_{13},F_{23}$ は一次独立であり、$W$ の基底である。よって
\[
E_{12}-E_{21},\quad E_{13}-E_{31},\quad E_{23}-E_{32}
\]
が基底で，$\dim W=3$。
\end{enumerate}
\end{proof}

\begin{problem}{第2問｜微分積分}{2022-2-2}
$a<b$，$u$ は $[a,b]^2$ 上連続とする。
\begin{enumerate}
  \item[(1)] $\displaystyle f(x)=\int_a^x\d s\int_a^s u(s,t)\,\d t$ が $(a,b)$ で微分可能で，$\displaystyle f'(x)=\int_a^x u(x,t)\,\d t$ となることを示せ。
  \item[(2)] $x>0$，$D_x=\{(s,t)\mid0\le s\le t\le x\}$ とし
  $\displaystyle g(x)=\iint_{D_x}s(1-2st)\,\d s\d t$ とおく。$x>0$ における最大値を求めよ。
\end{enumerate}
\end{problem}
\begin{memo*}
\textbf{解答の見通し。}
(1) では内側の積分全体を $F(s)=\int_a^s u(s,t)\,\d t$ と名付ける。
すると $f(x)=\int_a^xF(s)\,\d s$ なので、$F$ が連続だと確認できれば
微積分学の基本定理から $f'(x)=F(x)$ となる。
$F$ では積分区間と被積分関数の両方が $s$ とともに変わるため、差を二つに分けて連続性を示す。
(2) の不等式 $0\le s\le t\le x$ は、外側で $t$ を $0$ から $x$ まで動かし、
各 $t$ を固定した内側で $s$ を $0$ から $t$ まで動かす、という積分順序を表す。
\end{memo*}
\begin{proof}[解答]
\begin{enumerate}
\item[(1)] $F(s)=\int_a^s u(s,t)\,\d t$ とおく。$F$ の連続性を確認する。
$s,s+h\in[a,b]$ とし、まず $h>0$ の場合には
\begin{align*}
F(s+h)-F(s)
&=\int_a^s\{u(s+h,t)-u(s,t)\}\,\d t
  +\int_s^{s+h}u(s+h,t)\,\d t.
\end{align*}
$u$ はコンパクト集合 $[a,b]^2$ 上連続なので一様連続かつ有界であり、
$|u|\le M$ となる $M\ge0$ と、任意の $\varepsilon>0$ に対して
\[
|s-s'|<\eta\quad\Longrightarrow\quad
|u(s,t)-u(s',t)|<\varepsilon\qquad(t\in[a,b])
\]
を満たす $\eta>0$ が取れる。従って $0<h<\eta$ なら
\[
|F(s+h)-F(s)|
\le\int_a^s|u(s+h,t)-u(s,t)|\,\d t
 +\int_s^{s+h}|u(s+h,t)|\,\d t
\le\varepsilon(b-a)+Mh.
\]
右辺は $\varepsilon\to0$, $h\to0$ で0へ近づくので、第1項も第2項も0へ収束する。
$h<0$ でも積分区間の向きを入れ替えれば同じ評価ができるので、$F$ は連続である。

$f(x)=\int_a^xF(s)\,\d s$ だから、微積分学の基本定理より
\[
f'(x)=F(x)=\int_a^x u(x,t)\,\d t.
\]
\item[(2)] 積分領域の条件 $0\le s\le t\le x$ は、外側で $t$ を $0$ から $x$ まで動かし、
各 $t$ を固定した内側で $s$ を $0$ から $t$ まで動かすことを表す。
被積分関数は $s(1-2st)=s-2s^2t$ だから
\begin{align*}
g(x)
&=\int_0^x\int_0^t (s-2s^2t)\,\d s\d t\\
&=\int_0^x\left[\frac{s^2}{2}-\frac{2ts^3}{3}\right]_{s=0}^{s=t}\d t\\
&=\int_0^x\left(\frac{t^2}{2}-\frac{2t^4}{3}\right)\d t\\
&=\left[\frac{t^3}{6}-\frac{2t^5}{15}\right]_{t=0}^{t=x}
=\frac{x^3}{6}-\frac{2x^5}{15}.
\end{align*}
従って
\[
g'(x)=\frac{x^2}{2}-\frac{2x^4}{3}
=x^2\left(\frac12-\frac{2x^2}{3}\right).
\]
$x>0$ では $x^2>0$ なので、$g'(x)$ の符号は括弧の中だけで決まる。
\[
\frac12-\frac{2x^2}{3}=0
\quad\Longleftrightarrow\quad
x^2=\frac34
\quad\Longleftrightarrow\quad
x=\frac{\sqrt3}{2}\quad(x>0)
\]
であり、括弧の中は $x$ について単調減少だから、
$0<x<\sqrt3/2$ では $g'(x)>0$、$x>\sqrt3/2$ では $g'(x)<0$ である。
従って $g$ は $x=\sqrt3/2$ で最大になる。この点では
\[
x^2=\frac34,\qquad
x^3=\frac34\cdot\frac{\sqrt3}{2}=\frac{3\sqrt3}{8},\qquad
x^5=\left(\frac34\right)^2\frac{\sqrt3}{2}=\frac{9\sqrt3}{32}
\]
なので
\[
g\left(\frac{\sqrt3}{2}\right)
=\frac16\cdot\frac{3\sqrt3}{8}-\frac{2}{15}\cdot\frac{9\sqrt3}{32}
=\frac{\sqrt3}{16}-\frac{3\sqrt3}{80}
=\frac{5\sqrt3-3\sqrt3}{80}
=\frac{\sqrt3}{40}.
\]
よって $x>0$ における最大値は $\sqrt3/40$ である。
\end{enumerate}
\end{proof}

\begin{problem}{第3問｜位相空間}{2022-2-3}
\begin{enumerate}
  \item[(1)] 位相空間 $X$ の部分集合 $A,B$ に対し $\overline{A\cup B}=\overline A\cup\overline B$ を示せ。
  \item[(2)] Hausdorff 空間のコンパクト部分集合は閉集合であることを示せ。
  \item[(3)] 全射連続写像 $F:S\to T$ について，$S$ が連結なら $T$ も連結であることを示せ。
\end{enumerate}
\end{problem}
\begin{memo*}
\textbf{解答の見通し。}
(1) の閉包 $\overline A$ は「$A$ を含む最小の閉集合」である。
二つの集合が等しいことは、左が右に含まれることと、右が左に含まれることを別々に示せばよい。
(2) の Hausdorff 性は異なる二点を交わらない開近傍で分けられる性質、
コンパクト性は任意の開被覆から有限個を選べる性質である。
点 $x\notin K$ と各 $y\in K$ をまず個別に分け、有限個の近傍だけで $K$ 全体を覆う。
(3) の連結とは、空でない二つの開集合へ分割できないことである。
像が分割できると仮定し、連続写像の逆像を取ると定義域まで分割されることを示す。
\end{memo*}
\begin{proof}[解答]
\begin{enumerate}
\item[(1)] 閉包 $\overline{A\cup B}$ は $A\cup B$ を含む最小の閉集合である。
$\overline A\cup\overline B$ は閉集合であり $A\cup B$ を含むので
\[
\overline{A\cup B}\subset\overline A\cup\overline B.
\]
逆に $A\subset A\cup B$ と $B\subset A\cup B$ に閉包の単調性を用いると
\[
\overline A\subset\overline{A\cup B},\qquad
\overline B\subset\overline{A\cup B}.
\]
従って $\overline A\cup\overline B\subset\overline{A\cup B}$ であり、両方の包含から
\[
\overline{A\cup B}=\overline A\cup\overline B
\]
を得る。

\item[(2)] $K\subset X$ をコンパクト部分集合とし、$x\in X\setminus K$ を取る。
$X$ は Hausdorff 空間なので、各 $y\in K$ に対して互いに交わらない開近傍
$U_y\ni x$ と $V_y\ni y$ を取れる。$\{V_y\}_{y\in K}$ は $K$ の開被覆だから、
コンパクト性より $y_1,\ldots,y_m\in K$ が存在して
\[
K\subset V_{y_1}\cup\cdots\cup V_{y_m}
\]
となる。ここで
\[
U=U_{y_1}\cap\cdots\cap U_{y_m}
\]
とおく。$U$ は有限個の開集合の共通部分なので開であり、
各 $U_{y_i}$ が $x$ を含むので $x\in U$ である。すなわち $U$ は $x$ の開近傍である。
さらに $U$ は $K$ と交わらない。実際、$z\in U\cap K$ があるとすると、
$K\subset V_{y_1}\cup\cdots\cup V_{y_m}$ よりある $i$ について $z\in V_{y_i}$ であり、
一方 $z\in U\subset U_{y_i}$ でもあるから
\[
z\in U_{y_i}\cap V_{y_i}=\varnothing
\]
となって矛盾する。従って $U\subset X\setminus K$ である。
$x\in X\setminus K$ は任意だったので $X\setminus K$ は開集合であり、$K$ は閉集合である。

\item[(3)] $T$ が連結でないと仮定すると、非空な開集合 $U,V\subset T$ により
\[
T=U\cup V,\qquad U\cap V=\varnothing
\]
と書ける。$F$ は連続なので $F^{-1}(U),F^{-1}(V)$ は $S$ の開集合である。
また $F$ は全射だから、$U\ne\varnothing$ の点は像に現れ、$F^{-1}(U)\ne\varnothing$、
同様に $F^{-1}(V)\ne\varnothing$ である。さらに逆像は合併と共通部分を保つので
\[
F^{-1}(U)\cup F^{-1}(V)=F^{-1}(U\cup V)=F^{-1}(T)=S,
\]
\[
F^{-1}(U)\cap F^{-1}(V)=F^{-1}(U\cap V)=F^{-1}(\varnothing)=\varnothing.
\]
従って $S$ も空でない二つの開集合へ分割され、$S$ が連結であることに矛盾する。
よって $T$ は連結である。
\end{enumerate}
\end{proof}
