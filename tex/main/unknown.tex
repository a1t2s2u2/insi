\section{専門基礎科目と思われる問題}

\begin{problem}{行列｜直交対角化}{unknown-1}
\[
P=\begin{pmatrix}
1/\sqrt3&1/\sqrt2&p\\q&r&s\\-1/\sqrt3&1/\sqrt2&-1/\sqrt6
\end{pmatrix}
\]
は直交行列とする。$p,q,r,s$ を求め，$(P^{\mathsf T})^{10}P^{19}(P^{\mathsf T})^{10}$ を求めよ。さらに $A$ は $P$ で対角化され，固有多項式が $x^3+\sqrt[10]{2}x^2$，$(1,0,1)^{\mathsf T}$ が固有値 $-\sqrt[10]{2}$ の固有ベクトルであるとき $A^{10}$ を求めよ。直交行列で対角化される実行列が対称であることも示せ。
\end{problem}
\begin{proof}[解答]
直交行列の各列は正規直交系である。第1列と第2列の内積から
\[
\frac1{\sqrt6}+qr-\frac1{\sqrt6}=qr=0.
\]
第2列のノルムは $1/2+r^2+1/2=1$ なので $r=0$。第1列のノルムから
$1/3+q^2+1/3=1$、従って $q=\varepsilon/\sqrt3$（$\varepsilon=\pm1$）である。
第3列を $(p,s,-1/\sqrt6)^{\mathsf T}$ と書く。第2列との直交条件は
\[
\frac p{\sqrt2}-\frac1{\sqrt{12}}=0
\]
なので $p=1/\sqrt6$。第1列との直交条件は
\[
\frac p{\sqrt3}+qs+\frac1{\sqrt{18}}=0.
\]
$p=1/\sqrt6$, $q=\varepsilon/\sqrt3$ を代入すると
$s=-\varepsilon\sqrt{2/3}$ となる。従って全ての組は
\[
p=\frac1{\sqrt6},\quad q=\frac{\varepsilon}{\sqrt3},\quad r=0,
\quad s=-\varepsilon\sqrt{\frac23}.
\]
$P^{\mathsf T}=P^{-1}$ なので、指数法則を使うと
\begin{align*}
(P^{\mathsf T})^{10}P^{19}(P^{\mathsf T})^{10}
&=P^{-10}P^{19}P^{-10}\\
&=P^{-1}=P^{\mathsf T}.
\end{align*}

固有多項式は $x^2(x+\sqrt[10]{2})$ だから、固有値は0（重複度2）と
$-\sqrt[10]{2}$（重複度1）である。$A$ は直交行列で対角化されるため、
異なる固有空間は直交する。正規化した固有ベクトルを
$u=(1,0,1)^{\mathsf T}/\sqrt2$ とすれば、$u$ 方向への直交射影は $uu^{\mathsf T}$ なので
\[
A=-\sqrt[10]{2}\,uu^{\mathsf T}.
\]
ここで $(uu^{\mathsf T})^2=u(u^{\mathsf T}u)u^{\mathsf T}=uu^{\mathsf T}$ だから
\[
A^{10}=(-\sqrt[10]{2})^{10}uu^{\mathsf T}
=2uu^{\mathsf T}=\begin{pmatrix}1&0&1\\0&0&0\\1&0&1\end{pmatrix}.
\]
最後に、直交行列 $P$ について $P^{-1}AP=D$ が実対角なら
$A=PDP^{\mathsf T}$ である。転置すると
\[
A^{\mathsf T}=(PDP^{\mathsf T})^{\mathsf T}=PD^{\mathsf T}P^{\mathsf T}=PDP^{\mathsf T}=A
\]
となるので $A$ は対称である。
\end{proof}

\begin{problem}{微分積分｜積分記号下の微分}{unknown-2}
$f\in C^1(D)$ とする。固定端点および可変端点の積分に対する Leibniz 則を示せ。また
\[
p(x)=\int_{x^2}^{x^3}\frac{\sin(xy)}y\,\d y\qquad(x>0)
\]
の導関数を求めよ。
\end{problem}
\begin{proof}[解答]
固定端点では差商と微積分学の基本定理から
$\frac{d}{dx}\int_c^df(x,y)dy=\int_c^df_x(x,y)dy$。可変端点では上下端の細い区間を分けると
\[
\frac d{dx}\int_{\phi(x)}^{\psi(x)}f(x,y)dy
=f(x,\psi)\psi'-f(x,\phi)\phi'+\int_\phi^\psi f_x(x,y)dy.
\]
この公式を使うため
\[
f(x,y)=\frac{\sin(xy)}y,\qquad
f_x(x,y)=\cos(xy),\qquad
\phi(x)=x^2,\quad\psi(x)=x^3
\]
とおくと
\begin{align*}
p'(x)
&=f(x,x^3)\,3x^2-f(x,x^2)\,2x
  +\int_{x^2}^{x^3}\cos(xy)\,\d y\\
&=\frac{3\sin(x^4)-2\sin(x^3)}x
  +\left[\frac{\sin(xy)}x\right]_{y=x^2}^{x^3}\\
&=\frac{3\sin(x^4)-2\sin(x^3)}x
  +\frac{\sin(x^4)-\sin(x^3)}x.
\end{align*}
従って
\[
p'(x)=\frac{4\sin(x^4)-3\sin(x^3)}x.
\]
\end{proof}

\begin{problem}{距離空間｜コンパクト集合の近傍}{unknown-3}
コンパクト集合 $K$ までの距離 $\rho(x)=\inf_{z\in K}d(x,z)$ について，$|\rho(x)-\rho(y)|\le d(x,y)$，各 $x$ で最短点 $a\in K$ が存在することを示せ。また $K\subset L$ で $L$ が開なら，ある $r>0$ で $\{\rho<r\}\subset L$ を示せ。
\end{problem}
\begin{proof}[解答]
任意の $z\in K$ に対し三角不等式から
\[
d(x,z)\le d(x,y)+d(y,z)
\]
である。$z\in K$ について右辺の下限を取ると
$\rho(x)\le d(x,y)+\rho(y)$、すなわち $\rho(x)-\rho(y)\le d(x,y)$ を得る。
$x,y$ を交換した式と合わせて
\[
|\rho(x)-\rho(y)|\le d(x,y)
\]
となる。

$z\mapsto d(x,z)$ は上の評価と同様に1-Lipschitz、従って $K$ 上連続である。
$K$ はコンパクトなので最小値を取る点 $a\in K$ が存在し
\[
d(x,a)=\min_{z\in K}d(x,z)=\rho(x).
\]

最後に $F=X\setminus L$ とおく。$F$ は閉で $K\cap F=\varnothing$ である。
$h(z)=d(z,F)$ は $K$ 上連続であり、各 $z\in K$ について $h(z)>0$ だから、
コンパクト性により
\[
\delta:=\min_{z\in K}d(z,F)>0.
\]
$0<r<\delta$ とする。$\rho(x)<r$ なら、最短点の存在により
$d(x,a)=\rho(x)<r$ となる $a\in K$ がある。もし $x\in F$ なら
$d(a,F)\le d(a,x)<r<\delta$ となり矛盾する。従って $x\in L$ であり、
$\{x\mid\rho(x)<r\}\subset L$ を得る。
\end{proof}

\begin{problem}{回転面｜トーラス}{unknown-4}
$R>r>0$ とし，円 $(x-R)^2+z^2=r^2$ を $z$ 軸の周りに回転したトーラス $S$ の Gauss 曲率と平均曲率を求め，測地線を一つ挙げよ。
\end{problem}
\begin{proof}[解答]
\[
X(u,v)=((R+r\cos v)\cos u,(R+r\cos v)\sin u,r\sin v)
\]
とおく。まず偏微分は
\[
X_u=(-(R+r\cos v)\sin u,(R+r\cos v)\cos u,0),
\]
\[
X_v=(-r\sin v\cos u,-r\sin v\sin u,r\cos v).
\]
従って第1基本形式の係数は
\[
E=(R+r\cos v)^2,\qquad F=0,\qquad G=r^2.
\]
$R>r$ より $R+r\cos v>0$ なので $EG-F^2>0$、従ってこれは正則な曲面である。
外向き単位法線を
\[
N=(\cos v\cos u,\cos v\sin u,\sin v)
\]
と取る。二階偏微分と内積を取ると
\[
L=X_{uu}\cdot N=-(R+r\cos v)\cos v,\quad
M=X_{uv}\cdot N=0,\quad
N_2=X_{vv}\cdot N=-r.
\]
$F=M=0$ なので座標方向が主方向であり、主曲率は
\[
k_1=\frac LE=-\frac{\cos v}{R+r\cos v},\qquad
k_2=\frac{N_2}G=-\frac1r.
\]
従って
\[
K=\frac{\cos v}{r(R+r\cos v)},\qquad
H=-\frac{R+2r\cos v}{2r(R+r\cos v)}.
\]
$v=0$ と固定した外側の赤道を $\gamma(u)=X(u,0)$ とおくと
\[
\gamma''(u)=-(R+r)(\cos u,\sin u,0)=-(R+r)N(u,0).
\]
加速度の接平面成分が0、すなわち測地的曲率が0なので、この円は測地線である。
\end{proof}

\begin{problem}{Euler 型微分方程式}{unknown-5}
$x^2y''+pxy'+qy=0$ $(x\ge1)$，$y(1)=0,y'(1)=1$ を考える。$q=0,p\ne1$ の解，$x=e^t$ による変形，$p=2$ の解を求める。さらに $p=2$ で $x>1$ の零点が存在する $q$ の範囲と最小の零点を求めよ。
\end{problem}
\begin{proof}[解答]
$q=0$ では $x^2y''+pxy'=0$ である。$z=y'$ とおくと
\[
xz'+pz=0,\qquad \frac{z'}z=-\frac px.
\]
従って $z=Cx^{-p}$。初期条件 $y'(1)=1$ から $C=1$ で、$y(1)=0$ を使って
\[
y(x)=\int_1^x s^{-p}\,\d s=\frac{x^{1-p}-1}{1-p}\qquad(p\ne1).
\]

次に $x=e^t$、$Y(t)=y(e^t)$ とおく。連鎖律から
\[
Y'(t)=e^ty'(e^t)=xy'(x),
\]
\[
Y''(t)=xy'(x)+x^2y''(x),
\]
従って $x^2y''=Y''-Y'$ である。元の方程式は
\[
Y''+(p-1)Y'+qY=0,\qquad Y(0)=0, Y'(0)=1.
\]
$p=2$ では特性方程式が $\lambda^2+\lambda+q=0$ となる。
$d=1-4q$ とおく。

$d>0$ のとき、特性根は $(-1\pm\sqrt d)/2$ なので
\[
Y(t)=e^{-t/2}\left(C_1e^{\sqrt d\,t/2}+C_2e^{-\sqrt d\,t/2}\right).
\]
$Y(0)=0$ から $C_2=-C_1$。従って
$Y(t)=2C_1e^{-t/2}\sinh(\sqrt d\,t/2)$ であり、
\[
Y'(0)=2C_1\cdot\frac{\sqrt d}{2}=C_1\sqrt d=1
\]
から $C_1=1/\sqrt d$ を得る。

$d=0$ のときは重根 $-1/2$ を持つので
$Y(t)=(C_1+C_2t)e^{-t/2}$。初期条件から $C_1=0$, $C_2=1$ である。

$d<0$ のとき、$b=\sqrt{-d}$ とおけば特性根は $(-1\pm ib)/2$ であり
\[
Y(t)=e^{-t/2}\left(C_1\cos\frac{bt}{2}+C_2\sin\frac{bt}{2}\right).
\]
$Y(0)=0$ から $C_1=0$、さらに
$Y'(0)=C_2b/2=1$ から $C_2=2/b$ となる。

最後に $t=\log x$、$e^{-t/2}=x^{-1/2}$ を戻すと
\[
y=x^{-1/2}\begin{cases}
\dfrac{2}{\sqrt d}\sinh(\frac{\sqrt d}{2}\log x),&d>0,\\
\log x,&d=0,\\
\dfrac{2}{\sqrt{-d}}\sin(\frac{\sqrt{-d}}2\log x),&d<0.
\end{cases}
\]
$d\ge0$ の場合、$x>1$ では $\log x>0$ であり、$\sinh u$ と $u$ は正なので零点を持たない。
$d<0$、すなわち $q>1/4$ の場合は
\[
\sin\left(\frac{\sqrt{4q-1}}2\log x\right)=0
\]
より、正の零点は
\[
\frac{\sqrt{4q-1}}2\log x=k\pi\qquad(k=1,2,\ldots).
\]
従って $x>1$ の零点があるのは $q>1/4$ のときに限り、最小の零点は
$\exp(2\pi/\sqrt{4q-1})$ である。
\end{proof}

\begin{problem}{解析｜弱い特異核}{unknown-6}
$0<\alpha<1$，$\phi$ は $[0,\infty)$ 上有界可測とし $f(x)=\int_0^x\phi(t)dt$，
$g(x)=\int_0^x f(s)(x-s)^{-\alpha}ds$ とする。$f$ の連続性，
\[
g(x)=\frac1{1-\alpha}\int_0^x\phi(t)(x-t)^{1-\alpha}dt
\]
を示し，$g'(x)=\int_0^x\phi(t)(x-t)^{-\alpha}dt$ を示せ。
\end{problem}
\begin{proof}[解答]
$|f(y)-f(x)|\le\norm\phi_\infty|y-x|$ なので $f$ は連続。$f(s)=\int_0^s\phi(t)dt$ を代入し，Fubini で積分順序を交換すると
\begin{align*}
g(x)
&=\int_0^x\int_0^s\phi(t)(x-s)^{-\alpha}\,\d t\d s\\
&=\int_0^x\phi(t)\int_t^x(x-s)^{-\alpha}\,\d s\d t.
\end{align*}
ここで $|\phi(t)|(x-s)^{-\alpha}$ は三角形 $0\le t\le s\le x$ 上可積分である。実際、
\[
\int_0^x\int_0^s\norm\phi_\infty(x-s)^{-\alpha}\,\d t\d s
=\norm\phi_\infty\int_0^x s(x-s)^{-\alpha}\,\d s<\infty
\]
なので Fubini の定理を適用できる。内側を $u=x-s$ で積分すると
\[
\int_t^x(x-s)^{-\alpha}\,\d s
=\int_0^{x-t}u^{-\alpha}\,\d u
=\frac{(x-t)^{1-\alpha}}{1-\alpha},
\]
従って指定された表示式を得る。

導関数の候補を
\[
G(x)=\int_0^x\phi(t)(x-t)^{-\alpha}\,\d t
\]
とおく。$0<\alpha<1$ なのでこの積分は絶対収束する。さらに Fubini の定理により
\begin{align*}
\int_0^xG(u)\,\d u
&=\int_0^x\int_0^u\phi(t)(u-t)^{-\alpha}\,\d t\d u\\
&=\int_0^x\phi(t)\int_t^x(u-t)^{-\alpha}\,\d u\d t\\
&=\frac1{1-\alpha}\int_0^x\phi(t)(x-t)^{1-\alpha}\,\d t\\
&=g(x).
\end{align*}
$G$ は局所連続である。実際、核 $u^{-\alpha}\mathbf1_{(0,L)}(u)$ は
$L^1(\R)$ に属し、その平行移動は $L^1$ で連続、$\phi$ は有界だからである。
従って微積分学の基本定理を適用でき
\[
g'(x)=G(x)=\int_0^x\phi(t)(x-t)^{-\alpha}\,\d t
\]
を得る。
\end{proof}
