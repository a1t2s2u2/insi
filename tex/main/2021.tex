\begin{problem}{第1問｜Jordan 標準形}{2021-1-1}
$a$ を複素数とし，行列
\[
A=\begin{pmatrix}a-1&a&0\\-a&-a-1&0\\-1&1&-1\end{pmatrix}
\]
を考える。
\begin{enumerate}
  \item[(1)] $A$ の固有多項式を求めよ。
  \item[(2)] $A$ の固有値を求め，対応する固有空間の次元を求めよ。
  \item[(3)] $A$ のジョルダン標準形を与えよ。（変換行列は求めなくてもよい。）
\end{enumerate}
\end{problem}
\begin{memo*}
\textbf{解答の見通し。}
行列 $A$ の対角成分には全て $-1$ が含まれているので、$N=A+I_3$ とおいて
$-I_3$ の部分を取り除く。実際に掛け算すると $N^3=O$ になる。
このように何乗かすると零行列になる行列を冪零行列という。
$Nx=\mu x$ なら $N^3x=\mu^3x=0$ なので、$N$ の固有値は0だけである。
従って $A=N-I_3$ の固有値は $-1$ だけになる。

Jordan ブロック $J_k(-1)$ に $I_k$ を足すと、上の対角線に1が並ぶ冪零ブロックになる。
その $k$ 乗で初めて0になるため、$N^2$ と $N^3$ を調べれば最大ブロックの大きさが分かる。
$a=0$ のときだけ $N^2$ が0になるので場合分けする。
\end{memo*}
\begin{proof}[解答]
$N=A+I_3$ とおくと
\[
N=\begin{pmatrix}a&a&0\\-a&-a&0\\-1&1&0\end{pmatrix},\qquad
N^2=\begin{pmatrix}0&0&0\\0&0&0\\-2a&-2a&0\end{pmatrix},\qquad
N^3=O.
\]
\begin{enumerate}
  \item[(1)] $N$ は冪零だから
  \[
  \det(\lambda I_3-A)=\det\{(\lambda+1)I_3-N\}=(\lambda+1)^3.
  \]

  \item[(2)] 固有値は $-1$ のみである。固有空間は $\ker N$ であり、
  \[
  a\ne0:\quad \ker N=\left\langle(0,0,1)^{\mathsf T}\right\rangle,
  \quad \dim\ker N=1,
  \]
  \[
  a=0:\quad
  \ker N=\left\langle(1,1,0)^{\mathsf T},(0,0,1)^{\mathsf T}\right\rangle,
  \quad \dim\ker N=2.
  \]

  \item[(3)] $a\ne0$ なら $N^2\ne O$, $N^3=O$ だから
  \[
  J=J_3(-1).
  \]
  $a=0$ なら $N\ne O$, $N^2=O$ かつ $\dim\ker N=2$ だから
  \[
  J=J_2(-1)\oplus(-1).
  \]
\end{enumerate}
\end{proof}

\begin{problem}{第2問｜二変数関数}{2021-1-2}
$\R^2$ 上の関数 $f(x,y)$ を次のように定義する。
\[
f(x,y)=\begin{cases}\dfrac{\sin(x^2y)}{x^2+y^2},&(x,y)\ne(0,0),\\0,&(x,y)=(0,0).
\end{cases}
\]
\begin{enumerate}
  \item[(1)] $x=r\cos\theta$，$y=r\sin\theta$ $(r>0,\ 0\le\theta<2\pi)$ に対して，$|f(x,y)|\le r$ および $|f(x,y)|\le\dfrac1{r^2}$ が成り立つことを示せ。
  \item[(2)] $f(x,y)$ は点 $(0,0)$ において連続であることを示せ。
  \item[(3)] $R>0$，$p>1$ に対して
  \[
  I_p(R)=\iint_{B(R)}|f(x,y)|^p\,\d x\d y
  \]
  とおく。ただし $B(R)=\{(x,y)\in\R^2\mid\sqrt{x^2+y^2}\le R\}$ である。このとき $\lim_{R\to\infty}I_p(R)$ が存在することを示せ。
  \item[(4)] $\dfrac{\partial f}{\partial y}(x,0)$ を求めよ。また，$f(x,y)$ は $C^1$ 級の関数かどうか，理由をつけて答えよ。
\end{enumerate}
\end{problem}
\begin{memo*}
\textbf{解答の見通し。}
極座標の半径 $r=\sqrt{x^2+y^2}$ は、点 $(x,y)$ と原点の距離であるため
$|x|\le r$, $|y|\le r$ が成り立つ。
$|\sin u|\le|u|$ を使うと $|f|\le r$ となり、原点へ近づくと0へ近づくことが分かる。
一方、$|\sin u|\le1$ を使うと $|f|\le r^{-2}$ となり、遠方での減衰が分かる。
半径1の内側と外側で二つの評価を使い分け、面積要素
$\d x\d y=r\,\d r\d\theta$ を掛けた一変数積分が収束するか確認する。
最後の偏導関数は公式を先に使わず、定義の差商を直接計算する。
\end{memo*}
\begin{proof}[解答]
\begin{enumerate}
  \item[(1)] $r=\sqrt{x^2+y^2}$ とすると $|x|,|y|\le r$ だから
  \[
  |f(x,y)|\le\frac{|x|^2|y|}{r^2}\le r
  \]
  である。また $|\sin u|\le1$ より $|f(x,y)|\le r^{-2}$。

  \item[(2)] (1) より
  \[
  |f(x,y)-f(0,0)|\le\sqrt{x^2+y^2}\longrightarrow0.
  \]
  従って原点で連続である。

  \item[(3)] $0<r\le1$ では $|f|^p\le r^p$、$r\ge1$ では
  $|f|^p\le r^{-2p}$ だから
  \[
  I_p(R)\le
  2\pi\int_0^1r^{p+1}\,\d r+
  2\pi\int_1^\infty r^{1-2p}\,\d r
  =\frac{2\pi}{p+2}+\frac{\pi}{p-1}<\infty.
  \]
  $I_p(R)$ は $R$ について単調増加かつ上に有界なので、
  $\lim_{R\to\infty}I_p(R)$ は存在する。

  \item[(4)] $x\ne0$ なら
  \[
  \frac{\partial f}{\partial y}(x,0)
  =\lim_{h\to0}\frac{\sin(x^2h)}{h(x^2+h^2)}
  =\lim_{h\to0}\frac{\sin(x^2h)}{x^2h}\frac{x^2}{x^2+h^2}=1.
  \]
  $x=0$ では差商は0だから
  \[
  \frac{\partial f}{\partial y}(x,0)=
  \begin{cases}1&(x\ne0),\\0&(x=0).\end{cases}
  \]
  この偏導関数は原点で不連続なので、$f$ は $C^1$ 級でない。
\end{enumerate}
\end{proof}

\begin{problem}{第3問｜連続像}{2021-1-3}
\begin{enumerate}
  \item[(1)] 集合 $X,Y,Z$，写像 $f:X\to Y$, $g:Y\to Z$ で，$f$ は全射ではない，$g$ は単射ではない，$g\circ f$ は全単射である，という条件を全て満たす例をあげよ。
  \item[(2)] $X,Y$ を位相空間とし，$f:X\to Y$ を連続写像とする。$X$ の連結な部分集合 $U$ に対して，$f(U)$ は連結であるか。連結であれば，そのことを示せ。そうでなければ，反例を挙げよ。
  \item[(3)] $X,Y$ を位相空間とし，$f:X\to Y$ を連続写像とする。$X$ のコンパクトな部分集合 $U$ に対して，$f(U)$ はコンパクトであるか。コンパクトであれば，そのことを示せ。そうでなければ，反例を挙げよ。
\end{enumerate}
\end{problem}
\begin{memo*}
\textbf{解答の見通し。}
(2) (3) はどちらも「連続写像は連結性とコンパクト性を保つ」という一つの事実である。
証明の骨格も共通で、$f(U)$ 側の情報を $f^{-1}$ で $U$ 側へ引き戻し、
$U$ の仮定（連結・コンパクト）を使ってから、再び $f$ で $f(U)$ 側へ戻す。

(2) は対偶で考える。$f(U)$ が二つに分かれると仮定すると、
その逆像で $U$ も二つに分かれてしまい、$U$ の連結性に反する。
(3) は $f(U)$ の開被覆を逆像へ戻すと $U$ の開被覆になり、
$U$ のコンパクト性で有限個に絞れる。その有限個を $f$ で写せばよい。

(1) は一元集合と二元集合で足りる。元を列挙して各条件を確かめられる。
\end{memo*}
\begin{proof}[解答]
\begin{enumerate}
  \item[(1)] $X=Z=\{0\}$, $Y=\{0,1\}$ とし
  \[
  f(0)=0,\qquad g(0)=g(1)=0
  \]
  とする。$f$ は全射でなく、$g$ は単射でないが、
  $g\circ f:\{0\}\to\{0\}$ は全単射である。

  \item[(2)] 連結である。$f(U)$ に非自明な開分離 $A\cup B$ があれば、制限写像を用いて
  \[
  U=(f|_U)^{-1}(A)\cup(f|_U)^{-1}(B)
  \]
  は $U$ の非自明な開分離となり、連結性に反する。

  \item[(3)] コンパクトである。$f(U)$ の開被覆 $\{V_\lambda\}$ に対し
  $\{f^{-1}(V_\lambda)\}$ は $U$ の開被覆である。有限部分被覆を取って像へ戻せば、
  対応する有限個の $V_\lambda$ が $f(U)$ を覆う。
\end{enumerate}
\end{proof}

\begin{problem}{第1問｜環とイデアル}{2021-s-1}
環の準同型 $\Phi:\mathbb C[X,Y,Z]\to\mathbb C[t]$ を，$f(X,Y,Z)\in\mathbb C[X,Y,Z]$ に対し
$\Phi(f(X,Y,Z))=f(t^2,t^3,t^4)$ で定める。$I=\ker\Phi$，$R=\mathbb C[X,Y,Z]/I$ と定める。
多項式 $f\in\mathbb C[X,Y,Z]$ が定める剰余類 $f+I$ を $\bar f$ で表す。
\begin{enumerate}
  \item[(1)] $I$ は素イデアルかどうか，理由も含め答えよ。
  \item[(2)] $I$ の生成元を求めよ。それらの元が $I$ を生成することも示せ。
  \item[(3)] $\bar X,\bar Y,\bar Z$ で生成される $R$ のイデアル $\mathfrak m$ が極大であることを示せ。
  \item[(4)] $\mathfrak m$ が2元で生成されることを示せ。
  \item[(5)] $\mathfrak m$ が単項イデアルでないことを示せ。
\end{enumerate}
\end{problem}
\begin{memo*}
\textbf{解答の見通し。}
準同型の像を $\mathbb C[t^2,t^3]$ と同定して商環を具体化する。
核ではまず $Z=X^2$ を消去し、次に $Y^2=X^3$ で割った余りを偶数次数・奇数次数に分ける。
極大性は原点での値写像、生成元の最小個数は
$\mathfrak m/\mathfrak m^2$ のベクトル空間次元で判定する。
\end{memo*}
\begin{proof}[解答]
\begin{enumerate}
  \item[(1)] $t^4=(t^2)^2$ だから
  \[
  \operatorname{Im}\Phi=\mathbb C[t^2,t^3,t^4]
  =\mathbb C[t^2,t^3].
  \]
  第1同型定理より
  \[
  R=\mathbb C[X,Y,Z]/\ker\Phi
  \cong\mathbb C[t^2,t^3]\subset\mathbb C[t].
  \]
  $\mathbb C[t]$ は整域なので、その部分環 $\mathbb C[t^2,t^3]$ も整域である。
  従って $I=\ker\Phi$ は素イデアルである。
  \item[(2)] 生成元は
  \[
  I=(Z-X^2,\ Y^2-X^3).
  \]
  実際、$\Phi(Z-X^2)=t^4-(t^2)^2=0$、
  $\Phi(Y^2-X^3)=t^6-(t^2)^3=0$ だから、右辺を $J$ とおけば $J\subset I$。

  逆に $f\in I$ とする。$Z-X^2$ を用いて $Z$ を消去すれば、ある
  $g(X,Y)\in\mathbb C[X,Y]$ により
  \[
  f\equiv g(X,Y)\pmod{(Z-X^2)}
  \]
  と書ける。さらに $g$ を $Y$ の多項式と見て、モニック多項式
  $Y^2-X^3$ で割ると
  \[
  g(X,Y)=q(X,Y)(Y^2-X^3)+a(X)Y+b(X)
  \]
  となる。$f\in I$ より
  \[
  0=\Phi(f)=a(t^2)t^3+b(t^2).
  \]
  $a(t^2)t^3$ は奇数次数の項だけ、$b(t^2)$ は偶数次数の項だけからなるので、
  この多項式恒等式から $a=b=0$。従って $f\in J$ であり、$I=J$ を得る。

  \item[(3)] $R\cong\mathbb C[t^2,t^3]$ のもとで
  $\mathfrak m=(t^2,t^3)$ に対応する。定数項を取る準同型
  \[
  \mathbb C[t^2,t^3]\longrightarrow\mathbb C,\qquad h(t)\longmapsto h(0)
  \]
  は全射で、核は $(t^2,t^3)$ である。従って
  $R/\mathfrak m\cong\mathbb C$ は体だから $\mathfrak m$ は極大である。

  \item[(4)] $\bar Z=\bar X^2$ だから
  \[
  \mathfrak m=(\bar X,\bar Y,\bar Z)
  =(\bar X,\bar Y,\bar X^2)=(\bar X,\bar Y).
  \]

  \item[(5)] $R\cong\mathbb C[t^2,t^3]$ で考えると
  \[
  \mathfrak m=(t^2,t^3),\qquad
  \mathfrak m^2=(t^4,t^5,t^6).
  \]
  従って $\mathfrak m/\mathfrak m^2$ では $t^2,t^3$ の剰余類が一次独立であり、
  これらが全体を生成するから
  \[
  \dim_{R/\mathfrak m}(\mathfrak m/\mathfrak m^2)
  =\dim_{\mathbb C}(\mathfrak m/\mathfrak m^2)=2.
  \]
  もし $\mathfrak m=(h)$ が単項なら、$\mathfrak m/\mathfrak m^2$ は
  $h+\mathfrak m^2$ 一つで生成される $R/\mathfrak m$ ベクトル空間なので次元は高々1となり、矛盾する。
\end{enumerate}
\end{proof}

\begin{problem}{第2問｜曲率・捩率}{2021-s-2}
曲線 $p:\R\to\R^3$ は
\[
p(t)=(\cos t,\ \sin t,\ \cosh t)\qquad(t\in\R)
\]
で定義されるとする。
\begin{enumerate}
  \item[(1)] 曲線 $p$ の曲率および捩率を求めよ。
  \item[(2)] 円柱面 $S=\{(x,y,z)\in\R^3\mid x^2+y^2=1\}$ の主方向と $\dfrac{\d p}{\d t}(t)$ の角度が $\dfrac\pi4$ であるような $t$ を全て求めよ。
  \item[(3)] 曲線 $p$ の曲面 $S$ における測地的曲率ベクトルを求めよ。
\end{enumerate}
\end{problem}
\begin{memo*}
\textbf{解答の見通し。}
(1) は $p',p'',p'''$ を曲率・捩率の公式へ代入する。
(2) では円柱の二つの主方向へ $p'$ を分解し、内積から角度条件を作る。
(3) は単位接ベクトルを弧長で微分して曲率ベクトルを求め、
そこから円柱の法線方向成分を引けば測地的曲率ベクトルになる。
\end{memo*}
\begin{proof}[解答]
以下 $p'=\d p/\d t$ などと略記する。

(1) まず
\[
p'=(-\sin t,\cos t,\sinh t),\quad
p''=(-\cos t,-\sin t,\cosh t),
\]
\[
p'''=(\sin t,-\cos t,\sinh t)
\]
であり
\[
p'\times p''=(\cos t\cosh t+\sin t\sinh t,
\sin t\cosh t-\cos t\sinh t,1).
\]
従って
\[
\norm{p'}=\cosh t,\qquad
\norm{p'\times p''}=\sqrt2\cosh t,\qquad
(p'\times p'')\cdot p'''=2\sinh t.
\]
曲率・捩率の公式に代入すると
\[
\kappa(t)=\frac{\sqrt2}{\cosh^2t},\qquad
\tau(t)=\frac{\sinh t}{\cosh^2t}.
\]
(2) 円柱面 $S$ の単位主方向を
\[
e_1=(-\sin t,\cos t,0),\qquad e_2=(0,0,1)
\]
とする。$p'=e_1+\sinh t\,e_2$ かつ $\norm{p'}=\cosh t$ なので、
$p'$ と $e_1$ のなす角が $\pi/4$ である条件は
\[
\frac{|p'\cdot e_1|}{\norm{p'}}=\frac1{\cosh t}=\frac1{\sqrt2}.
\]
$e_2$ との角についても $|\sinh t|/\cosh t=1/\sqrt2$ となり、同じ条件
$\cosh t=\sqrt2$ を与える。従って
$t=\pm\operatorname{arcosh}\sqrt2=\pm\log(1+\sqrt2)$。

(3) 単位接ベクトルは
\[
T=\frac{p'}{\norm{p'}}
=(-\operatorname{sech}t\sin t,
  \operatorname{sech}t\cos t,\tanh t).
\]
$\d s/\d t=\norm{p'}=\cosh t$ より $\d/\d s=\operatorname{sech}t\,\d/\d t$。
従って成分ごとに微分すると
\[
\frac{\d T}{\d s}
=\operatorname{sech}t\frac{\d T}{\d t}
\]
\[
=\bigl(
\operatorname{sech}^2t\tanh t\sin t-\operatorname{sech}^2t\cos t,
-\operatorname{sech}^2t\tanh t\cos t-\operatorname{sech}^2t\sin t,
\operatorname{sech}^3t
\bigr).
\]
円柱の単位法線は $n=(\cos t,\sin t,0)$ であり
\begin{align*}
\frac{\d T}{\d s}\cdot n
&=\operatorname{sech}^2t\tanh t\sin t\cos t
-\operatorname{sech}^2t\cos^2t\\
&\quad-\operatorname{sech}^2t\tanh t\cos t\sin t
-\operatorname{sech}^2t\sin^2t\\
&=-\operatorname{sech}^2t.
\end{align*}
従って法線成分を引けば
\[
\boldsymbol{k}_g
=\frac{\d T}{\d s}
-\left(\frac{\d T}{\d s}\cdot n\right)n
=\operatorname{sech}^2t\,
(\tanh t\sin t,-\tanh t\cos t,\operatorname{sech}t).
\]
\end{proof}

\begin{problem}{第3問｜収束べき級数}{2021-s-3}
複素数を係数とする収束べき級数
\[
\varphi(z)=\sum_{n=0}^\infty a_nz^n
\]
を考える。$a_0\ne0$ とする。
\begin{enumerate}
  \item[(1)] $1/\varphi(z)$ が $z=0$ で収束べき級数に展開できることを示せ。
  \item[(2)] $1/\varphi(z)$ の $z=0$ におけるべき級数展開の収束半径が $\varphi(z)$ の収束半径より真に小さくなることはあるか。あるとするならその理由を述べ，ないとするならそれを証明せよ。
  \item[(3)] $1/\varphi(z)=\sum_{n=0}^\infty b_nz^n$ とおくとき，$b_0,b_1,b_2$ を $a_n$ $(n=0,1,2,\ldots)$ を用いて表せ。
  \item[(4)] $n=0,1,2,3$ に対し $\displaystyle\int_{|z|=1}\frac{\d z}{z^n\sin^2z}$ を求めよ。
\end{enumerate}
\end{problem}
\begin{memo*}
\textbf{解答の見通し。}
$\varphi(0)\ne0$ なら逆数は原点近くで正則だが、その収束半径は
$\varphi$ 自身の収束半径ではなく、逆数が最初に出会う特異点で決まる。
係数は積が1になる条件から再帰的に求める。
積分では $1/\sin^2z$ を必要な次数まで Laurent 展開し、$z^{-1}$ の係数だけを読む。
\end{memo*}
\begin{proof}[解答]
\begin{enumerate}
\item[(1)] $\varphi(0)\ne0$ なので、連続性により0の十分小さい近傍で $\varphi(z)\ne0$。
従って $1/\varphi$ はその近傍で正則であり、Taylor 展開できる。
\item[(2)] ある。例えば $\varphi(z)=1-2z$ は整関数だが
$1/\varphi(z)=\sum_{n=0}^\infty(2z)^n$ の収束半径は $1/2$。
\item[(3)] 恒等式
\[
(a_0+a_1z+a_2z^2+\cdots)(b_0+b_1z+b_2z^2+\cdots)=1
\]
で $z^0,z^1,z^2$ の係数を比較すると
\[
a_0b_0=1,\quad a_0b_1+a_1b_0=0,\quad
a_0b_2+a_1b_1+a_2b_0=0.
\]
これを順に解いて
\[
b_0=\frac1{a_0},\quad b_1=-\frac{a_1}{a_0^2},\quad
b_2=\frac{a_1^2-a_0a_2}{a_0^3}.
\]
\item[(4)]
\[
\sin z=z-\frac{z^3}{6}+\frac{z^5}{120}+O(z^7)
\]
を二乗すると
\begin{align*}
\sin^2z
&=z^2-\frac13z^4
+\left(\frac1{36}+\frac1{60}\right)z^6+O(z^8)\\
&=z^2\left(1-\frac13z^2+\frac{2}{45}z^4+O(z^6)\right).
\end{align*}
$u=-z^2/3+2z^4/45+O(z^6)$ とおき
$(1+u)^{-1}=1-u+u^2+O(u^3)$ を使うと
\begin{align*}
\left(1-\frac13z^2+\frac{2}{45}z^4+O(z^6)\right)^{-1}
&=1+\frac13z^2
+\left(\frac19-\frac{2}{45}\right)z^4+O(z^6)\\
&=1+\frac13z^2+\frac1{15}z^4+O(z^6).
\end{align*}
従って
\[
\frac1{\sin^2z}=\frac1{z^2}+\frac13+\frac{z^2}{15}+O(z^4).
\]
従って
\[
\frac1{z^n\sin^2z}
=z^{-n-2}+\frac13z^{-n}+\frac1{15}z^{2-n}+O(z^{4-n}).
\]
$z^{-1}$ の係数、すなわち留数を $n=0,1,2,3$ の順に読むと
$0,1/3,0,1/15$。留数定理から積分値は順に
\[
0,\qquad \frac{2\pi i}{3},\qquad0,\qquad\frac{2\pi i}{15}.
\]
\end{enumerate}
\end{proof}

\begin{problem}{第4問｜測度収束}{2021-s-4}
$E\subset\R^d$ はルベーグ可測集合で $|E|<\infty$ とする。ここで，$|E|$ は $E$ のルベーグ測度を表す。
$\{f_n(x)\}$ $(n=1,2,\ldots)$ は $E$ 上ルベーグ可測関数列とする。
正数 $\varepsilon$ に対して $A_{n,\varepsilon}=\{x\in E\mid|f_n(x)|\ge\varepsilon\}$ $(n=1,2,\ldots)$ とおくとき，以下の問いに答えよ。
\begin{enumerate}
  \item[(1)] $E$ 上ルベーグ積分可能な関数 $g$ に対して $F=\{x\in E\mid|g(x)|\ge1\}$ とおくとき，$|F|\le\int_E|g(x)|\,\d x$ を示せ。
  \item[(2)] $\int_E|f_n|\to0$ なら全ての $\varepsilon>0$ で $|A_{n,\varepsilon}|\to0$ を示せ。
  \item[(3)] $f_n\to0$ がほとんど至る所なら $\displaystyle\int_E\frac{|f_n|}{1+|f_n|}\to0$ を示せ。
  \item[(4)] 全ての $\varepsilon>0$ で $|A_{n,\varepsilon}|\to0$ なら，ほとんど至る所で0に収束する部分列があることを示せ。
\end{enumerate}
\end{problem}
\begin{memo*}
\textbf{解答の見通し。}
(1), (2) は集合の定義関数を積分して Chebyshev 型の評価を得る。
(3) は被積分関数が1以下なので、有限測度という仮定が優関数を与える。
(4) は閾値を $1/k$、例外集合の測度を $2^{-k}$ とする部分列を選び、
例外集合へ無限回入る点が零集合であることを使う。
\end{memo*}
\begin{proof}[解答]
\begin{enumerate}
\item[(1)] $F=\{|g|\ge1\}$ 上では $1\le|g|$ だから、$E$ 上で
$\mathbf1_F\le|g|$ である。両辺を積分して
\[
|F|=\int_E\mathbf1_F\le\int_E|g|.
\]

\item[(2)] $A_{n,\varepsilon}$ 上では $|f_n|/\varepsilon\ge1$ なので
\[
\mathbf1_{A_{n,\varepsilon}}
\le\frac{|f_n|}{\varepsilon}.
\]
従って
\[
|A_{n,\varepsilon}|
\le\frac1\varepsilon\int_E|f_n|\longrightarrow0.
\]

\item[(3)]
\[
0\le\frac{|f_n(x)|}{1+|f_n(x)|}\le1
\]
であり、$f_n(x)\to0$ となる点では左辺も0へ収束する。
$|E|<\infty$ なので優関数 $\mathbf1_E$ は可積分であり、優収束定理から
\[
\int_E\frac{|f_n|}{1+|f_n|}\longrightarrow0.
\]

\item[(4)] 各 $k$ について $|A_{n,1/k}|\to0$ だから、$n_1<n_2<\cdots$ を
\[
|A_{n_k,1/k}|<2^{-k}
\]
となるよう帰納的に選べる。すると
\[
\sum_{k=1}^\infty|A_{n_k,1/k}|
\le\sum_{k=1}^\infty2^{-k}<\infty.
\]
Borel--Cantelli の補題より、ほとんど全ての $x$ は
$A_{n_k,1/k}$ に有限回しか属さない。従ってそのような $x$ では、十分大きい $k$ に対し
\[
|f_{n_k}(x)|<\frac1k,
\]
ゆえに $f_{n_k}(x)\to0$ である。
\end{enumerate}
\end{proof}

\section{第2期・専門基礎科目（3題必修）}

\begin{problem}{第1問｜等角ベクトルと対称行列}{2021-2-1}
次の問いに答えよ。
\begin{enumerate}
  \item[(1)] 3次実対称行列全体のなす線形空間の次元を求めよ。
  \item[(2)] $x,y\in\R^3$ に対し，$(x,y)$ を $x$ と $y$ の標準内積とする。$n$ を正の整数とする。
  $x_1,\ldots,x_n\in\R^3$ が条件
  \[
  |(x_i,x_j)|=\begin{cases}r&i\ne j\text{ のとき}\\1&i=j\text{ のとき}\end{cases}
  \]
  を満たすとする。ただし，$r$ は $r\ne1$ となる定数とする。このとき，実数 $a_i$ $(i=1,2,\ldots,n)$ が
  \[
  \sum_{i=1}^na_i(x_i,x_j)^2=0,\qquad j=1,2,\ldots,n
  \]
  を満たすならば，$a_1=a_2=\cdots=a_n=0$ が成り立つことを示せ。
  \item[(3)] (2) の $x_i$ に対し，行列 $x_ix_i^{\mathsf T}$ $(i=1,2,\ldots,n)$ は一次独立であることを示せ。ただし，$x_i^{\mathsf T}$ は $x_i$ を転置して得られる行ベクトルである。
  \item[(4)] (2) の $n$ に対し，$n\le6$ となることを示せ。
\end{enumerate}
\end{problem}
\begin{memo*}
\textbf{解答の見通し。}
目標の $n\le6$ を得るには、$n$ 個の対象を6次元の空間へ入れて一次独立だと示せばよい。
そこで各ベクトルから $3\times3$ 行列 $x_ix_i^{\mathsf T}$ を作る。
これは対称行列なので、(1) で数えた6次元の空間に住んでいる。

一次独立性は、一次関係 $\sum_ia_ix_ix_i^{\mathsf T}=O$ の両側から
$x_j^{\mathsf T}$ と $x_j$ を掛けると (2) のスカラー方程式に化けることから従う。
行列の等式を数の等式へ落とすのがこの一手である。

(2) 自体は、$j$ を固定した式で $i=j$ の項だけ係数が違うことに注目し、
$\sum_{i\ne j}a_i=\sum_ia_i-a_j$ と書き直す。すると $a_j$ の項と $j$ に依らない項に分かれ、
$a_j$ が $j$ によらないことが読み取れる。
\end{memo*}
\begin{proof}[解答]
\begin{enumerate}
  \item[(1)] 3次実対称行列は
  \[
  \begin{pmatrix}a&b&c\\b&d&e\\c&e&f\end{pmatrix}
  \]
  と一意に書けるので、次元は $6$。

  \item[(2)] $S=\sum_{i=1}^na_i$ とおく。$n=1$ なら与式から直ちに $a_1=0$。
  $n\ge2$ では $r\ge0$, $r\ne1$ だから $r^2\ne1$ であり、各 $j$ について
  \[
  0=a_j+r^2\sum_{i\ne j}a_i=(1-r^2)a_j+r^2S.
  \]
  従って全ての $a_j$ は同じ値 $a$ である。これを戻すと
  \[
  \{1+(n-1)r^2\}a=0,
  \]
  よって $a=0$。

  \item[(3)] $\sum_i a_ix_ix_i^{\mathsf T}=O$ とする。両側から
  $x_j^{\mathsf T}$ と $x_j$ を掛けると
  \[
  \sum_{i=1}^na_i(x_i,x_j)^2=0\qquad(j=1,\ldots,n).
  \]
  (2) より全ての $a_i$ は0だから、$x_ix_i^{\mathsf T}$ は一次独立である。

  \item[(4)] 各 $x_ix_i^{\mathsf T}$ は3次実対称行列である。
  (1), (3) より6次元空間内に一次独立な元が $n$ 個あるので、$n\le6$。
\end{enumerate}
\end{proof}

\begin{problem}{第2問｜Gamma 関数と Beta 関数}{2021-2-2}
$p>0$，$q>0$ とする。ガンマ関数 $\Gamma(p)$ とベータ関数 $B(p,q)$ は次のように定義される。
\[
\Gamma(p)=\int_0^\infty x^{p-1}e^{-x}\,\d x,
\qquad B(p,q)=\int_0^1x^{p-1}(1-x)^{q-1}\,\d x
\]
ガンマ関数とベータ関数の間には関係式
\[
B(p,q)=\frac{\Gamma(p)\Gamma(q)}{\Gamma(p+q)}
\]
が成り立つことを，
\[
\Gamma(p)\Gamma(q)=\iint_Dx^{p-1}y^{q-1}e^{-x-y}\,\d x\d y,
\qquad D=\{(x,y)\mid x>0,\ y>0\}
\tag{$*$}
\]
を利用して，以下の2通りの方法で示せ。
\begin{enumerate}
  \item[(1)] ($*$) において $x=st$，$y=s(1-t)$ と変数変換することにより，$\Gamma(p)\Gamma(q)=B(p,q)\Gamma(p+q)$ となることを示せ。
  \item[(2)] ($*$) において $x=(r\cos\theta)^2$，$y=(r\sin\theta)^2$ と変数変換することにより，$\Gamma(p)\Gamma(q)=B(p,q)\Gamma(p+q)$ となることを示せ。
\end{enumerate}
\end{problem}
\begin{memo*}
\textbf{解答の見通し。}
出発点の ($*$) は、二つの一変数積分の積を $D$ 上の二重積分として書いたものである。
被積分関数が非負なので、Tonelli の定理により積分順序を変えても値は変わらない。
変数変換では、面積が何倍になるかを表す Jacobian の絶対値を掛ける。

どちらの変換でも、狙いは被積分関数を「一方の変数だけの因子」と
「もう一方の変数だけの因子」の積にすることである。
そうなれば二重積分は二つの一変数積分の積に分かれ、
片方が $\Gamma(p+q)$、もう片方が $B(p,q)$ になる。

一つ目では $s=x+y$ が二数の和、$t=x/(x+y)$ が和に占める $x$ の割合であり、
変換後はそのまま分離する。
二つ目では動径 $r$ と角度 $\theta$ に分かれるが、
$\Gamma$ と $B$ の定義の形に戻すため $u=r^2$ と $t=\cos^2\theta$ を代入する。
\end{memo*}
\begin{proof}[解答]
\begin{enumerate}
  \item[(1)] $x=st$, $y=s(1-t)$ とおくと
  \[
  s>0,\quad 0<t<1,\qquad
  \left|\frac{\partial(x,y)}{\partial(s,t)}\right|=s.
  \]
  従って
  \begin{align*}
  \Gamma(p)\Gamma(q)
  &=\int_0^\infty\int_0^1
  s^{p+q-1}e^{-s}t^{p-1}(1-t)^{q-1}\,\d t\d s\\
  &=\Gamma(p+q)B(p,q).
  \end{align*}

  \item[(2)] $x=(r\cos\theta)^2$, $y=(r\sin\theta)^2$ とおくと
  \[
  r>0,\quad0<\theta<\frac{\pi}{2},\qquad
  \left|\frac{\partial(x,y)}{\partial(r,\theta)}\right|
  =4r^3\sin\theta\cos\theta.
  \]
  よって
  \begin{align*}
  \Gamma(p)\Gamma(q)
  &=4\left(\int_0^\infty r^{2p+2q-1}e^{-r^2}\,\d r\right)
  \left(\int_0^{\pi/2}\cos^{2p-1}\theta\sin^{2q-1}\theta\,\d\theta\right)\\
  &=4\left\{\frac12\Gamma(p+q)\right\}
       \left\{\frac12B(p,q)\right\}\\
  &=\Gamma(p+q)B(p,q).
  \end{align*}
  最後はそれぞれ $u=r^2$, $t=\cos^2\theta$ と置換した。
\end{enumerate}
従って、いずれの方法でも
\[
B(p,q)=\frac{\Gamma(p)\Gamma(q)}{\Gamma(p+q)}.
\]
\end{proof}

\begin{problem}{第3問｜距離空間の連続写像}{2021-2-3}
距離空間 $(X,d_X)$ から距離空間 $(Y,d_Y)$ への写像 $f:X\to Y$ について次の問いに答えよ。
\begin{enumerate}
  \item[(1)] $\delta>0$，$\varepsilon>0$，$x\in X$，$y\in Y$ に対して
  \[
  N_X(x;\delta)=\{u\in X\mid d_X(u,x)<\delta\},\qquad
  N_Y(y;\varepsilon)=\{v\in Y\mid d_Y(v,y)<\varepsilon\}
  \]
  とおく。これらの記号を用いて「写像 $f$ が点 $x\in X$ で連続である」ということの定義を述べよ。
  \item[(2)] $f$ が $X$ のすべての点で連続であれば，$Y$ の任意の閉集合 $F$ について $f^{-1}(F)$ が $X$ の閉集合であることを証明せよ。
  \item[(3)] $f$ が $X$ のすべての点で連続であれば，$X$ の任意の部分集合 $A$ に対して $f(\overline A)\subset\overline{f(A)}$ であることを証明せよ。ここで $\overline A$ は集合 $A$ の閉包を表すものとする。
\end{enumerate}
\end{problem}
\begin{memo*}
\textbf{解答の見通し。}
問題文の $N_X(x;\delta)$ は、$x$ からの距離が $\delta$ 未満の点全体、すなわち開球である。
連続とは、出力側で許す誤差 $\varepsilon$ を先に指定したとき、
入力側の十分小さい誤差 $\delta$ を選べることをいう。量化記号はこの順序で書く。
(2) は $x\notin f^{-1}(F)$ を一つ取り、その周りに逆像の補集合へ含まれる開球を作る。
(3) の $x\in\overline A$ は、$x$ のどんな開球にも $A$ の点があるという意味である。
連続性が与える $\delta$-球からその点を取り、像が任意の $\varepsilon$-球へ入ることを示す。
\end{memo*}
\begin{proof}[解答]
\begin{enumerate}
  \item[(1)] $f$ が $x$ で連続であるとは
  \[
  \forall\varepsilon>0\ \exists\delta>0,\qquad
  f(N_X(x;\delta))\subset N_Y(f(x);\varepsilon)
  \]
  が成り立つことである。

  \item[(2)] $F\subset Y$ を閉集合とする。$x\notin f^{-1}(F)$ なら、
  ある $\varepsilon>0$ で
  $N_Y(f(x);\varepsilon)\subset Y\setminus F$。
  連続性より、ある $\delta>0$ で
  \[
  N_X(x;\delta)\subset f^{-1}(Y\setminus F)
  =X\setminus f^{-1}(F).
  \]
  従って補集合は開であり、$f^{-1}(F)$ は閉である。

  \item[(3)] $x\in\overline A$ とする。任意の $\varepsilon>0$ に対し、
  連続性から
  $f(N_X(x;\delta))\subset N_Y(f(x);\varepsilon)$ となる $\delta>0$ を取る。
  $N_X(x;\delta)\cap A\ne\varnothing$ だから
  \[
  N_Y(f(x);\varepsilon)\cap f(A)\ne\varnothing.
  \]
  よって $f(x)\in\overline{f(A)}$、従って
  $f(\overline A)\subset\overline{f(A)}$。
\end{enumerate}
\end{proof}
